## 자동 재화 생성하기
<p align="Center">
<img src="../../_images/MSW-100_Defense_7_001.png" alt="MSW-100_Defense_7_001"
</p>

- **MainGame** 진입 시점 이후로 매 초마다 플레이어와 적(AI) 기지는 재화를 자동으로 생산하여 전투를 수행할 유닛을 생성할 수 있습니다.
- **StageData** DataSet에 기록된 스테이지 별 초당 생성량만큼 재화를 자동으로 생산합니다.
- BaseComponent에 선언된 **isUser** 변숫값을 활용해 필요한 재화 생산량의 데이터 값을 활용합니다.

```lua
BaseComponent의 AutoGetGold 함수 동작과정

Property:
-- 해당 기지가 유저(플레이어)의 것인지 판단하기 위한 상태 변수입니다.
[None]
boolean isUser
-- 초당 재화 생산량의 값을 저장하기 위한 변수입니다.
[None]
integer goldPerSecond
-- 일정 시간마다 재화를 생산하기 위해 측정하는 타이머입니다.
[None]
number goldTimer
-- 각 기지에 축적된 골드량을 기록하기 위한 변수입니다.
[None]
integer gold

Method:
[clientOnly]
void OnUpdate(number delta)
{
    -- 매 프레임마다 AutoGetGold 함수를 호출합니다.
    self:AutoGetGold(number delta)
}

[clientOnly]
void Init()
{
    -- 스테이지가 시작될 때 호출되는 Init 함수로 다음과 같이 스테이지에 맞는 재화 생산량을 설정합니다.

    self.goldTimer = 0
    self.gold = 0

    if isUser == true then
        self.goldPerSecond = _DataSetLogic.stageData[_StageLogic.currentStage].PlayerGoldPerSecond
    else
        self.goldPerSecond = _DataSetLogic.stageData[_StageLogic.currentStage].EnemyGoldPerSecond
    end
}

[clientOnly]
void AutoGetGold(number delta)
{
    -- 1초의 단위를 측정하기 위한 타이머값을 증가시킵니다.
    self.goldTimer += delta

    -- 현재 스테이지의 소지 가능한 최대 재화량 값을 기록합니다.
    -- ---@ type을 통해 해당 자료형이 정수임을 설정합니다.
    ---@ type integer
    local maxGold = _DataSetLogic.stageData[_StageLogic.currentStage].MaxGold

    -- 만약 1초 이상인 상태라면
    if self.goldTimer >= 1 and self.gold < maxGold then
        -- 타이머를 0으로 다시 초기화합니다.
        self.goldTimer = 0
        -- 설정된 초당 골드값을 현재 골드값에 추가합니다.
        self.gold += self.goldPerSecond
        self.gold = math.clamp(self.gold,0,maxGold)
    end
}
```

### DeckLoader 컴포넌트 제작하기
- MainGame 맵에 진입할때, 현재 유저의 설정된 덱 데이터를 기반으로 내부 덱 요소의 정보로 초기화하기 위한 컴포넌트입니다.
- DeckLoader 컴포넌트는 **DeckContainer**에 등록되어 활용됩니다.
- DeckLoader 컴포넌트의 코드는 다음과 같이 작성합니다.
- 해당 코드는 Plain Text 환경으로 작성되었습니다.

```lua
@Component
script DeckLoader extends Component

property string deckItemEntryID = "model://7ed4ff59-e40b-4ff9-a9bf-d22a306afb70" -- 유닛을 생산하기 위한 UI Deck의 Model EntryID 값입니다.

@ExecSpace("ClientOnly")
method void Init()
-- 여러개의 Deck들을 관리하는 DeckLoader를 초기화하는 함수입니다.

-- GameLogic에 등록된 유저 덱의 정보가 담긴 테이블을 가져옵니다.
---@type table
local userDecks = _GameLogic.userDeck
-- 이미 기존에 등록된 덱 개수를 확인합니다.
-- 게임에 처음 진입 시 덱이 구성되지 않았으므로 덱 UI 엔티티를 생성하기위함 입니다.
local currentDeckCount = #self.Entity.Children
-- 현재 필요한 덱 생산 수를 확인합니다.
local needDeckCount = #userDecks - currentDeckCount
-- 필요한 수량만큼 Deck을 생산하도록 합니다.
self:RegistDeckItem(needDeckCount)

-- 자식 엔티티(Deck)에 순차적으로 접근하여 각 덱을 초기화합니다.
for index, child in pairs(self.Entity.Children) do
	local deck = child.Deck
	-- 각 덱의 초기화는 GameLogic에 등록된 덱 테이블 내에 등록된 유닛 명으로 설정합니다.
	deck:Init(userDecks[index])
	-- MainGame내에서 활용될 덱이므로 기지에 덱의 정보에 등록된 유닛을 스폰할 수 있는 기능을 가진 'SpawnRequester' 컴포넌트를 부착합니다.
	deck.Entity:AddComponent("SpawnRequester")
end
end

@ExecSpace("ClientOnly")
method void RegistDeckItem(integer count)
-- 요청 수량만큼 덱을 등록하는 함수입니다.

-- 매개변수 1 count : 생산하고자 하는 덱의 개수입니다.

-- 필요 수량이 0보다 큰 경우에만 생산을 진행합니다.
if count > 0 then

	for i = 1, count, 1 do
		_SpawnService:SpawnByModelId(self.deckItemEntryID,"Deck",Vector3.zero,self.Entity)
	end
	
end
end

end
```

- 덱 UI 자체에 등록된 컴포넌트는 다음과 같습니다.
- SpriteGUIComponent : 덱 UI의 레이아웃을 표기하기 위한 컴포넌트입니다.
- Deck : 유닛으로써 UI로 표기되어야 하는 정보를 등록하기 위한 컴포넌트입니다.
- ButtonComponent : 버튼 입력을 통해 유저와 상호작용하기 위한 컴포넌트입니다.
- Deck 컴포넌트의 코드는 다음과 같이 작성합니다.
- 해당 코드는 Plain Text 환경으로 작성되었습니다.

```lua
@Component
script Deck extends Component

property string deckUnitName = "" -- 하나의 Deck에 등록된 유닛의 이름을 기록합니다.

@ExecSpace("ClientOnly")
method void Init(string unitName)
-- Deck을 초기화 하는 함수입니다.

-- 매개변수 1 : 덱에 담겨야 할 유닛의 이름을 전달 받습니다.

-- 내부 멤버로 유닛의 이름을 기록해둡니다.
self.deckUnitName = unitName

-- 해당 유닛의 모든 정보를 담는 테이블을 DataSetLogic으로부터 전부 가져옵니다.
---@type table
local unitData = _DataSetLogic.unitData[unitName]
-- 유닛의 외형을 결정하는 RenderID 값을 기록합니다.
local unitRenderID = unitData["RenderID"]
-- 대상 유닛의 IDLE 상태의 애니메이션 RUID 값을 기록합니다.
local unitIDLE_RUID = _DataSetLogic.unitRenderData[unitRenderID].IDLE
-- 대상 유닛의 생산 비용을 기록합니다.
local unitCost = unitData["Cost"]

-- 자식엔티티에서 유닛의 외형이 등록되는 SpriteGUIRenderer 컴포넌트를 가져옵니다.
local imageRenderer = self.Entity:GetChildByName("Image").SpriteGUIRendererComponent
-- 자식엔티티에서 유닛의 이름이 등록되는 Text 컴포넌트를 가져옵니다.
local nameText = self.Entity:GetChildByName("Name").TextComponent
-- 자식엔티티에서 유닛의 생산 비용이 등록되는 Text 컴포넌트를 가져옵니다.
local costText = self.Entity:GetChildByName("Cost").TextComponent

-- SpriteGUIRenderer 컴포넌트의 RUID 값을 유닛의 IDEL 리소스 값으로 설정합니다.
imageRenderer.ImageRUID = unitIDLE_RUID
-- 이름이 기입되는 Text 컴포넌트의 text를 유닛 명으로 설정합니다.
nameText.Text = unitName
-- 생산 비용이 기입되는 Text 컴포넌트의 text를 생산 비용 값으로 설정합니다.
costText.Text = unitCost

-- SpriteGUIRenderer 컴포넌트에는 유닛의 정적인 형태로만 출력하도록 하기위해
-- 프레임 재생속도를 0으로 설정합니다.
imageRenderer.PlayRate = 0
end

end
```

## 재화를 사용해 유닛 생성하기
- 유닛 생성 조건으로는 현재 축적된 재화의 값이 생성하고자 하는 유닛의 비용보다 큰 경우 생성이 가능합니다.
- 플레이어는 필드에 배치되는 유닛을 생산하기 위해서는 덱 UI에 등록된 ButtonComponent를 통해 실제 유닛을 생산할 수 있습니다.
- 적(AI)는 재화가 매초 축적되며 생산하고자 하는 유닛의 생산 비용만큼 축적될 경우 자동으로 유닛을 생산합니다.

- 유닛 자동생산 기능은 이전 단원 `4. 유닛 스폰 시스템 구현하기` 단원의 참고 자료를 확인합니다.
- 재화를 통한 적의 자동생산 기능은 다음과 같이 동작합니다.

### 플레이어의 유닛 생산 기능 제작하기

### Deck의 다양한 동작 구현 방식
<p align="Center">

- 디펜스 샘플월드에서는 유저에게 표시되는 Deck UI 요소에 필요한 컴포넌트를 실시간으로 등록하는 방식을 활용하고 있습니다.
- Deck 컴포넌트의 동작은 등록되는 유닛의 이름에 맞게 Deck UI 내부에 유닛 명, 외형, 비용 정보가 표기됩니다.
- Deck UI 엔티티에 TotalDeckSelector 컴포넌트가 추가될 경우 덱 편성에 필요한 기능을 추가적으로 수행하게 됩니다.
- Deck UI 엔티티에 SpawnRequester 컴포넌트가 추가될 경우 **MainGame** 맵에서 유저의 BaseComponent에게 Deck에 기록 중이던 유닛 생산을 요청하게 됩니다.
- Deck UI 엔티티에 PlayerDeckSelector 컴포넌트가 추가될 경우 덱 편성 시 유저의 실제 전투에 활용될 덱 교체 기능이 추가적으로 수행하게 됩니다.

<br>

### 유닛 스폰 요청하기
- SpawnRequester 컴포넌트 내에서 버튼 클릭 감지 시 호출되는 **HandleButtonClickEvent** 함수를 구현합니다.
- Deck 컴포넌트에서 기록 중인 유닛 이름을 확인하고 플레이어의 BaseComponent를 찾아 유닛 생성을 요청하게 됩니다.

```lua
SpawnRequester

Event Handler:
[self]
HandleButtonClickEvent(ButtonClickEvent event)
{
    -- 현재 Deck 컴포넌트에서 기록중인 이름을 기록합니다.
    local unitName = self.Entity.Deck.deckUnitName

    -- 플레이어의 baseComponent를 기록합니다.
    local playerBaseComponent = _EntityService:GetEntityByPath("/maps/MainGame/MyBase").BaseComponent
    -- Deck 버튼 클릭 시, 내부 멤버로 등록한 유닛 생산을 요청합니다.
    playerBaseComponent:PlayerRequestSpawn(unitName)
}

```

<br>

## 최소 구현 기준
- 유저가 덱 UI를 사용해 플레이어의 기지로부터 전투를 수행할 유닛이 생성되어야 합니다.
- 재화를 활용해 적 기지로부터 자동으로 유닛이 생성되어야 합니다.

<br>

## 응용 및 확장 방향
- 플레이어가 일정량의 재화를 지불하면 초당 재화 생산량이나 최대 보유 가능량을 영구적으로 늘려주는 '기지 업그레이드'버튼을 추가해 게임의 전략성을 높여볼 수 있습니다.
- 재화가 부족할 경우 덱 UI의 이미지를 어둡게 처리하거나 버튼 입력을 막아, 플레이어가 생산 불가 상태임을 직관적으로 인지할 수 있도록 발전시켜 봅니다.
- 적군 AI는 비용이 모이는 즉시 유닛을 생산하지만, 조건문을 확장하여 특정 타이밍까지 재화를 모았다가 여러 마리의 유닛을 한꺼번에 스폰하는 '웨이브'형태의 패턴을 구현해 볼 수 있습니다.
