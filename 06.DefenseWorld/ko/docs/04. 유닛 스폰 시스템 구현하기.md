## 해당 영상 시청 후 수행해 볼 내용
- UnitData의 'Usable' 항목의 정보를 'T' 또는 'F'값으로 변경해보며 덱 설정 환경에서 활용 가능한 유닛인지 직접 테스트해 봅니다.
- Base 엔티티의 자식으로 등록된 'SpawnLocation' 엔티티의 위치를 변경해 보며 해당 위치에 실제 유닛이 생성되는지 확인합니다.

## 플레이어의 사용 가능한 유닛 설정하기
<p align="Center">
<img src="../../_images/MSW-100_Defense_4_001.png" alt="MSW-100_Defense_4_001"
</p>

- 디펜스 샘플 월드는 유저가 전투로 활용할 수 있는 유닛을, DataSet을 통해 제한할 수 있습니다.
- 설정 옵션은 **UnitData** DataSet에 등록된 모든 유닛 목록 중 유저에게 제공하지 않도록 설정하기 위해선 **Usable**의 값을 **F**값으로 등록합니다.
- 이와 같은 기능을 제작하기 위해선 다음과 같은 코드를 작성할 수 있습니다.

```lua
void RegistUseableDecks()
{
    -- DataSet에 등록된 모든 유닛 데이터 테이블을 가져옵니다.
    local unitData = _DataSetLogic.unitData

    -- 테이블을 전체적으로 순회합니다.
    for unitName,unitData in pairs(unitData)
        -- 만약 해당 데이터의 'Useable'값이 'F'로 등록되어 있다면
        if unitData["Useable"] == "F" then
            -- 다음 순회로 넘어갑니다.
            continue
        end

        -- 값이 'T'일 경우 Deck을 생성하는 함수 호출을 진행합니다.
        RegistDeckItem(unitName)
    end
}
```

## 적(AI)이 생산하는 유닛 설정하기
<p align="Center">
<img src="../../_images/MSW-100_Defense_4_002.png" alt="MSW-100_Defense_4_002"
</p>

- 적의 기지로부터 생산할 수 있는 유닛은 **UnitData** DataSet에 등록된 모든 유닛을 등록해 활용할 수 있습니다.
- **StageData**에서는 UnitData의 Usable의 값과 상관없이 Unit들의 이름을 등록해 스테이지별 생산 가능한 유닛을 설정합니다.
- **StageData**의 맨 앞 **ID** 값은 각 스테이지 번호로 활용됩니다.
  - 2 스테이지에서 적이 생산가능한 유닛목록 : Panda, Pepe
- 적 유닛 목록을 테이블로 정리하는 과정은 다음 코드와 같이 작성할 수 있습니다.

```lua
Property:
Integer currentStage = 2

table GetEnemyUnitTable()
{
    -- 현재 스테이지에서 적이 생산 가능한 유닛 데이터를 가져옵니다.
    -- 전달받는 데이터는 String 문자열 형태입니다.
    local enemyUnitData = _DataSetLogic.stageData[currentStage].EnemyUnit -- string : "Panda,Pepe"

    -- UtilLogic의 Split 함수를 사용해 문자열을 분할해 테이블로 변환합니다.
    -- enemyUnitData("Panda,Pepe")를 콤마(,)값을 기준으로 분할하여 테이블로 저장합니다.
    -- table { "Panda" , "Pepe" }
    return _UtilLogic:Split(enemyUnitData,",")
} 
```

### Base의 스폰지점 생성하기
<p align="Center">
<img src="../../_images/MSW-100_Defense_4_003.png" alt="MSW-100_Defense_4_003"
</p>

- Player, Enemy의 각 기지로부터 생산되는 유닛의 스폰지점을 나타내는 자식 엔티티를 생성 후 배치해줍니다.

### 생산할 유닛 선정 과정

1. **StageData**에 등록된 EnemyUnit에 등록된 유닛 중 무작위 유닛을 선정합니다.
2. 선정된 유닛의 생산 조건(Cost)을 충족할 때까지 대기합니다.
3. 대상 유닛을 생산할 재화가 충분하다면, 유닛을 생산합니다.
4. 다음으로 생산할 유닛을 EnemyUnit 목록에서 다시 무작위로 선정합니다.
5. 2번의 과정부터 다시 반복합니다.

- 각 기지에 등록되는 BaseComponent의 코드는 다음과 같이 작성합니다.
- 해당 코드는 Plain Text 환경으로 작성되었습니다.
```lua
@Component
script BaseComponent extends Component

property boolean isUser = true -- 해당 기지가 유저의 소유인지 나타내는 변수입니다.

property integer hp = 0 -- 해당 기지의 현재 체력값을 나타냅니다.

property Entity spawnLocation = nil

property number gold = 0 -- 유닛을 생산할 수 있는 재화 값입니다.

property number goldTimer = 0 -- 일정 시간마다 재화를 생산하기 위해 측정하는 타이머입니다.

property number goldPerSecond = 0 -- 초당 몇 골드를 생산할지 결정하는 변수입니다.

property string nextSpawnUnitName = "" -- 다음으로 생산되어야 하는 유닛입니다.(Enemy 전용 변수)

property integer needGold = 0 -- 행동 시 필요한 재화값을 나타냅니다.

@ExecSpace("ClientOnly")
method void OnBeginPlay()
-- 월드 실행 시 바로 호출되는 함수입니다.

-- 자신의 초기화를 진행합니다.
self:Init()
end

@ExecSpace("ClientOnly")
method void Init()
-- 컴포넌트의 초기화를 수행합니다.

-- 자식엔티티 중 유닛 스폰 위치를 결정할 'SpawnLocation' 엔티티를 찾아 등록합니다.
self.spawnLocation = self.Entity:GetChildByName("SpawnLocation")

-- 현재의 재화값을 0으로 초기화합니다.
self.gold = 0
-- 재화를 생산하기 위한 타이머의 값을 0으로 설정합니다.
self.goldTimer = 0

-- 해당 기지가 유저의 것이라면,
if self.isUser == true then
	-- 초당 재화 생산량을 StageData의 PlayerGoldPerSecond값으로 설정합니다. (현재 스테이지 번호를 기준으로 합니다.)
	self.goldPerSecond = _DataSetLogic.stageData[_StageLogic.currentStage].PlayerGoldPerSecond
	-- 플레이어가 지켜야 하는 기지의 초기 체력값을 설정합니다.
	self.hp = _DataSetLogic.stageData[_StageLogic.currentStage].PlayerBaseHp
	
	_UI_MainGame:UpdateGoldText(self.gold)
	
-- 해당 기지가 적의 것이라면
else
	-- 초당 재화 생산량을 StageData의 EnemyGoldPerSecond값으로 설정합니다. (현재 스테이지 번호를 기준으로 합니다.) 
	self.goldPerSecond = _DataSetLogic.stageData[_StageLogic.currentStage].EnemyGoldPerSecond
	-- 다음으로 생산해야할 유닛을 결정합니다.
	self.nextSpawnUnitName = self:SelectRandomUnitForEnemyName()
	-- 적이 지켜야 하는 기지의 초기 체력값을 설정합니다.
	self.hp = _DataSetLogic.stageData[_StageLogic.currentStage].EnemyBaseHp
	-- 다음으로 생산해야할 유닛의 필요한 재화량 값을 기록합니다.
	self.needGold = _DataSetLogic.unitData[self.nextSpawnUnitName].Cost
end
end

@ExecSpace("ClientOnly")
method void SpawnUnit(string unitName)
-- 유닛을 생성하는 함수입니다.

-- 매개변수 1 unitName : 생산하는 유닛의 이름값을 전달 받습니다.

-- 오브젝트 풀로부터 유닛 엔티티를 반환받도록 요청합니다.
-- 전달받는 엔티티의 부모를 자신으로 설정하도록 매개변수를 전달합니다.
local unit = _ObjectPoolLogic:GetUnit(self.Entity)

-- 유닛 엔티티의 위치 좌표를 자식 엔티티인 SpawnLocaion의 위치와 동일하게 설정합니다.
unit.TransformComponent.Position = self.spawnLocation.TransformComponent.Position
-- Unit의 UnitComponet에 접근하여 초기화를 진행시킵니다.
unit.UnitComponent:Init(unitName,self.isUser)
end

@ExecSpace("ClientOnly")
method void OnDamage(integer damageValue)
-- 상대방 유닛으로 인한 피격시 호출되는 함수입니다.

-- 매개변수 1 damageValue : 상대방으로부터 얼마나의 피격 데미지를 전달받는지 결정합니다.

-- 체력값을 감소시킵니다.
self.hp -= damageValue

-- 체력이 0 이하라면
if self.hp <= 0 then
	-- 파괴 기능을 수행합니다.
	self:Destory()
	-- 스테이지의 진행이 종료되었음을 설정합니다.
	_StageLogic:SetStageState(false)
end
end

@ExecSpace("ClientOnly")
method void Destory()
-- 해당 엔티티의 파괴를 수행하는 함수입니다.

--파괴된 엔티티가 유저의 기지라면 패배라는 결과를 출력하는 팝업정보를 띄웁니다.
_UI_Result:ShowWithWinInfo(not self.isUser)

-- 컴포넌트가 부착된 엔티티를 비활성화 합니다.
self.Entity.Enable = false
end

@ExecSpace("ClientOnly")
method void OnUpdate(number delta)
-- 매 프레임마다 호출되는 함수입니다.

-- 스테이지가 종료되었다면
if _StageLogic.isStageStart == false then
	-- 이후의 기능을 수행하지 않고 종료합니다.
	return
end

-- 게임이 진행되는 동안 재화가 자동으로 축적되도록 합니다.
self:AutoGetGold(delta)
end

@ExecSpace("ClientOnly")
method void AutoGetGold(number delta)
-- 자동으로 재화를 축적하는 함수입니다.

-- 매개변수 1 delta : 프레임간 시간변화량의 값을 전달받습니다.

-- 1초의 단위를 측정하기 위한 타이머값을 증가시킵니다.
self.goldTimer += delta

-- 현재 스테이지의 소지 가능한 최대 재화량 값을 기록합니다.
-- ---@ type을 통해 해당 자료형이 정수임을 설정합니다.
---@ type integer
local maxGold = _DataSetLogic.stageData[_StageLogic.currentStage].MaxGold

-- 만약 1초 이상인 상태라면
if self.goldTimer >= 1 and self.gold < maxGold then
	-- 타이머를 0으로 다시 초기화합니다.
	self.goldTimer = 0
	-- 설정된 초당 골드값을 현재 골드값에 추가합니다.
	self.gold += self.goldPerSecond
	self.gold = math.clamp(self.gold,0,maxGold)
	
		-- 만약 해당 Base가 적의 기지라면
		if self.isUser == false then
			-- 자동 유닛 스폰함수를 동작하도록 합니다.
			self:EnemyAutoSpawn()
		-- 만약 해당 Base가 유저의 기지라면
	elseif self.isUser == true then
			-- 재화 정보를 나타내는 UI 텍스트의 값을 갱신하도록 합니다.
			_UI_MainGame:UpdateGoldText(self.gold)
		end
end
end

@ExecSpace("ClientOnly")
method string SelectRandomUnitForEnemyName()
-- StageData에 등록된 현재 스테이지에 맞는 EnemyUnit중 랜덤한 유닛 이름을 가져오는 함수입니다.

-- 유닛들의 이름을 담아둘 테이블을 생성합니다.
local enemyUnits = {} 
-- DataSetLogic으로부터 현재 스테이지에서 생성할 수 있는 유닛 이름 테이블을 그대로 가져옵니다.
enemyUnits = _DataSetLogic.stageData[_StageLogic.currentStage].EnemyUnit

-- 테이블의 1번요소부터 테이블의 끝 요소중 랜덤한 인덱스 값을 생성합니다.
local selectRandomIndex = _UtilLogic:RandomIntegerRange(1,#enemyUnits)

-- 해당 인덱스에 맞는 유닛의 이름을 가져옵니다.
local randomUnitName = enemyUnits[selectRandomIndex]

-- 해당 유닛의 이름을 반환합니다.
return randomUnitName
end

@ExecSpace("ClientOnly")
method void EnemyAutoSpawn()
-- 적 기지 전용 함수로, 자동으로 유닛을 생성합니다.

-- 만약 적 기지의 골드값이 다음 유닛생성에 필요한 골드값만큼 충족하다면
if self.gold >= self.needGold then
	-- 필요한 유닛을 스폰하도록 합니다.
	self:SpawnUnit(self.nextSpawnUnitName)
	-- 현재 재화에서 필요한 재화값을 감소시킵니다.
	self.gold -= self.needGold
	-- 다음으로 생성할 유닛을 다시 선택합니다.
	self.nextSpawnUnitName = self:SelectRandomUnitForEnemyName()
end
end

@ExecSpace("ClientOnly")
method void PlayerRequestSpawn(string unitName)
-- 매개변수로 전달받는 유닛을 스폰해달라고 요청하는 함수입니다.(플레이어 전용 함수)

-- 매개변수 1 unitName : 스폰하고자 하는 유닛의 유닛명

-- 대상 유닛의 생산 비용값을 확인합니다.
local targetUnitCost = _DataSetLogic.unitData[unitName].Cost

-- 현재 소지 골드값이 대상 유닛의 생산 비용보다 높다면
if self.gold >= targetUnitCost then
	-- 대상 유닛을 그대로 스폰하도록 요청합니다.
	self:SpawnUnit(unitName)
	-- 현재 재화량에서 유닛의 생산비용만큼 재화량을 감소시킵니다.
	self.gold -= targetUnitCost
	-- 재화량을 나타내는 TextUI를 소모된 현재 재화량으로 새로 갱신합니다.
	_UI_MainGame:UpdateGoldText(self.gold)
end
end

@ExecSpace("ClientOnly")
method void ReturnAllUnit()
-- 생산된 유닛을 전부 오브젝트 풀로 다시 반환하는 함수입니다.

-- 자식 엔티티중에서 UnitComponent가 등록된 엔티티들을 Table로 기록하도록 합니다.
local units = self.Entity:GetChildComponentsByTypeName("UnitComponent")

-- 테이블을 전부 순회합니다.
for _, unit in pairs(units) do
	-- 오브젝트 풀에 해당 유닛을 반환하도록 합니다.
	_ObjectPoolLogic:ReturnUnit(unit.Entity)
end
end

end
```

## 최소 구현 기준
- 적과의 전투를 진행하기 위해 적의 기지로부터 유닛이 자동으로 생산되어야 합니다.
- 최소 1개 이상의 플레이어, 적(AI)이 유닛을 생산할 수 있는 환경이 제작되어야 합니다.

## 응용 및 확장 방향
- 적 기지의 여러 스폰지점을 생성하여 하나의 스테이지에 다양한 상황을 연출할 수 있습니다.
