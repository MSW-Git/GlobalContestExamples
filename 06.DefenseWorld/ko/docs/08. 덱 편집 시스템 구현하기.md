## 덱 편집 기능 제작하기
<p align="Center">
<img src="../../_images/MSW-100_Defense_8_005.png" alt="MSW-100_Defense_8_005"
</p>

1. 덱 설정 기능의 핵심으로 사용할 유닛을 선택한 뒤, 편성할 플레이어 덱을 설정하는 기능이 담긴 DeckEditor 컴포넌트를 등록합니다.
2. 게임이 실행되는 도중 생성되는 덱 요소에 TotalDeckSelector 컴포넌트가 등록됩니다.
3. 게임이 실행되는 도중 생성되는 덱 요소에 PlayerDeckSelector 컴포넌트가 등록됩니다.

<br>

- DeckEditor 스크립트는 다음과 같이 작성해야 합니다.
- 해당 코드는 Plain Text 환경으로 작성되었습니다.

```lua
@Component
script DeckEditor extends Component

property table tempSelectedDecks = {} -- 임시 덱 테이블니다. 플레이어가 확인 버튼을 누를시 GameLogic의 덱 정보가 해당 테이블의 값으로 그대로 반영합니다.

property string seletedDeckUnitName = "" -- 현재 선택중인 덱의 유닛명을 기록하는 변수입니다.

property Entity totalDeckContentParent = "63ea1601-803c-4336-9ed4-4f26d4b45aad"

property Entity playerDeckContentParent = "0ae318bb-9902-4c7e-8823-b58d5f34d6c0"

property table playerDeckButtons = {} -- 유저의 덱 정보를 표기하는 팝업 내 버튼요소를 테이블에 기록합니다.

property Entity okButton = "7b9b83fd-d956-499a-ab69-6aca967a1cc8"

property Entity cancelButton = "43860e7a-219f-4c89-8247-6182c5a6533d"

@ExecSpace("ClientOnly")
method void OnBeginPlay()
-- 월드 실행 시 호출되는 함수입니다.



-- 덱 설정 팝업에서 확인 버튼 클릭시 SaveDeckData 함수가 호출되도록 이벤트를 등록합니다.
self.okButton:ConnectEvent(ButtonClickEvent,self.SaveDeckData)
-- 덱 설정 팝업에서 취소 버튼 클릭시 ClosePopup 함수가 호출되도록 이벤트를 등록합니다.
self.cancelButton:ConnectEvent(ButtonClickEvent,self.ClosePopup)

-- 유저가 사용할 수 있는 모든 덱 정보들을 등록합니다.
self:RegistTotalDeck()
wait(0.2)
-- 유저의 현재 덱 정보들을 등록합니다.
self:InitUserDeck()
end

@ExecSpace("ClientOnly")
method void InitUserDeck()
-- GameLogic으로부터 현재 유저의 덱 정보들을 가져옵니다.
self.tempSelectedDecks = _GameLogic.userDeck
-- DeckItem(프리팹)을 생성하기 위한 모델의 EntryID 값입니다.
local deckModelEntryID = "model://7ed4ff59-e40b-4ff9-a9bf-d22a306afb70"

-- 이미 기존에 등록된 덱 개수를 확인합니다.
-- 게임에 처음 진입 시 덱이 구성되지 않았으므로 덱 UI 엔티티를 생성하기위함 입니다.
local currentDeckCount = #self.playerDeckContentParent.Children
-- 현재 필요한 덱 생산 수를 확인합니다.
local needDeckCount = #self.tempSelectedDecks - currentDeckCount

-- 필요한 덱 생성 수만큼 순회를 진행합니다.
for i = 1, needDeckCount, 1 do
	
	-- SpawnService로 새로운 덱을 생성하여, 유저의 덱이 담기는 엔티티의 자식요소로 설정합니다.
	local newDeck = _SpawnService:SpawnByModelId(deckModelEntryID,"",Vector3.zero,self.playerDeckContentParent)
	-- 새로 생성된 덱의 정보를 대상 유닛의 정보로 설정합니다.
	newDeck.Deck:Init(self.tempSelectedDecks[i])
	-- 유저 팝업내에서 관리되는 덱으로써 'PlayerDeckSeletor' 컴포넌트를 추가 등록합니다.
	newDeck:AddComponent("PlayerDeckSelector")
	-- 해당 버튼이 몇 번째 유닛을 가리키고 있는지 인덱스 번호를 설정합니다.
	newDeck.PlayerDeckSelector.index = i
	-- 덱에 등록중인 Button 컴포넌트를 내부멤버인 playerDeckButtons에 추가합니다.
	table.insert(self.playerDeckButtons,newDeck.ButtonComponent)
end

-- 유저 덱의 모든 덱 추가가 완료되었으면 모든 버튼을 선택할 수 없도록 기능을 비활성화 합니다.
self:ActivePlayerDecks(false)
end

@ExecSpace("ClientOnly")
method void ActivePlayerDecks(boolean active)
-- 유저 덱을 표기하는 팝업 내 모든 덱의 버튼의 기능 활성화 여부를 설정하는 함수입니다.

-- 매개변수 1 active : 버튼들의 활성화 상태를 매개변수의 값 그대로 반영합니다.

-- 버튼 요소만큼 순회를 진행합니다.
for i = 1, #self.playerDeckButtons, 1 do
	
	-- 각 버튼요소를 확인합니다.
	---@type ButtonComponent
	local button = self.playerDeckButtons[i]
	
	-- 버튼의 기능 활성화 여부를 설정합니다.
	button.Enable = active
end
end

@ExecSpace("ClientOnly")
method void RegistTotalDeck()
-- 유저가 선택할 수 있는 덱 요소를 등록합니다.

-- 유닛의 정보가 담긴 DataSet Table을 그대로 가져옵니다.
local unitTable = _DataSetLogic.unitData
-- 덱의 모델 EntryID 값을 기록합니다.
local deckModelEntryID = "model://7ed4ff59-e40b-4ff9-a9bf-d22a306afb70"

-- 테이블 순회를 진행합니다.
for unitName,unitData in pairs(unitTable) do
	-- 유저가 사용할 수 있는지 확인하고 사용할 수 없는 덱이라면 다음 덱요소로 건너뜁니다.
	if unitData["Useable"] == "F" then
		continue
	end
	
	-- SpawnService를 통해 모든 덱의 요소에 새로운 덱을 생성합니다.
	local newDeck = _SpawnService:SpawnByModelId(deckModelEntryID,"",Vector3.zero,self.totalDeckContentParent)
	-- 덱을 해당 유닛의 이름으로 초기화합니다.
	newDeck.Deck:Init(unitName)
	-- 덱 편성을 위한 TotalDeckSelector 컴포넌트를 부착합니다.
	newDeck:AddComponent("TotalDeckSelector")
end
end

@ExecSpace("Client")
method void SaveDeckData()
-- 편성한 덱 요소를 기록하는 함수입니다.

-- 실제로 사용할 덱 데이터를 임시 테이블에 기록된 덱 정보로 그대로 반영합니다.
_GameLogic.userDeck = self.tempSelectedDecks
-- 덱 정보를 데이터 스토리지에 저장합니다.
_GameLogic:SaveUserDeck(_GameLogic.userDeck)
-- 덱 설정팝업을 종료합니다.
self:ClosePopup()
end

@ExecSpace("ClientOnly")
method void ClosePopup()
-- 덱 설정 팝업을 종료하는 함수입니다.

-- 자신의 엔티티를 비활성화 합니다.
self.Entity.Enable = false
end

end
```

<br>

- PlayerDeckSelector 스크립트는 다음과 같이 작성해야 합니다.
- 해당 코드는 Plain Text 환경으로 작성되었습니다.

```lua
@Component
script PlayerDeckSelector extends Component

property integer index = -1

property DeckEditor mainDeckEditor = "b887294b-9d5c-44fd-9c14-f830cbd749c3"

@EventSender("Self")
handler HandleButtonClickEvent(ButtonClickEvent event)
-- mainDeckEditor에 기록된 편성 대기중인 덱의 유닛이름을 기록합니다.
local deckUnitName = self.mainDeckEditor.seletedDeckUnitName

-- 현재 덱의 정보를 해당 유닛이름으로 초기화합니다.
self.Entity.Deck:Init(deckUnitName)
-- mainDeckEditor의 임시 덱 Table의 index 번째 요소를 현재 유닛명으로 교체합니다.
self.mainDeckEditor.tempSelectedDecks[self.index] = deckUnitName
-- 유저의 덱 교체가 완료되었으므로 다시, 유저의 덱 버튼을 비활성화 합니다.
self.mainDeckEditor:ActivePlayerDecks(false)
end

end
```

<br>

- TotalDeckSelector 스크립트는 다음과 같이 작성해야 합니다.
- 해당 코드는 Plain Text 환경으로 작성되었습니다.

```lua
@Component
script TotalDeckSelector extends Component

property DeckEditor mainDeckEditor = "b887294b-9d5c-44fd-9c14-f830cbd749c3"

@EventSender("Self")
handler HandleButtonClickEvent(ButtonClickEvent event)
-- 버튼 클릭시 호출되는 이벤트 함수입니다.

-- mainDeckEditor에서 기록중인 덱 정보들을 가져옵니다.
local currentDecks = self.mainDeckEditor.tempSelectedDecks
-- 현재 덱의 유닛명을 확인합니다.
local unitName = self.Entity.Deck.deckUnitName

-- 테이블의 총 개수만큼 순회를 진행합니다.
for i = 1, #currentDecks, 1 do
	-- 테이블 내에 이미 사용중인 덱이 있다면 함수 자체를 종료합니다.
	if currentDecks[i] == unitName then
		return
	end	
end

-- mainDeckEditor에 현재 선택중인 유닛이름을 현재 덱의 유닛명으로 설정합니다.
self.mainDeckEditor.seletedDeckUnitName = unitName
-- 편성할 유닛을 선택했으니, 유저의 덱을 선택할 차례이므로 유저의 덱의 버튼을 활성화 합니다.
self.mainDeckEditor:ActivePlayerDecks(true)
end

end
```

## 덱 저장 및 불러오기
- DataStroageService에서 제공되는 DataStorage 기능을 활용해 유저의 정보를 저장하거나 불러올 수 있습니다.
- 월드 입장 및 종료 시 데이터 저장 또는 불러오기 기능이 필요할 경우, 라이프 사이클 함수를 활용하는 방식보다 EventHandler(UserEnterEvent, UserLeaveEvent)활용을 권장합니다.

### DataStorage 기능을 활용한 덱 저장
- DataStorage에서 활용할 수 있는 SetAndWait 함수를 활용해 유저별 정보를 저장할 수 있습니다.
- 데이터 저장 시 String 자료형 변환 과정이 필요합니다.
- 서버의 데이터 저장 과정은 다음과 같이 진행될 수 있습니다.

```lua
Method:
[server]
void SaveDeck()
{
  local DeckString = "Tiger,Bear,Panda"
  local userKey = "TestUserKey"

  local userData = _DataStroageService:GetUserDataStroage(userKey)
  local errorCode = userData:SetAndWait("Deck",DeckString)

  if errorCode ~= 0 then
    log_error("Save deck error : "..errorCode)
  end
}
```

<br>

### DataStorage 기능을 활용한 덱 불러오기
- DataStorage에서 활용할 수 있는 GetAndWait 함수를 활용해 유저별 정보를 불러올 수 있습니다.
- GetAndWait의 첫 번째 반환 값은 에러가 발생될 경우 에러 코드값을 반환하며, 두 번째 매개변수로는 서버로부터 불러오는 데이터를 전달받습니다.
- 데이터를 불러올 경우 String 자료형으로 데이터를 전달받습니다.
- 서버의 데이터를 불러오는 과정은 다음과 같이 진행될 수 있습니다.

```lua
[server]
void LoadDeck()
{
  local userKey = "TestUserKey"
  local userStorage = _DataStorageService:GetUserDataStorage(userKey)
```

- 게임 내 유저 및 스테이지의 정보를 저장하는 기능을 포함하는 GameLogic의 스크립트는 다음과 같습니다.

```lua
@Logic
script GameLogic extends Logic

property table userDeck = nil

@Sync

local userData = _DataStorageService:GetUserDataStorage(self.playerStorageKey)
```

## 최소 구현 기준
- DataSet에 등록된 유닛 중 플레이어가 사용 가능한 유닛들만 전체 덱 설정 팝업(Contents)에 중복 없이 정상적으로 나열되어야 합니다.
- 전체 덱 목록에서 특정 유닛을 클릭한 후 플레이어 덱 슬롯을 클릭하면, 해당 슬롯의 유닛이 선택한 유닛으로 정상 교체되어야 합니다.
- 플레이어의 덱 내에 동일한 유닛이 2개 이상 중복으로 편성되지 않도록 예외 처리가 작동해야 합니다.
- 덱 편성 완료 후 DataStorageService를 통해 유저의 덱 정보가 서버에 문자열(String) 형태로 안전하게 저장되고, 재접속 시 정상적으로 로드되어야 합니다.

<br>

## 응용 및 확장 방향
- 각 유닛마다 배치 '비용(Cost)' 또는 '등급'을 부여하고, 플레이어 덱에 편성된 유닛들의 총 코스트 합이 일정 수치를 넘지 못하도록 제한하는 전략적 요소를 추가해 볼 수 있습니다.
- 전체 덱 목록에서 유닛을 선택했을 때, 현재 어떤 유닛이 선택되었는지 플레이어가 명확히 알 수 있도록 해당 유닛 UI에 테두리 하이라이트 연출이나 선택 이펙트를 추가하여 사용자 경험을 개선합니다.
- 단 하나의 덱만 사용하는 것을 넘어, '덱 A', '덱 B', '덱 C'처럼 여러 개의 덱 프리셋을 저장해 두고 스테이지의 특성이나 적의 속성에 맞춰 덱을 통째로 빠르게 스왑(Swap)할 수 있는 시스템으로 확장해 봅니다.
