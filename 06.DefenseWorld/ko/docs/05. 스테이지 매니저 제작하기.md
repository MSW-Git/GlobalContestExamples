## 디펜스 월드의 스테이지 관리하기
- 유저에게 다양한 스테이지를 제공하기 위해 각 스테이지 번호에 맞는 게임 환경이 구성되어야 합니다.
- 각 스테이지 번호, 스테이지 설정을 담당하는 중간 역할을 수행할 중간 관리자가 필요합니다.
- 프로그램 내 전역적으로 접근할 수 있는 **Logic** 스크립트를 활용해 **StageLogic**을 제작합니다. 

<br>

### StageLogic 제작하기
- StageLogic은 다음과 같은 역할을 수행합니다.
  - currentStage : 플레이어가 진행할 현재 스테이지 번호를 기록합니다.
  - StageReset : 본격적인 MainGame 내에서 현재 스테이지 번호에 맞는 게임 환경으로 초기화를 진행합니다.
  - OpenNextStage : 유저가 스테이지를 클리어하고 다음 스테이지로 넘어가는 기능을 수행합니다.
  - SetStageBackGround : 현재 스테이지의 배경화면을 설정합니다.

<br>

- 다음과 같이 코드를 작성해야 합니다.
- 해당 코드는 Plain Text 환경으로 작성되었습니다.
```lua
@Logic
script StageLogic extends Logic

property integer currentStage = 1 -- 현재 스테이지의 번호를 기록합니다.

property boolean isStageStart = false -- 현재 스테이지가 진행중인지 나타냅니다.

property string backGroundEntityPath = "/maps/MainGame/Background" -- BackGround 엔티티의 경로를 기록합니다.

@ExecSpace("ClientOnly")
method void StageReset()
-- 현재 스테이지를 처음부터 다시 시작하도록 설정하는 함수입니다.

-- 결과창 팝업을 비활성화 하도록 설정합니다.
_UI_Result:Hide()

-- 플레이어의 baseComponent를 저장합니다.
local myBase = _EntityService:GetEntityByPath("/maps/MainGame/MyBase").BaseComponent
-- 적의 baseComponent를 저장합니다.
local enemyBase = _EntityService:GetEntityByPath("/maps/MainGame/EnemyBase").BaseComponent

-- 플레이어의 기지 엔티티를 활성화 합니다.
myBase.Entity.Enable = true
-- 적의 기지 엔티티를 활성화 합니다.
enemyBase.Entity.Enable = true

-- 플레이어의 baseComponent로부터 생성된 유닛을 전부 다시 오브젝트 풀로 반환하도록 합니다.
myBase:ReturnAllUnit()
-- 적의 baseComponent로부터 생성된 유닛을 전부 다시 오브젝트 풀로 반환하도록 합니다.
enemyBase:ReturnAllUnit()

-- 플레이어의 baseComponent를 초기화 합니다.
myBase:Init()
-- 적의 baseComponent를 초기화 합니다.
enemyBase:Init()
-- 스테이지의 BackGround를 설정합니다.
self:SetStageBackGround()
-- 스테이지가 진행중인 상태로 설정합니다.
self:SetStageState(true)

-- 상단의 스테이지 번호 UI Text를 설정합니다.
_UI_MainGame:UpdateStageNumberText()
end

@ExecSpace("Client")
method void OpenNextStage()
-- 다음 스테이지로 전환하는 함수입니다.

-- 모든 스테이지의 수를 확인합니다.
local totalStageCount = #_DataSetLogic.stageData

-- 만약 해당 스테이지가 마지막 스테이지보다 낮은 스테이지라면
if totalStageCount > self.currentStage then
	-- 현재 스테이지의 번호를 1 증가합니다.
	self.currentStage += 1
	-- 데이터 스토리지에 유저의 마지막으로 클리어한 스테이지 번호를 기록합니다.
	_GameLogic:SaveLastStage(self.currentStage)
end

-- 스테이지 상태를 처음 상태로 다시 되돌립니다.
self:StageReset()
end

@ExecSpace("ClientOnly")
method void SetStageState(boolean isStart)
-- 스테이지가 진행중인 상태를 변경하는 함수입니다.

-- 매개변수 1 isStart : true 라면 스테이지가 진행중인 상태입니다.

-- 스테이지의 상태를 내부 멤버로 기록합니다.
self.isStageStart = isStart
end

@ExecSpace("ClientOnly")
method void SetStageBackGround()
-- 배경화면을 전환하는 함수입니다.

-- 내부 멤버로 기록된 BackGround를 관리하는 엔티티로 접근하여 BackgroundComponent에 접근합니다.
local backGroundComponent = _EntityService:GetEntityByPath(self.backGroundEntityPath).BackgroundComponent
-- 배경 요소를 현재 스테이지 번호에 맞는 배경으로 설정합니다.
backGroundComponent:ChangeBackgroundByTemplateRUID(_DataSetLogic.stageData[self.currentStage].BackGroundRUID)
end

end
```

<br>
<br>

<p align="Center">
<img src="../../_images/MSW-100_Defense_5_002.png" alt="MSW-100_Defense_5_002"
</p>

- 게임 화면 출력의 기준이 되는 캐릭터 엔티티를 **Title** 맵에서 **MainGame** 맵으로 이동시켜 본격적인 게임을 수행할 수 있습니다.
- 캐릭터 엔티티를 다른 맵으로 이동시키기 위해선 메이플스토리 월드에서 제공하는 기능인 **TeleportService**를 활용해야 합니다.
- 맵 전환을 위한 TeleportService는 다음과 같이 사용할 수 있습니다.
```lua
[server only]
void TeleportToMainGameMap(Entity playerEntity)
{
    -- Hierarchy에 등록된 MainGame 맵 엔티티의 경로를 등록합니다.
    string destinationPath = "/maps/MainGame"

    -- 함수의 매개변수로 전달받는 대상 엔티티와 경로값을 전달합니다. 
    -- 대상을 destinationPath에 배치된 엔티티의 자식으로 이동시킵니다.
    _TeleportService:TeleportToEntityPath(playerEntity,destinationPath)
}
```

### TeleportLogic 제작하기
- 디펜스 월드의 TeleportService를 활용한 TeleportLogic은 다음과 같이 제작합니다.
- 해당 코드는 Plain Text 환경으로 작성되었습니다.

```lua
@Logic
script TelePortLogic extends Logic

property string localPlayerName = "" -- 월드 실행 시 등장하는 플레이어 엔티티의 이름을 담기 위한 변수입니다.

@ExecSpace("Server")
method void SaveLocalPlayerNametoServer(string userName)
-- 서버측에 로컬플레이어의 엔티티명을 등록하는 함수입니다.

-- 매개변수 1 userName : 로컬 플레이어가 조작하는 캐릭터의 엔티티명을 전달받습니다.

-- 내부 멤버로 매개변수(플레이어 엔티티명)을 기록해 둡니다.
self.localPlayerName = userName
end

@ExecSpace("ServerOnly")
method void TelePort(string mapName)
-- Teleport기능을 수행하는 함수입니다.

-- 매개변수 1 mapName : 이동하고자 하는 맵 이름을 전달받습니다.

-- 내부 멤버로 등록한 대상 플레이어 엔티티를 찾아 기록합니다.
local playerEntity = _UserService:GetUserEntityByUserId(self.localPlayerName)

-- 이동하고자 하는 맵의 path값을 담기위한 변수를 선언합니다.
local mapPath = ""

-- 만약 매개변수의 값이 'Title' 이라면
if mapName == "Title" then
	-- mapPath의 값을 아래와 같이 설정합니다.
	mapPath = "/maps/Title"

-- 만약 매개변수의 값이 'MainGame' 이라면	
elseif mapName == "MainGame" then
	-- mapPath의 값을 아래와 같이 설정합니다.
	mapPath = "/maps/MainGame"

-- 잘못된 매개변수의 값을 전달받았다면
else
	-- 아래와 같은 에러값이 로그로 출력되도록 설정합니다.
	log_error("Invalid map name. Please check again.")
end

-- 대상 플레이어 엔티티와 맵 경로가 설정되었다면 TeleportService의 기능을 활용해 맵 전환을 수행합니다.
_TeleportService:TeleportToEntityPath(playerEntity,mapPath)
end

end
```

## HandleEntityMapChangedEvent를 활용한 맵 전환 시점 확인하기
<p align="Center">
<img src="../../_images/MSW-100_Defense_5_003.png" alt="MSW-100_Defense_5_003"
</p>

- 본격적인 스테이지를 진행하기 위해선 MainGame 맵으로 전환된 시점 이후에 각 스테이지를 초기화하는 **StageLogic**의 **StageReset** 함수가 호출되어야 합니다.
- Teleport 기능을 수행함과 동시에 StageReset을 호출하게 될 경우 동기화 시점이 일치하지 않는 문제가 발생될 수 있습니다.
  1. 텔레포트를 수행하는 시점과 동시에 **StageReset** 이 진행될 경우 **MainGame** 맵 내에 등록된 엔티티, 참조를 연결하는 작업 수행 시 문제가 발생될 수 있습니다.
  2. 텔레포트 작업을 요청한 후 맵이 완전히 전환된 시점 이후에 **StageReset** 함수를 호출하여 이와 같은 문제를 방지할 수 있습니다.

## ButtonComponent의 클릭 이벤트 등록하기
<p align="Center">
<img src="../../_images/MSW-100_Defense_5_005.gif" alt="MSW-100_Defense_5_005"
</p>

- Title 맵에서 본격적인 게임으로 진입하기 위해선, 플레이어가 `게임 시작`을 누를 경우 실제로 맵을 전환하는 상호작용이 필요합니다.
- 월드 실행 시 호출되는 **OnBeginPlay** 함수 내에서 ButtonComponent의 Event 기능을 활용해 **MainGame**맵으로 전환하는 기능을 미리 등록해 둡니다. 

```lua
UI_Title Logic에서 버튼의 기능을 연동하는 과정

Property:
-- 버튼 엔티티를 변수로 할당하며 미리 참조를 연결해 둡니다.
[None]
Entity gameStartButton = /ui/Title/GameStartButton

Method:
[clientOnly]
void OnBeginPlay()
{
  -- Entity의 ConnectEvent함수를 활용해 대상 엔티티의 버튼클릭 이벤트가 발생될 때, EnterMainGame 함수를 호출하도록 연결합니다.
  self.gameStartButton:ConnectEvent(ButtonClickEvent,self.EnterMainGame)
}

[server]
void EnterMainGame()
{
  _TelePortLogic:TelePort("MainGame")
}
```

<br>

## 최소 구현 기준
- Title 화면에서 '게임 시작' 버튼 클릭 시 에러 없이 MainGame 맵으로 텔레포트되어야 합니다.
- 스테이지 클리어 시 currentStage가 1 증가하며, 다음 스테이지 데이터(배경, 적 정보)에 맞춰 게임이 온전히 재시작되어야 합니다.

<br>

## 응용 및 확장 방향
- currentStage가 5의 배수일 때 특수한 배경과 강력한 보스 몬스터가 등장하도록 전용 로직을 추가해 봅니다.
- 텔레포트 시 화면이 끊기는 느낌을 없애기 위해, 화면을 덮는 검은색 UI의 투명도를 조절하는 부드러운 씬 전환 연출을 추가해 봅니다.
- 마지막 스테이지 클리어 시 게임을 끝내지 않고, 적의 능력치가 끝없이 상승하는 무한 모드로 진입하도록 난이도 공식을 확장해 봅니다.
