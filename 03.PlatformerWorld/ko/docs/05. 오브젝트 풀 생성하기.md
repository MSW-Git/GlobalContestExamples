# [💡기초 학습]오브젝트 풀 생성하기
<br>
<br>
<br>

## 영상 타임라인
- 강의 영상 내 아래의 타임라인에서 학습할 수 있습니다.
- 오브젝트 풀 생성하기: 00:44:52
<br>

## 학습 목표
- 엔티티(오브젝트) 생성과 파괴가 반복될 때 발생하는 성능 저하의 원인을 파악하고, 이를 해결하기 위한 오브젝트 풀링 기법의 필요성을 이해합니다.
- 게임 시작 시 필요한 맵 청크나 장애물 엔티티를 미리 생성하여 대기 상태로 보관하는 풀 자료구조를 직접 구현할 수 있습니다.
- 화면 밖으로 벗어난 엔티티를 삭제하지 않고 비활성화하여 풀에 반환하고, 필요할 때 위치를 재설정하여 다시 꺼내 쓰는 재사용 메서드를 작성합니다.
<br>

## 해당 영상 시청 후 수행해 볼 내용
- 로직 스크립트를 활용하여 게임 초기화 단계에서 지정된 수량만큼의 타일셋 엔티티를 미리 스폰한 뒤, 비활성화 상태로 보관하는 코드를 작성해 봅니다.
- 스크롤되어 화면 왼쪽 끝의 DeadZone에 닿은 타일이 삭제되지 않고, 속성이 초기화된 채 다시 오브젝트 풀로 돌아가 대기 상태가 되는지 테스트해 봅니다.
- 절차적 맵 생성 시 매번 새로운 엔티티를 만드는 대신, 오브젝트 풀에서 대기 중인 엔티티를 꺼내어 이전 타일의 끝 좌표에 정상적으로 이어 붙이는지 확인해 봅니다.
<br>

## 오브젝트 풀이란?
- 게임을 진행하는 데 있어 수많은 엔티티가 사용됩니다.
- 이러한 엔티티 중, 빈번하게 스폰되는 총알이나 발판 같은 요소는 일정 시간이 지나면 화면에서 사라져 소멸해야 합니다. 
- 하지만 생성(Spawn)과 소멸(Destroy) 작업은 메모리 할당과 해제를 동반하므로, 이 과정이 짧은 시간 동안 반복될 경우 컴퓨터 연산에 큰 부하(프레임 드랍)를 유발합니다.
- 하지만 미리 엔티티를 생성하고, 소멸시키는 대신 활성화만 해제한다면 컴퓨터의 부하를 줄일 수 있습니다.
- 스위치를 켜고 끄듯 상태만 전환하여 메모리를 재사용하기 때문입니다.
- 즉, **오브젝트 풀(Object Pool)**은 미리 만들어 둔 엔티티들을 '수영장(Pool)'에 담아두고 필요할 때마다 꺼내 쓰며 반환하는 방식으로 불필요한 생성과 소멸을 최소화하는 핵심 최적화 기법입니다.
<br>

## 오브젝트 풀 제작하기
- 샘플 월드의 오브젝트 풀은 GameLogic이나 DataSetLogic처럼 어디서든 전역적으로 접근하여 쉽게 엔티티를 대여하고 반환받을 수 있는 중앙 관리(Manager) 구조로 제작됩니다.
- Workspace에서 새로운 로직을 생성하고 ObjectPoolLogic으로 작성합니다.

## ObjectPoolLogic 스크립트 작성하기
- 다음과 같이 코드를 작성해야 합니다.
- 해당 코드는 Plain Text 환경으로 작성되었습니다.

```lua
@Logic
script ObjectPoolLogic extends Logic

property number discript = 0 -- 생성과 파괴가 빈번하게 일어나는 오브젝트들을 효율적으로 관리하기 위한 로직입니다.

property table tilePool = {} -- 타일셋 오브젝트를 관리하기 위한 풀 역할을 수행하는 테이블입니다.

property table itemPool = {} -- 아이템 오브젝트를 관리하기 위한 풀 역할을 수행하는 테이블입니다.

property Entity ItemPoolParent = "9e9e98bd-4f7b-46ed-9a93-f0e7d74fac1c"

property table itemDataSetIDTable = {["Coin"] = 1 ,["Health"] = 2 ,["Booster"] = 3} -- key : 아이템명 , value : 데이터셋의 Index 번호를 저장하기 위한 테이블입니다.

@ExecSpace("ClientOnly")
method void ReturnTile(Entity entity)
-- 타일 오브젝트를 오브젝트 풀로 반환하는 메서드입니다.

-- 매개변수 1 entity : 반환(비활성)하고자 하는 오브젝트를 전달받습니다.
-- 매개변수 2 tileNumber : 타일 번호를 전달받습니다.

-- 타일셋 엔티티의 번호를 추출합니다. (테이블 내 번호별로 타일셋 오브젝트를 관리하기 위함입니다.)
local tileNumber = self:GetTileNumber(entity)

-- 타일 번호에 해당하는 풀(테이블)이 유효한지 검사합니다.
if isvalid(self.tilePool[tileNumber]) == false or self.tilePool[tileNumber] == nil then
	-- 테이블이 없었다면 새로운 테이블 공간을 새로 생성합니다.
	self.tilePool[tileNumber] = {}
end

-- 대상 테이블에 대상 엔티티(오브젝트)를 저장합니다.
table.insert(self.tilePool[tileNumber],entity)
-- 엔티티(오브젝트)를 비활성화합니다.
entity.Enable = false
end

@ExecSpace("ClientOnly")
method Entity GetTile(number tileNumber)
-- 타일 풀로부터 지정된 번호 타일 엔티티 반환받는 메서드입니다.

-- 매개변수 1 number tileNumber : 특정 타일셋 번호를 전달받습니다.

-- math.floor메서드를 활용해 tileNumber의 소수점 부분을 버리기 위함입니다. ex) 4.0 => 4
tileNumber = math.floor(tileNumber)

-- 대상 풀(테이블) 유효성 검사를 진행합니다.

-- 전달받은 tileNumber 타일 오브젝트를 관리하는 풀(테이블)이 없다면
if isvalid(self.tilePool[tileNumber]) == false or self.tilePool[tileNumber] == nil then
	-- 새로운 테이블 공간을 새로 생성합니다.
	self.tilePool[tileNumber] = {}
end

-- /분기1/ 없으면 새로 생성해서 반환합니다.

-- 만약 대상 풀에 담긴 오브젝트가 더 이상 없다면 (모델(프리팹)으로 부터 새로 오브젝트를 생성을 진행합니다.)
if #self.tilePool[tileNumber] <= 0 then
	
	-- 테이블 셋에 저장된 EntryID를 확인합니다.
	local entryID = _DataSetLogic.tileSetInfoTable[tileNumber].EntryID
	-- 사전에 미리 저장된(프리팹) 이름의 형태로 이름을 저장합니다.
	local name = "Model_TileSet0"..tileNumber
	
	-- 임시 위치에 대상 모델(프리팹) 오브젝트를 생성합니다.
	local tempTile = _SpawnService:SpawnByModelId(entryID,name,Vector3.up*100,_EntityService:GetEntity("d24b5917-1821-4743-ae07-77cc9bfc2d0a"))
	
	-- 해당 오브젝트를 반환합니다.
	return tempTile
end

-- /분기2/ 있었다면 마지막 요소를 반환합니다.

-- 현재 대상 테이블 수를 확인합니다.
local tableCount = #self.tilePool[tileNumber]

-- 대상 테이블의 마지막 요소를 반환받으며, 해당 엔티티를 테이블에서 제거합니다.
local tempTile = table.remove(self.tilePool[tileNumber],tableCount)

-- 선택한 오브젝트를 활성화합니다.
tempTile.Enable = true

-- 오브젝트를 반환합니다.
return tempTile


end

@ExecSpace("ClientOnly")
method Entity GetItem(string itemName)
-- 오브젝트 풀로부터 아이템 오브젝트를 반환받는 메서드입니다.

-- 매개변수 1 string itemName : 아이템의 이름을 전달받습니다.

-- 아이템 풀에서 대상 아이템들이 담긴 테이블을 가져옵니다.
local targetTable = self.itemPool[itemName]

-- 테이블 내부 요소가 비어있을때
if #targetTable == 0 then
	-- 대상 아이템을 config에서 설정한 생성 개수만큼 추가 생성을 진행합니다.
	self:CreateItem(itemName,_Config.itemPoolInitCount)
end


-- 가져오고자 하는 테이블로부터 엔티티를 불러옵니다.(테이블의 마지막요소)
---@type Entity
local targetEntity = targetTable[#targetTable]

-- 엔티티를 저장하였으므로 테이블의 마지막 요소를 제거합니다.
table.remove(targetTable,#targetTable)

-- 엔티티(오브젝트)를 활성화합니다.
targetEntity.Enable = true

-- 저장했던 엔티티를 반환합니다.
return targetEntity
end

@ExecSpace("ClientOnly")
method void ReturnItem(Entity entity)
-- 아이템 오브젝트를 다시 오브젝트 풀로 반환하는 메서드입니다.

-- 매개변수 1 Entity entity : 반환하고자 하는 아이템 오브젝트 엔티티를 전달합니다.

-- 반환하는 아이템의 이름을 확인합니다.
local itemName = entity.Item.name
-- 대상 테이블로 엔티티를 집어넣습니다.
table.insert(self.itemPool[itemName],entity)

-- 타일셋의 기존 Parent(타일셋의 아이템을 배치하기 위해 사전에 등록한 오브젝트)으로부터 
-- 다시 ItemPool 로 Parent를 이전합니다.
entity:AttachTo(self.ItemPoolParent)
-- 엔티티(오브젝트)를 비활성화합니다.
entity.Enable = false
end

@ExecSpace("ClientOnly")
method void CreateItem(string itemName, number createCount)
-- 모델(프리팹)으로부터 아이템 오브젝트를 새로 생성하여 대상 풀로 집어넣는 메서드입니다.

-- 매개변수 1 string itemName : 생성하고자 하는 아이템 이름을 전달받습니다.
-- 매개변수 2 number createCount : 생성할 개수, 오브젝트 풀로 몇개의 오브젝트를 생성해 넣어둘지를 결정합니다.

-- 임시 테이블을 선언합니다.
local tempTable

-- 테이터셋에서 참조할 Id 번호를 설정합니다.
local dataSetID = self.itemDataSetIDTable[itemName]

-- 데이터셋의 대상 아이템 모델의 EntryID 값을 불러옵니다.
local entryID = _DataSetLogic.itemInfoTable[dataSetID].EntryID
-- 데이터셋의 대상 아이템 모델의 Name 값을 불러옵니다.
local name = _DataSetLogic.itemInfoTable[dataSetID].Name
-- 데이터셋의 대상 아이템 모델의 Value 값을 불러옵니다.
local value = _DataSetLogic.itemInfoTable[dataSetID].Value

-- 대상 테이블의 유효성 검사를 진행합니다.

-- /분기 1/ 대상 아이템 풀 내에 대상 풀(테이블) 이 없을 경우
if isvalid(self.itemPool[name]) == false or self.itemPool[name] == nil then
	-- 새로운 임시 테이블을 생성합니다.
	tempTable = {}
	-- 아이템 풀에서 내부 풀(테이블)을 임시테이블로 설정합니다.
	self.itemPool[name] = tempTable
	
-- /분기 2/ 대상 아이템 풀 내에 대상 풀이 이미 있을 경우
else
	-- 임시 테이블(풀) 을 기존에 활용하고 있던 테이블로 선언합니다.
	tempTable = self.itemPool[name]
end

-- 매개변수2 createCount 만큼의 반복문을 수행합니다.
for i = 1 , createCount, 1 do 
	-- SpawnService를 활용해 오브젝트를 생성합니다.
	local entity = _SpawnService:SpawnByModelId(entryID,name,Vector3.up * 100, self.ItemPoolParent)
	
	-- 생성한 오브젝트의 설정작업을 진행합니다.
	
	-- 생성한 오브젝트의 아이템의 고윳값을 설정합니다.
	entity.Item.value = value
	-- 생성한 오브젝트의 아이템의 name 변수를 설정합니다.
	entity.Item.name = name
	
	-- 담겨야하는 테이블(풀)로 생성된 엔티티를 삽입합니다.
	table.insert(tempTable,entity)
	
	-- 생성된 엔티티 오브젝트를 비활성화합니다.
	entity.Enable = false
end
end

@ExecSpace("ClientOnly")
method void InitItemPool()
-- 아이템 풀을 초기화하는 메서드입니다.

-- DataSetLogic 으로부터 아이템 데이터셋이 담긴 테이블을 불러옵니다.
local itemDataTable = _DataSetLogic.itemInfoTable

-- Coin 아이템을 Config에서 설정한 생성개수만큼 생성합니다.
self:CreateItem("Coin",_Config.itemPoolInitCount)
-- Health 아이템을 Config에서 설정한 생성개수만큼 생성합니다.
self:CreateItem("Health",_Config.itemPoolInitCount)
-- Booster 아이템을 Config에서 설정한 생성개수만큼 생성합니다.
self:CreateItem("Booster",_Config.itemPoolInitCount)
end

@ExecSpace("ClientOnly")
method void Init()
-- ObjectPool Logic의 초기화를 수행하는 메서드입니다.

-- 아이템 풀 초기화를 진행합니다.
self:InitItemPool()
end

@ExecSpace("ClientOnly")
method number GetTileNumber(Entity entity)
-- 타일 번호를 추출하는 메서드입니다.

-- 매개변수 1 Entity entity : 대상 엔티티를 전달받습니다.

-- 엔티티의 이름을 추출합니다.
local tileSetName = entity.Name
-- 이름의 마지막 문자열을 추출합니다. ('Model_TileSet02' 문자열에서 14,15번째 문자 추출 > 02)
local numberStr = string.sub(tileSetName,14,15)
-- 문자열을 숫자값으로 변환해 반환합니다.
return tonumber(numberStr)
end

end
```

- ObjectPoolLogic은 주기적으로 생성되며 파괴되는 바닥 타일셋, 각 아이템 요소를 관리합니다.
- 작성된 GetTile 메서드는 타일 풀에서 대기 중인 바닥 타일을 꺼내오는 기능을 수행하며, ScrollingController 컴포넌트에서 새로운 맵 조각을 스폰해야 할 때 호출됩니다.
- ReturnTile 메서드는 화면 밖으로 지나간 바닥 타일이 DeadZone과 충돌했을 때, 해당 타일을 파괴하는 대신 풀로 반환(비활성화)하기 위해 TileDeadZone 스크립트에서 사용됩니다.
<br>

## 최소 구현 기준
- 맵 스크롤 진행 시 타일셋 엔티티가 반복적으로 생성 및 소멸(Destroy)되지 않고, 오브젝트 풀 시스템을 통해 정상적으로 비활성화 및 재활성화되며 순환하도록 구현합니다.
- 모든 타일셋은 오브젝트 풀에서 생성 또는 사용 가능한 엔티티를 반환하는 형태로 구현합니다.
<br>

## 응용 및 확장 방향
- 오브젝트 풀의 원칙인 **미리 생성**하여 사용할 수 있도록 코드를 수정합니다.
- 수정된 코드를 바탕으로 바닥 타일뿐만 아니라 게임 내에 빈번하게 등장하는 장애물이나 아이템 요소들도 오브젝트 풀을 통해 정상적으로 재사용되는지 확인합니다.
- 무조건 많은 양을 초기에 생성해 두는 것이 아니라, 오브젝트 풀 내의 대기 엔티티가 부족할 때만 유동적으로 풀의 크기를 늘려(추가 생성) 메모리를 더욱 최적화하는 확장 로직을 구현해 봅니다.
