## 장애물 배치 방식

<img src = "https://develrocket-bucket.s3.ap-northeast-2.amazonaws.com/learning/MSW-100/Running/MSW-100_Running_9_003.png" alt="MSW-100_Running_8_005.png">

- 장애물을 코드로 위치를 제어하기에는 어려움이 있으므로 장애물 엔티티를 미리 배치해두는 방식으로 제작을 진행합니다.
- 플레이어가 장애물을 회피할 수 있는지 테스트를 진행해 보며 배치 및 수정 작업을 진행합니다.
- 장애물 엔티티 관리 방식이 원활하도록, 장애물들의 부모 엔티티를 준비합니다.
<br>

### 무작위로 장애물 배치하기
- 만약 준비한 장애물을 타일마다 항상 배치한다면 게임이 단조로워질 수 있으므로, 난수(Random) 기능을 활용해 장애물이 불규칙한 패턴으로 등장하도록 구현합니다.
- 미리 배치된 장애물을 무작위로 배치하는 기능은 바닥 타일셋에 등록된 TileSet 컴포넌트 스크립트에서 타일셋이 활성화될 때 진행됩니다.
- 무작위 장애물 배치 기능은 다음 코드와 같이 처리할 수 있습니다.

```lua
TileSet 스크립트 내 SpawnObstacle메서드

-- 장애물 엔티티들의 부모 엔티티입니다.
Entity obstacleParent

void SpawnObstacle()
{
    -- 타일셋 내에 사전에 미리 설치한 장애물 오브젝트 수를 확인합니다.
    -- obstacle들을 보관한 부모 엔티티의 자식(장애물엔티티) 개수를 확인합니다.
    local spawnLocationCount = self.obstacleParent.Children.Count

    -- obstacleParent 오브젝트의 자식 오브젝트를 순회하며
    for index , child in pairs(self.obstacleParent.Children) do
        
        -- 0부터 1까지 무작위 숫자를 생성합니다.(구현 의도 : 0 일 경우 장애물 배치 O , 1 일 경우 장애물 배치 X 로 50% 확률)
        local randomNumber = math.random(0,1)
        
        -- /분기1/ 만약 숫자가 0 이라면
        if randomNumber == 0 then
            -- 장애물 오브젝트를 활성화합니다.
            child.Enable = true
            -- 스테이지 번호에 맞는 리소스 이미지로 덮어 씌웁니다.
            child.SpriteRendererComponent.SpriteRUID = _DataSetLogic.stageRenderInfoTable[_GameLogic.currentStage].ObstacleRUID
            
        
        -- /분기2/ 만약 숫자가 0이 아니라면
        else
            -- 장애물 오브젝트를 비활성화합니다.
            child.Enable = false
        end	
    end
}

```

## 장애물 피격처리
<p align = "Center">
<img src = "https://develrocket-bucket.s3.ap-northeast-2.amazonaws.com/learning/MSW-100/Running/MSW-100_Running_9_004.png" alt="MSW-100_Running_8_005.png">
</p>

- 캐릭터의 TriggerComponent와 장애물의 TriggerComponent가 정상적으로 등록되어 있다면, 상호 간 Trigger 이벤트를 활용할 수 있습니다.
- 캐릭터의 기능 및 동작을 직접 제어하기 위해 제작된 CustomPlayer 스크립트 내에서 HandleTriggerEnterEvent 이벤트 메서드를 활용합니다.
- CustomPlayer코드의 내용이 필요한 경우 '6.플레이어 캐릭터 구현하기' 단원에 첨부된 CustomPlayer스크립트 내용이 첨부되어 있습니다.
- 충돌 처리 기능은 다음 코드와 같이 처리할 수 있습니다.

```lua
CustomPlayer의 이벤트 핸들러를 활용한 충돌처리

HandleTriggerEnterEvent(TriggerEnterEvent event)
{
    -- 충돌된 대상의 엔티티 정보를 저장합니다.
    local TriggerBodyEntity = event.TriggerBodyEntity

    -- 대상 엔티티의 CollisionGroup에 설정된 충돌체 그룹 ID 번호를 추출합니다.
    local TriggerGroup = tostring(TriggerBodyEntity.TriggerComponent.CollisionGroup.Id)

    -- 해당 충돌체의 그룹명이 "Obstacle"과 동일하다면
    if _Config.triggerGroup[TriggerGroup] == "Obstacle" then
        -- 충돌 판정 처리를 수행합니다.
        self:ContactObstacle()
    end
}

void ContactObstacle()
{
    -- GameLogic에게 Config에 설정된 장애물 피격 데미지값 만큼 체력을 감소시키도록 합니다.
    _GameLogic:AddHP(_Config.obstacleDamage)
}

```

## 캐릭터 무적 기능 추가하기
<p align = "Center">
<img src = "https://develrocket-bucket.s3.ap-northeast-2.amazonaws.com/learning/MSW-100/Running/MSW-100_Running_9_005.gif" alt="MSW-100_Running_8_005.gif">
</p>

- 일정 무적 시간이 주어지지 않으면서 장애물의 피격처리를 HandleTriggerStayEvent 이벤트 메서드로 제작할 경우, 매 프레임마다 데미지를 입게 되어 순식간에 체력이 0이 되는 현상이 발생될 수 있습니다.
- 장애물에 부딪힌 직후 짧은 시간 동안 추가 피해를 막아줌으로써, 플레이어가 상황을 인지하고 자세를 재정비할 수 있는 여유 시간을 제공합니다.
- 일정시간 피격 무시 기능은 다음 코드와 같이 처리할 수 있습니다.

```lua
CustomPlayer의 invincibleTimer 활용

-- 일시 무적 타이머 변수를 선언합니다.
number invincibleTimer
-- 피격 상태임을 나타내는 변수를 선언합니다.
boolean isHit

void ContactObstacle()
{
    -- 만약 일시 무적 타이머 변수의 값이 0보다 크다면(일시 무적 타이머가 동작 중인 상황)
    if self.invincibleTimer > 0 then
    -- 장애물로 인한 피격처리를 수행하지 않고 메서드를 종료합니다.
        return
    end

    -- 위의 조건문이 수행되지 않았다면 (일시 무적 타이머가 동작 중이지 않은 상황)

    -- 체력을 감소시킵니다.
    _GameLogic:AddHP(_Config.obstacleDamage)
    
    -- 피격 상태를 true로 전환합니다.
    self.isHit = true;

    -- 플레이어에게 무적 상태임을 인지할 수 있도록 캐릭터 외형의 알파값을 절반으로 감소시켜 반투명 상태로 설정합니다.
    self.Entity.AvatarRendererComponent:SetAlpha(0.5)

    -- 일시 무적 타이머가 동작을 수행할 수 있도록 설정할 무적시간 값을 새로 갱신합니다.
    self.invincibleTimer = _Config.invincibleTime
}

void OnUpdate(number delta)
{
    -- isHit 값을 활용해 isHit 상황이 아닐때는 아래의 코드가 수행되지 않도록 방지합니다.
    if self.isHit == true then

        -- 일시 무적 타이머의 값이 0보다 큰 상황이라면,
        if self.invincibleTimer > 0 then
        -- 타이머의 값을 delta값만큼 감소시킵니다. 
            self.invincibleTimer -= delta
        
        -- 일시 무적 타이머가 남은 지속시간이 없다면,
        else
        -- 플레이어에게 무적상황이 끝났음을 인지할 수 있도록 캐릭터 외형의 알파값을 원래 값(1)으로 증가시켜 불투명 상태로 설정합니다.
            self.Entity.AvatarRendererComponent:SetAlpha(1);
        end
    end
}

```

## 최소 구현 기준
- 장애물 엔티티가 바닥 타일과 동일하게 이동되어야 합니다.
- 플레이어 캐릭터와 장애물 충돌 시 피격 판정이 발생해야 하며, 장애물로 인해 캐릭터가 뒤로 밀리지 않아야 합니다.
- 장애물로 인한 피격 시 일시 무적 기능이 활성화되어야 하며 타이머가 동작하는 동안에는 데미지를 입지 않아야 합니다.
<br>

## 응용 및 확장 방향
- 고정된 외형의 장애물뿐만 아니라, 제자리에서 위아래로 솟아오르는 동적인 애니메이션을 가진 엑티브 장애물을 추가할 수 있습니다.
- 게임의 진행도(시간의 흐름)에 따라 장애물 생성 확률을 높이거나 다르게 설정하여 게임의 난이도를 조절할 수 있습니다.
