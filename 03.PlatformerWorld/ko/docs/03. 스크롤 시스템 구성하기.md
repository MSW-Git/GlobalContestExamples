## 학습 목표
- 캐릭터가 계속 달리는 듯한 연출을 위해 배경과 타일을 반대 방향으로 이동시키는 스크롤 시스템의 핵심 원리를 이해합니다.
- OnUpdate메서드와 delta시간을 활용하여 프레임에 구애받지 않고 시간당 일정한 속도로 엔티티를 부드럽게 이동시키는 스크립트를 작성합니다.
- 배경화면과 바닥 타일의 속도를 제어해 게임의 난이도를 설정할 수 있습니다.
- 게임의 진행에 따라 스테이지 환경이 동적으로 어떻게 변경되는지 이해합니다.

## 스크롤 시스템의 기본 구조 및 원리 이해
- 러닝 플랫포머 게임에서 캐릭터가 끝없이 오른쪽으로 달려가게 만들면, 맵을 무한정 길게 제작해야 하는 한계에 부딪히게 됩니다.
- 캐릭터의 좌표값이 무한히 커질 경우 시스템상 치명적인 오류가 발생할 수 있습니다.
- 물리적 한계를 극복하고 무한한 맵을 안정적으로 연출하기 위해 스크롤 시스템의 도입이 필요합니다.

## ScrollingController 제작하기
<p align="Center">
<img src ="https://develrocket-bucket.s3.ap-northeast-2.amazonaws.com/learning/MSW-100/Running/MSW-100_Running_3_002.png" alt="MSW-100_Running_3_002.png">
</p>

- 빈 엔티티에 ScrollingController 컴포넌트를 등록합니다.
  - ScrollingController는 게임의 전반적인 스크롤 환경을 관리하는 주체입니다.
  - 생성된 엔티티의 TransformComponent의 Position 값을 (0,0,0)으로 설정합니다.
- 빈 엔티티를 생성하여 ScrollingController컴포넌트가 부착된 엔티티의 자식 엔티티로 설정합니다.
  - 해당 엔티티는 타일셋 전체를 이동시키기 위한 부모 엔티티 역할을 수행합니다.
  - 생성된 엔티티의 TransformComponent의 position 값을 (0,0,0)으로 설정해 부모와 동일선상에 위치시킵니다.
<br>

### ScrollingController Component 구성하기
- 배경, 바닥 타일을 제어하기 위한 코드를 작성해야 합니다.
- 다음과 같이 코드를 작성합니다.
- 해당 코드는 Plain Text를 기준으로 작성되어 있습니다.

```lua
@Component
script ScrollingController extends Component

property string disscript = "" -- 러닝 플랫포머 게임 내에 횡이동을 전반적으로 담당하기 위한 로직입니다.

property Entity tileParent = nil

property number scrollDist = -1 -- 현재 스크롤된 거리를 저장하는 변수입니다. // config에 설정된 타일셋 하나당 거리는 10

property number lastSpawnedTileNumber = -1 -- 가장 마지막으로 생성된 타일번호입니다.

property number spawnedCount = -1 -- 몇개의 타일이 스폰되었는지에 대한 기록용 변수입니다.

property BackgroundComponent background = nil

property number currentScrollSpeed = 0 -- 현재 스크롤링 속도값을 저장하기 위한 변수입니다.

property number currentBackgroundScrollSpeed = 0 -- 현재 배경 스크롤링 속도값을 저장하기 위한 변수입니다.

property number lastTileFloor = 1 -- 마지막으로 스폰된 마지막 타일층수를 기록하기 위한 변수입니다.

property number defaultSpeed = -1 -- 캐릭터 선택시 설정되는 스크롤 속도값 입니다.

@ExecSpace("ClientOnly")
method void Scroll(number moveDistance)
-- 현재 설정중인 스크롤을 전달받은 이동거리만큼 활성화된 TileSet 오브젝트 이동합니다.

-- 부모 오브젝트(TileParent)를 이동시킵니다.
self.tileParent.TransformComponent:Translate(moveDistance,0)
end

@ExecSpace("ClientOnly")
method void ConnectRefs()
-- 내부에 필요한 멤버들의 참조를 연결하는 메서드입니다.

-- 모든 타일셋의 부모가 될 오브젝트의 Entity를 연결합니다.
self.tileParent = _EntityService:GetEntityByPath("/maps/MainGame/Scrolling/TileParent")

-- Background 컴포넌트를 연결합니다.
self.background = _EntityService:GetEntityByPath("/maps/MainGame/Background").BackgroundComponent
end

@ExecSpace("ClientOnly")
method void Init()
-- Scrolling Controller 컴포넌트의 내부 초기화를 진행하는 메서드입니다.

-- 내부 참조연결을 진행합니다.
self:ConnectRefs()

-- 배경 화면 스크롤링 속도값을 Config에 설정한 배경 스크롤링 속도로 설정합니다.
self.background.ScrollRate = _Config.backgroundScrollSpeed

-- 실제 배경오브젝트가 스크롤링 되는 속도값을 지정합니다.
self.currentBackgroundScrollSpeed = _Config.backgroundScrollSpeed

-- 생성된 타일셋의 개수를 초기화합니다.
self.spawnedCount = 0
end

@ExecSpace("ClientOnly")
method void OnUpdate(number delta)
-- 매 프레임마다 호출되는 메서드입니다.

-- 만약 메인게임이 종료되었다면
if _GameLogic.isEnd == true then
	-- Update 메서드 동작을 중단합니다.
	return
end

-- 설정된 스크롤링 속도로 바닥 오브젝트 이동을 수행합니다.
self:Scroll(-1 * delta * self.currentScrollSpeed)
end

@ExecSpace("ClientOnly")
method void SpawnRandomTile()
-- 무작위 타일셋을 생성하는 메서드입니다.

-- 마지막으로 생성한 타일 번호와 중복되지 않은 무작위 번호를 추출합니다.
local randomNumber = self:GetRandomTileNumber()

-- 랜덤으로 추출한 번호의 타일을 생성합니다.
self:SpawnTile(randomNumber)
end

@ExecSpace("ClientOnly")
method number GetRandomTileNumber()
-- 무작위 타일 번호를 추출하기 위한 메서드입니다.

-- 마지막으로 스폰된 타일셋의 끝나는 지점의 층수와 연결될 수 있는 타일셋 번호들이 담긴 테이블을 가져오는 과정입니다.
-- ex) 끝난 지점의 층수 : 2층 / 2층으로 시작하는 타일셋 번호가 담긴 테이블 O
-- ex) 끝난 지점의 층수 : 1층 / 2층으로 시작하는 타일셋 번호가 담긴 테이블 X
-- 연결된 타일셋을 생성하기 위함입니다.

-- 절차적 맵 생성의 핵심 로직입니다.

-- 마지막으로 스폰된 타일의 종료지점의 층수를 N 층으로 가정할때,
-- N 층으로 시작할 수 있는 타일셋 번호들이 담긴 테이블을 가져옵니다.
local tileSetFloorInfoTable = _DataSetLogic.tilesFloorInfoTable[self.lastTileFloor]

-- 해당 테이블의 총 개수를 확인합니다.
local tileSetTableTotalCount = #tileSetFloorInfoTable

-- 1 부터 타일 종류 수까지를 범위로 무작위 번호를 추출합니다.
local randomNumber = math.random(1,tileSetTableTotalCount)

-- 대상 층수만 담긴 테이블에서 'randomNumber'번째 타일셋 번호를 추출합니다.
local selectedTileNumber = tileSetFloorInfoTable[randomNumber]



-- 연속된 타일 스폰을 방지하기 위한 로직입니다.(연속된 타일셋 생성을 허용하길 원한다면 아래 구문을 수행하지 않으면 됩니다.)

-- 임의로 뽑은 숫자의 번호가 이전에 스폰된 타일 번호와 같다면 (중복된 타일선택을 막기 위해 다시 랜덤 돌리기를 진행합니다.)
if selectedTileNumber == self.lastSpawnedTileNumber then
	
	-- 다시 임의의 무작위 숫자를 추출합니다. (해당 층수에 맞는 타일셋 요소들중 랜덤요소를 추출합니다.)
	local tempNumber = math.random(1,tileSetTableTotalCount)
	
	-- 임시의 숫자가 또 다시 같은 번호라면
	if tempNumber == randomNumber then
		-- 임시로 뽑은 숫자의 값에 1 증가시킨 뒤 나머지 연산자를 활용합니다.
		tempNumber = (tempNumber + 1) % tileSetTableTotalCount+1
		
		-- 데이터셋의 첫 시작값은 1부터 시작되는 이유로, 만약 나머지 연산처리를 한 결과가 0이라면
		if tempNumber == 0 then
			-- tempNumber 값을 1로 설정합니다.
			tempNumber = 1
		end
		
	end
	-- 대상 층수만 담긴 테이블에서 임의로 뽑은 'tempNumber' 번째 요소를 선택합니다.
	selectedTileNumber = tileSetFloorInfoTable[tempNumber]
end

-- 해당 타일셋 번호를 반환합니다.
return selectedTileNumber
end

@ExecSpace("ClientOnly")
method void SetScrollSpeed(number speed)
-- 바닥 타일 오브젝트의 이동속도를 조절하는 메서드입니다.

-- 매개변수 1 number speed : 설정하고자 하는 속도값을 전달받습니다.

-- 현재 속도값을 매개변수로 전달받은 속도값으로 설정합니다.
self.currentScrollSpeed = speed
end

@ExecSpace("ClientOnly")
method void SetBackgroundScrollSpeed(number speed)
-- 배경화면 스크롤 속도를 조절하는 메서드입니다.

-- 매개변수 1 number speed : 설정하고자 하는 속도값을 전달받습니다.

-- 배경의 스크롤 속도를 매개변수 값으로 설정합니다.
self.background.ScrollRate = speed
end

@ExecSpace("ClientOnly")
method void ReturnAllTiles()
-- 생성된 모든 타일셋 오브젝트들을 전부 반환(비활성)하는 메서드

-- 'tileParent'의 자식들 중에서 재귀적으로 접근하여 'TileSet' 컴포넌트들을 테이블로 담습니다.
local tileSetTable = self.tileParent:GetChildComponentsByTypeName("TileSet",true)

-- 'TileSet'컴포넌트들이 담긴 테이블의 수만큼 순회를 진행합니다.
for	i = 1, #tileSetTable, 1 do
	
	-- 만약 각 요소의 오브젝트가 켜져(활성화되어) 있는 상태라면 (켜져있는 오브젝트들만 반환을 진행하기 위함입니다.)
	if tileSetTable[i].Entity.Enable == true then
			
		-- 각 요소들의 타일셋 컴포넌트로 접근합니다.
		local tempTileset = tileSetTable[i].Entity.TileSet
		-- 타일셋의 자식으로 배치된 아이템들을 전부 회수합니다.
		tempTileset:ReturnAllItems()
		-- 해당 타일셋을 다시 오브젝트 풀로 반환합니다.
		_ObjectPoolLogic:ReturnTile(tempTileset.Entity)
	end
	
	
end
end

method void SpawnTile(number tileSetNumber)
-- 타일셋 오브젝트를 스폰하는 메서드입니다.

-- 매개변수 1 number tileSetNumber : 해당 매개변수 번호의 타일셋을 생성하기 위해 전달받습니다.

-- 오브젝트 풀로부터 '매개변수'번의 타일 오브젝트를 전달받습니다.
local spawnedTile = _ObjectPoolLogic:GetTile(tileSetNumber)

-- 오브젝트 풀로부터 스폰받은 타일셋의 끝나는 지점의 층수를 기록합니다.
self.lastTileFloor = _DataSetLogic.tileSetInfoTable[tileSetNumber].LastFloor

-- 타일셋의 기준이 되는 부모 오브젝트로부터의 x축 거리를 설정합니다.
-- spawnedCount(기존에 생성된 타일셋 수) * 10(타일셋 프리팹의 기준 거리단위)
spawnedTile.TransformComponent.Position = Vector3(self.spawnedCount * 10,0,0)

-- 생성된 타일셋 수를 1 증가시킵니다.
self.spawnedCount += 1
-- 마지막으로 생성된 타일셋 번호를 오브젝트 풀로부터 가져온 타일셋 번호로 교체합니다.
self.lastSpawnedTileNumber = tileSetNumber

-- 오브젝트 풀로부터 가져온 타일셋의 초기화를 진행합니다.
spawnedTile.TileSet:Init()
end

@ExecSpace("ClientOnly")
method void Reset()
-- 게임 진행에 필요한 멤버 요소를 초기화하는 메서드입니다.

-- 생성되었던 타일수를 0으로 초기화합니다.
self.spawnedCount = 0
-- 모든 타일셋 오브젝트들의 부모가 되는 'tileParent' 오브젝트의 위치를 원점으로 초기화합니다.
self.tileParent.TransformComponent.Position = Vector3(0,0,0)
-- 배경화면 스크롤링 속도를 Config에서 기준으로 설정했던 값으로 설정합니다.
self:SetBackgroundScrollSpeed(_Config.backgroundScrollSpeed)

-- 게임을 시작할때 1부터 5번까지 순서대로 타일셋을 생성합니다.

-- 1부터 5번까지 반복문을 수행하며
for	i = 1, 5, 1 do
	-- i 번 타일을 스폰합니다.
	self:SpawnTile(i)
end
end

@ExecSpace("ClientOnly")
method void ChangeBackground()
-- 배경화면을 변경하는 메서드입니다.

-- 게임로직에 등록된 현재 스테이지 번호를 참고하여 데이터셋에 저장된 배경 RUID 값으로 배경화면을 설정합니다.
self.background:ChangeBackgroundByTemplateRUID(_DataSetLogic.stageRenderInfoTable[_GameLogic.currentStage].BackgroundRUID)
end

end
```

### 코드로 ScrollRate 변경하기
- 게임 진행 도중에 실시간으로 배경 스크롤속도를 제어하는 방법입니다.
- ScrollingController 스크립트 내 등록된 SetBackgroundScrollSpeed 메서드를 참고합니다.
```lua
void SetBackgroundScrollSpeed (number speed)
{
    self.background.ScrollRate = speed
}
```

### 스테이지 변경 방식
- 스테이지 번호를 관리하는 데이터는 GameLogic의 currentStage의 number형 변수로 관리됩니다.
- GameLogic의 OnUpdate메서드에서 일정 시간이 지나면 스테이지가 변경되는 기능을 호출합니다.
- 샘플 월드의 경우 총 3개의 스테이지만 등록되어 있지만, %연산자를 활용하여 1,2,3 -> 1,2로 스테이지를 순환하는 구조로 제작되어 있습니다.
- 바닥 타일, 장애물의 경우 엔티티가 활성화될 때, GameLogic의 currentStage 번호를 확인하고 현재 번호에 맞는 RUID값을 적용합니다.
- 다음 코드는 GameLogic의 코드 중 일부입니다.

```lua
GameLogic
number currentStage = 1 -- 현재 스테이지 정보를 기록하는 변수입니다.
number stageTimer = 0 -- 스테이지 전환까지 몇초가 남았는지 기록합니다.

void OnUpdate(number delta) -- 매 프레임마다 호출되는 메서드입니다.
{
    if(stageTimer > 0) then -- 타이머에 남은 시간이 남아있다면
        self.stageTimer -= delta -- 타이머에 delta값 감소

    else -- 타이머에 남은 시간이 없다면
        self.StageChange() -- StageChange 메서드를 호출합니다.
        self.stageTimer = _Config.stageChagneTime -- 타이머를 초기 시간값으로 초기화 합니다.
}

void StageChange() -- 스테이지를 전환하는 메서드입니다.
{
    -- 스테이지 번호를 1,2,3,1 로 계속 순환할 수 있는 구조로 제작합니다.
    self.currentStage = (self.currentStage % (totalStageCount+1))+1
    scrollingController:ChangeBackground()
}
```
