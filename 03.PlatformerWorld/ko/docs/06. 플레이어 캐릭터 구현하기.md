## 영상 타임라인
- 강의 영상 내 아래의 타임라인에서 학습할 수 있습니다.
- 플레이어 캐릭터 구현하기: 00:57:50
<br>

## 학습 목표
- 키보드 입력을 감지하여 캐릭터에 물리적인 힘을 가하는 점프 기능을 구현할 수 있습니다.
- 다양한 캐릭터를 생성하는 방법에 대해 학습합니다.
- 캐릭터별 사용 가능한 스킬을 정의하고, 스킬 사용시 해당 효과를 구현하는 방법을 학습합니다.
- 플레이어에게 다양한 캐릭터를 선택할 수 있는 환경을 제공할 수 있습니다.
<br>

## 해당 영상 시청 후 수행해 볼 내용
- 샘플 월드에서 제공하는 스킬 키 외 다른 입력 방식으로 스킬을 발동할 수 있도록 세팅해 봅니다.
- 캐릭터의 점프 힘을 변경하여 원하는 물리적 연출을 변경해 봅니다.
- CharacterInfo DataSet에 새로운 캐릭터를 추가하고 캐릭터 선택 창에 새로운 캐릭터로 플레이가 가능한지 확인해 봅니다.

## CustomPlayer 스크립트 작성하기

- 다음과 같이 코드를 작성해야 합니다.
- 해당 코드는 Plain Text 환경으로 작성되었습니다.

```lua
@Component
script CustomPlayerController extends Component

property string discript = "" -- 캐릭터의 기능 및 상태를 정의하기 위한 컴포넌트입니다.

property PhysicsRigidbodyComponent rigidBody = nil

property number jumpForce = 730 -- 캐릭터의 점프 수행 시 필요한 힘 값입니다.

property boolean isGrounded = false -- 캐릭터가 현재 바닥에 있는 상태인지를 확인하는 변수입니다.

property StateComponent stateComponent = nil

property number invincibleTimer = 0 -- 장애물 충돌 시 무적시간을 제어하기 위한 타이머 변수입니다.

property boolean isHit = false -- 캐릭터가 현재 장애물 피격상태인지 확인하는 변수입니다.

@ExecSpace("ClientOnly")
method void Init()
-- CustomPlayerController 컴포넌트 초기화를 진행하는 메서드입니다.

-- 내부 변수 참조 연결을 수행합니다.
self:ConnectRefs()

-- 기본 상태를 MOVE 상태로 진행합니다.
self:ChangeState("MOVE")

end

@ExecSpace("ClientOnly")
method void ConnectRefs()
-- 내부에 필요한 멤버들의 참조를 연결하는 메서드입니다.

-- 캐릭터에 부착된 Rigidbody 컴포넌트를 연결합니다.
self.rigidBody = self.Entity.PhysicsRigidbodyComponent
-- 캐릭터에 부착된 State 컴포넌트를 연결합니다.
self.stateComponent = self.Entity.StateComponent
end

@ExecSpace("ClientOnly")
method void Jump()
-- 캐릭터의 점프동작을 수행하는 메서드입니다.

-- 만약 바닥 감지를 못했다면
if self.isGrounded == false then
	-- 점프 수행을 중단합니다.
	return
end

-- 캐릭터의 y 축으로 적용되고 있던 속도값을 0으로 설정함으로 y 축 운동을 제거합니다.
self.rigidBody:SetLinearVelocityY(0)

-- y 축 방향으로 rigidbody를 통해 Force값 적용합니다.
self.rigidBody:ApplyForce(Vector2.up * self.jumpForce)

-- 점프 로직을 수행할때 누워 있는 상태가 아니라면
if self.stateComponent.CurrentStateName ~= "CROUCH" then
	-- 점프 상태로 전환합니다.
	self:ChangeState("JUMP")
end

_SoundService:PlaySound("24ee4f2425924a1abd4f655e8faf2791",1)

-- 점프를 수행하므로 캐릭터가 현재 바닥에 붙어있는 상태가 아님을 설정합니다.
self.isGrounded = false
end

@ExecSpace("ClientOnly")
method void ChangeState(string state)
-- 전달받은 상태로 캐릭터의 상태를 변경시키는 메서드입니다. (캐릭터의 애니메이션 제어 용도입니다.)

-- 매개변수 1 string state : 문자열 형태로 변경하려는 상태를 전달받습니다.

-- 멤버로 선언된 state 컴포넌트로 상태를 설정합니다.
self.stateComponent:ChangeState(state)

-- 트리거 콜라이더 박스 크기 조정 : 캐릭터가 엎드려있을때와 서 있는 상태의 콜라이더 상태를 구분짓기 위함입니다.

-- 매개변수로 전달받은 상태가 CROUCH 값 이라면
if state == "CROUCH" then
	-- 자신(캐릭터) 엔티티에 포함된 Trigger 컴포넌트의 트리거 Box 사이즈를 Config 에서 설정한 사이즈로 설정합니다.	
	self.Entity.TriggerComponent.BoxSize = _Config.crouchTriggerSize
	-- Trigger 컴포넌트의 트리거 Box 가 Pivot(기준)으로 부터 얼마나 떨어져 있어야하는지 오프셋값을 설정합니다.
	self.Entity.TriggerComponent.BoxOffset = _Config.crouchTriggerOffset
	
-- 매개변수로 전달받은 상태가 MOVE 값 이라면
elseif state == "MOVE" then
	-- 자신(캐릭터) 엔티티에 포함된 Trigger 컴포넌트의 트리거 Box 사이즈를 Config 에서 설정한 사이즈로 설정합니다.
	self.Entity.TriggerComponent.BoxSize = _Config.standTriggerSize
	-- Trigger 컴포넌트의 트리거 Box 가 Pivot(기준)으로 부터 얼마나 떨어져 있어야하는지 오프셋값을 설정합니다.
	self.Entity.TriggerComponent.BoxOffset = _Config.standTriggerOffset
end
end

@ExecSpace("ClientOnly")
method void ContactGround()
-- 바닥을 감지했을 때 수행하는 메서드입니다.

-- /분기1/ 현재 이미 바닥에 접촉해있는 중이라면
if self.isGrounded == true then
	-- 아무것도 수행하지 않고 메서드를 종료합니다.
		return
end
	
-- /분기2/ 기존에 바닥과 접촉이 아닌 상태에서 바닥을 접촉했다면

-- 현재 바닥상태임으로 상태를 변경합니다.
	self.isGrounded = true

-- 바닥 감지 시 이동 애니메이션을 재생합니다.
	self:ChangeState("MOVE")
end

@ExecSpace("ClientOnly")
method void ContactItem(Item item)
-- 아이템별로 수행되어야 할 동작을 진행합니다.

-- 매개변수 1 Item item : 아이템 오브젝트에 포함된 Item 컴포넌트를 전달받습니다.

-- 아이템 습득 시 효과음 재생
_SoundService:PlaySound("ef4235b158414f5283c068bad71f403a",1)

-- 전달받은 아이템 컴포넌트의 name 변수가 Coin 이라면
if item.name == "Coin" then
	-- 게임로직의 점수 추가 메서드를 수행하며 전달받은 아이템의 Value 값을 전달합니다.
	_GameLogic:AddScore(item.value)
end

-- 전달받은 아이템 컴포넌트의 name 변수가 Health 라면
if item.name == "Health" then
	-- 게임로직의 체력 추가 메서드를 수행하며 전달받은 아이템의 Value 값을 전달합니다.
	_GameLogic:AddHP(item.value)
end

-- 전달받은 아이템 컴포넌트의 name 변수가 Booster 라면
if item.name == "Booster" then
	-- 아이템로직에서 부스터 아이템 사용 메서드를 호출합니다.
 	_ItemLogic:UseBooster()
end

-- 아이템 습득 시 멀리 떨어진 곳으로 위치 이동합니다.
-- (오브젝트 풀로 부터 재생성시 문제가 있으므로 아이템 오브젝트의 위치를 일시적으로 보이지 않는 곳으로 이동합니다.)
item.Entity.TransformComponent.Position = Vector3.one * 100

-- 오브젝트 풀로 아이템 오브젝트를 반환하는 메서드를 수행합니다.
_ObjectPoolLogic:ReturnItem(item.Entity)
end

@ExecSpace("ClientOnly")
method void ContactObstacle()
-- 장애물을 접촉했을때 수행되는 메서드입니다.

-- 무적 지속시간이 남아있다면
if self.invincibleTimer > 0 then
	-- 아무것도 하지 않고 종료합니다.
	return
end

-- Config에 정의한 장애물 접촉 데미지를 게임로직의 체력 추가 메서드로 전달합니다.
_GameLogic:AddHP(_Config.obstacleDamage)

-- 피격 중이으로 상태를 설정합니다.
self.isHit = true

-- 무적 지속시간 타이머를 Config에서 설정한 초기값으로 설정합니다.
self.invincibleTimer = _Config.invincibleTime

-- 피격 이펙트를 재생합니다.
self:PlayOnDamagedEffect()

-- 아바타 렌더러의 투명도를 설정합니다. (피격상태를 전달하기 위함)
self.Entity.AvatarRendererComponent:SetAlpha(0.5)
end

@ExecSpace("ClientOnly")
method void PlayOnDamagedEffect()
-- 피격 시 발생되는 이펙트를 재생하는 메서드입니다.

-- 피격 이펙트 리소스 ID 값을 저장합니다.
local effectRUID = "41e13add2ba746d9bbe624a31205a848"
-- 맵 엔티티를 가져옵니다.
local mapEntity = _EntityService:GetEntityByPath("/maps/MainGame")
-- 이펙트 발생지점을 설정합니다.
local position = self.Entity.TransformComponent.Position

-- 이펙트 서비스를 활용해 이펙트를 재생합니다.
_EffectService:PlayEffect(effectRUID,mapEntity,position,0,Vector3.one * 3)
end

@ExecSpace("ClientOnly")
method void OnUpdate(number delta)
-- 매 프레임마다 호출되는 메서드입니다.


-- 피격중인 상태라면
if self.isHit == true then
	-- 피격 타이머가 남은 지속시간이 있다면
	if self.invincibleTimer > 0 then
		-- 피격 타이머에 지속시간을 감소시킵니다.
		self.invincibleTimer -= delta
	-- 피격 타이머가 남은 지속시간이 없다면
	else
		-- 피격상태 해제상태로 변경합니다.
		self.isHit = false
		-- 아바타 렌더러의 투명도값을 1로 초기화합니다.
		self.Entity.AvatarRendererComponent:SetAlpha(1)
	end
end
end

@ExecSpace("ClientOnly")
method void ResetCharacter()
-- 게임 재시작 시 필요한 초기값 설정을 진행하는 메서드입니다.

-- 해당 엔티티의 위치값을 절대좌표 x = 0, y = 1 로 설정합니다.
self.Entity.TransformComponent.Position = Vector3(0,1,0)
-- Rigidbody의 BodyType 을 유동적으로 설정하기 위해 Dynamic으로 설정합니다.
self.rigidBody.BodyType = BodyType.Dynamic

-- 캐릭터를 보이도록 설정합니다.
self.Entity.Visible = true
end

@ExecSpace("ClientOnly")
method void OnBeginPlay()
-- 월드 실행시 호출되는 메서드입니다.

-- 해당 컴포넌트 초기화 진행합니다.
self:Init()
end

@EventSender("Service", "InputService")
handler HandleKeyDownEvent(KeyDownEvent event)
-- 유저가 키 입력을 눌렀을때 호출되는 메서드입니다.

-- 게임이 종료된 상태라면
if _GameLogic.isEnd == true then
-- 해당 메서드 수행을 중단합니다.
	return
end

-- 입력된 키 정보를 저장합니다.
local key = event.key

-- 스페이스바 키 입력 시 
if key == KeyboardKey.Space then
	-- 점프 동작을 수행합니다.
	self:Jump()
end

-- 방향키 아래 키 입력 시
if key == KeyboardKey.DownArrow then
	-- 캐릭터의 상태를 'CROUCH' 로 설정합니다.
	self:ChangeState("CROUCH")
end

-- 'K' 키 입력 시
if key == KeyboardKey.K then
-- 캐릭터의 스킬을 사용합니다.
	_SkillLogic:UseSkill()
end


end

@EventSender("Service", "InputService")
handler HandleKeyUpEvent(KeyUpEvent event)
-- 유저의 키 입력을 종료할때 호출되는 메서드입니다.

-- 게임이 종료된 상태라면
if _GameLogic.isEnd == true then
-- 해당 메서드 수행을 중단합니다.
	return
end

-- 입력된 키 정보를 저장합니다.
local key = event.key
---------------------------------------------------------

-- 방향키 아래 키 입력 해제 시
if key == KeyboardKey.DownArrow then
	-- 다시 Crouch 상태를 해제합니다.
	self:ChangeState("MOVE")
end
end

@EventSender("Self")
handler HandleTriggerEnterEvent(TriggerEnterEvent event)
-- 캐릭터의 트리거 박스 내 다른 트리거와 충돌 시 호출되는 메서드입니다.

-- 게임이 종료된 상태라면
if _GameLogic.isEnd == true then
-- 해당 메서드 수행을 중단합니다.
	return
end

-- 트리거 충돌의 대상 엔티티를 저장합니다.
local TriggerBodyEntity = event.TriggerBodyEntity
-- 충돌된 트리거의 그룹명을 저장합니다.
local TriggerGroup = tostring(TriggerBodyEntity.TriggerComponent.CollisionGroup.Id)
---------------------------------------------------------

-- 대상 트리거가 아이템 오브젝트의 트리거라면
if _Config.triggerGroup[TriggerGroup] == "Item" then
	-- 아이템을 감지했을 때의 로직을 수행합니다.
   self:ContactItem(TriggerBodyEntity.Item)
end

-- 대상 트리거가 장애물 요소 오브젝트의 트리거라면
if _Config.triggerGroup[TriggerGroup] == "Obstacle" then
	-- 장애물을 감지했을 때의 로직을 수행합니다.
	self:ContactObstacle()
end

-- 대상 트리거가 DeadZone 요소 오브젝트의 트리거라면
if _Config.triggerGroup[TriggerGroup] == "DeadZone" then
	
	-- 현재 캐릭터와 데드존이 만나는 조건은 화면 하단으로 캐릭터가 떨어지는 상황이므로
	-- 게임 종료 명령 수행이 필요합니다.
	
	-- Rigidbody의 BodyType 을 Static으로 물리운동이 진행되지 않도록 설정합니다.
	self.rigidBody.BodyType = BodyType.Static
	-- 게임 로직의 게임 종료 메서드를 수행합니다.
	_GameLogic:EndMainGame()
end
end

@EventSender("Self")
handler HandlePhysicsContactBeginEvent(PhysicsContactBeginEvent event)
-- Physics 강체의 물리 충돌이 감지되었을때 발생되는 이벤트입니다.

-- 충돌한 강체의 엔티티를 가져옵니다.
local ContactedBodyEntity = event.ContactedBodyEntity

-- 충돌한 강체의 부모 오브젝트의 TileSet 컴포넌트를 가져옵니다.
local tileSet = ContactedBodyEntity.Parent.TileSet

-- 접촉한 오브젝트에서 타일셋이 검출되었다면
if tileSet ~= nil then
	
	-- 접촉한 오브젝트의 Transform 컴포넌트를 가져옵니다.
	local contactTransform = ContactedBodyEntity.TransformComponent
	-- 캐릭터의 Transform 컴포넌트를 가져옵니다.
	local playerTransform = self.Entity.TransformComponent
	
	-- 만약 충돌체 대상의 오브젝트의 위치 y 값이 캐릭터의 위치 y 값 보다 더 높다면
	if contactTransform.Position.y >= playerTransform.Position.y then
		-- 해당 메서드 수행을 중단합니다.
		return
	end
	
	-- 현재 바닥에 있는 상태임을 전달합니다.
	self.isGrounded = true
	-- 만약 현재 상태가 'JUMP' 라면
	if self.stateComponent.CurrentStateName == "JUMP" then
		-- 'MOVE' 상태로 전환합니다.
		self:ChangeState("MOVE")
	end
end
end

end
```

### 점프 기능 수행과정 이해하기
- 설정된 환경에서 활용할 수 있는 점프기능은 캐릭터에 등록한 PhysicsRigidbody의 물리적인 기능을 활용하는 방법이 있습니다.
- CustomPlayerController 컴포넌트 내에서 캐릭터의 물리 기능을 제어하기 위해 PhysicsRigidbodyComponent 타입의 변수를 선언하고 컴포넌트 참조를 연결합니다.
- 엔티티에 위쪽방향으로 힘을 가하기 위해서는 PhysicsRigidbody의 ApplyForce메서드를 활용합니다.
- 캐릭터가 점프상태임을 알 수 있도록 점프 애니메이션을 재생시키도록 합니다.

```lua
캐릭터 점프기능 구현 예시

PhysicsRigidbodyComponent rigidBody;
StateComponent state;

number jumpForce = 730
boolean isGrounded = false


void OnBeginPlay()
{
    -- 물리적인 힘을 가할 PhysicsRigidbodyComponent 연결
    self.rigidBody = self.Entity.PhysicsRigidbodyComponent
    -- 캐릭터의 상태전환을 위한 StateComponent 연결
    self.state = self.Entity.StateComponent
}

void Jump()
{
    if self.isGround == false then 
        return -- 만약 캐릭터가 지면에 있지 않다면, 점프기능을 수행하지 않고 종료합니다. 
    end

    -- 점프 수행이 시작될때, 캐릭터가 받고있는 Y축 속도,힘을 소멸합니다.
    self.rigidBody:SetLinearVelocityY(0)

    -- Vector2.up 캐릭터의 위쪽 방향으로 jumpForce 만큼 힘을 가합니다.
    self.rigidBody:ApplyForce(Vector2.up * self.jumpForce)

    -- 점프 애니메이션을 적용하기 위해 StateComponent에게 Jump상태로 변경을 요청합니다.
    self.state:ChangeState("Jump")

    -- 점프를 수행하므로 캐릭터가 현재 바닥에 있지 않음으로 상태를 전환합니다.
    self.isGround = false;
}

HandleKeyDownEvent(KeyDownEvent event)
{
    -- 키보드의 키가 눌렸을때 호출되는 메서드입니다.

    -- 입력된 키 정보를 저장합니다.
    local key = event.key

    -- 만약 입력된 키가 Space 키라면
    if key == KeyboardKey.Space then
    -- Jump 메서드를 수행합니다.
        self:Jump()
    end
}

```

### 바닥 감지 시 캐릭터 상태 전환 방식
- 캐릭터의 연속적인 점프 수행을 방지하기 위해 boolean isGround 상태 변수를 활용하고 있습니다.
- 캐릭터의 점프가 시작될 때 해당 변수를 false로 전환합니다.
- 점프가 완료되고 중력에 의해 바닥에 착지했을 때, isGrounded 상태 변수를 다시 true로 전환하여 점프 가능한 상태로 복구해야 합니다.
- PhysicsCollider와 EventHandler를 통해 상태 전환이 가능합니다.

```lua
바닥 충돌 상태전환 예시코드

StateComponent state

void OnBeginPlay()
{
    -- 캐릭터의 상태전환을 위한 StateComponent 연결
    state = self.Entity.StateComponent
}

HandlePhysicsContactBeginEvent(PhysicsContactBeginEvent event)
{
    -- PhysicsCollider로 부터 다른 충돌체와 충돌이 감지되었을때 호출되는 메서드입니다.

    -- 충돌체 대상의 엔티티를 저장합니다.
    local ContactedBodyEntity = event.ContactedBodyEntity
    -- 충돌체 대상으로부터 TileSet 컴포넌트를 가져오도록 합니다.
    local tileSet = ContactedBodyEntity.TileSet

    -- 만약 TileSet 컴포넌트를 확인했다면 (바닥 타일임을 확인했다면)
    if tileSet ~= nil then
        -- isGround 상태를 true로 전환합니다.
        isGround = true
        -- 점프 상태였던 애니메이션을 다시 걷는 애니메이션으로 전환합니다.
        self.state.ChangeState("Move")
    end
}
```

## SkillLogic 스크립트 작성하기
- 다음과 같이 코드를 작성해야 합니다.
- 해당 코드는 Plain Text 환경으로 작성되었습니다.

```lua
@Logic
script SkillLogic extends Logic

property number discript = 0 -- 캐릭터의 고유 스킬 사용시 구체적인 동작을 수행하기 위해 선언한 로직입니다.

property number skillId = 0 -- 캐릭터의 고유 SkillId 값을 설정하기 위한 변수입니다.

property boolean useable = true -- 스킬사용 가능 상태를 저장하기 위한 상태변수입니다.

@ExecSpace("ClientOnly")
method void UseSkill()
-- 현재 등록된(SkillId)의 구체적인 동작을 수행하는 메서드입니다.

-- 해당 메서드 호출 시 사용할수 있는 상태가 아니라면
if self.useable == false then
	-- 메서드를 종료합니다.
	return
end

-- 현재 캐릭터 스킬명을 저장합니다.
local skillName = _DataSetLogic.skillInfoTable[self.skillId].Name
-- 현재 캐릭터 스킬고윳값을 저장합니다.
local skillValue = _DataSetLogic.skillInfoTable[self.skillId].Value

-- 스킬명이 'GetLife' 라면
if skillName == "Heal" then
-- 스킬 고윳값만큼 체력을 회복합니다.
	_GameLogic:AddHP(skillValue)
	

-- 스킬명이 'GetScore' 라면
elseif skillName == "Bonus" then
-- 스킬 고윳값만큼 추가 점수를 획득합니다.
	_GameLogic:AddScore(skillValue)
end

-- 스킬 이펙트를 재생합니다.
self:PlaySkillEffect()

-- 스킬 효과음 재생
_SoundService:PlaySound("56c1b5e3c7fc4abda561c1cc00813bda",1)

-- 스킬을 사용했으므로 사용할 수 없음으로 상태를 전환합니다.
self.useable = false

-- 스킬 아이콘 UI 의 렌더러의 투명도 값을 낮게 설정합니다.
_UIManagerLogic.skillRenderer.Color.a = 0.3
end

@ExecSpace("ClientOnly")
method void Reset()
-- 내부 상태를 초기화하는 메서드입니다.

-- 스킬을 사용할수 있게 전환합니다.
self.useable = true
end

@ExecSpace("ClientOnly")
method void PlaySkillEffect()
-- 스킬 이펙트를 재생하기 위한 메서드입니다.

-- SkillInfoTable에 저장된 스킬 이펙트 RUID 값을 가져옵니다.
local effect = _DataSetLogic.skillInfoTable[self.skillId].EffectRUID

-- EffectService를 활용해 캐릭터 오브젝트에 부착하여 스킬 이펙트를 재생합니다.
_EffectService:PlayEffectAttached(effect,_UserService.LocalPlayer,Vector3.zero,0,Vector3.one)
end

end
```

## 캐릭터 Slot에 캐릭터별 정보 등록하기
- 월드 실행 시 모든 캐릭터의 Slot 아이템을 미리 전부 생성합니다.
- 각 슬롯은 CharaterInfo DataSet의 모든 요소와 캐릭터의 데이터를 전부 출력해야 합니다.
- Slot에 캐릭터 정보를 등록하는 과정은 다음과 같습니다.
  - 월드 실행 -> 자동으로 호출되는 OnBeginPlay내에서 RegistCharacter메서드를 호출하도록 합니다.
  - RegistCharacter메서드의 기능은 다음과 같습니다.

```lua
UICharacterSelectPopup 의 RegistCharacter메서드

Entity scrollViewEntity; -- Slot 아이템이 배치되어야할 부모 엔티티

void RegistCharacter()
{
	-- Workspace에 등록된 Slot 아이템의 ModelID를 저장합니다.
	local modelId = _EntryService:GetModelIdByName("UICharacterSelect")
	-- 캐릭터 정보 테이블을 호출하는 코드를 간결하게 작성하기 위해 지역변수로 저장합니다.
	local characterTable = _DataSetLogic.characterInfoTable

	-- 캐릭터 정보 테이블에 등록된 개수만큼 순회를 진행합니다. (CharacterInfo DataSet에 등록된 캐릭터 수)
	for i = 1, #characterTable, 1 do

		-- SpawnService의 SpawnByModelID메서드를 활용해 하나의 Slot아이템을 생성합니다.
		local characterInfo = _SpawnService:SpawnByModelID(
			modelId,"characterInfoPopup",Vector3.zero,scrollViewEntity).UICharacterInfoComponent

		-- 캐릭터의 이름을 표기하는 UI의 텍스트를 캐릭터 이름으로 설정합니다.
		characterInfo.nameText.Text = characterTable[i].Name

		-- Slot에 미리 등록된 캐릭터 외형(CustomBodyEquip)의 리소스 ID 값을 설정합니다.
		characterInfo.costumImage.CustomBodyEquip = characterTable[i].RUID

		-- CharacterInfo DataSet에 저장된 SkillId 번호를 추출합니다.
		local skillId = tonumber(characterTable[i].SkillId)
		-- SkillInfo DataSet에 저장된 Skill 정보를 불러와 해당 스킬의 이름을 등록합니다.
		characterInfo.skillText.Text = _DataSetLogic.skillInfoTable[skillId].Name

		-- 캐릭터의 스크롤 속도를 표기하는 UI의 텍스트를 캐릭터 고유 스크롤 속도로 설정합니다.
		characterInfo.speedText.Text = characterTable[i].Speed

	end
}
```

### 캐릭터 선택창 제작하기
<p align = "Center">
<img src = "../../_images/MSW-100_Running_6_009.png" alt="MSW-100_Running_6_009.png">
</p>

- 캐릭터 선택 창의 최상위 엔티티에 월드 실행 시 캐릭터 Slot을 생성 기능을 수행하는 컴포넌트를 등록합니다.
- 제작이 완료된 경우, 플레이어가 게임 시작 버튼을 클릭하기 전까지 해당 팝업이 보이지 않아야하므로 엔티티를 비활성화 해둡니다.
- 다음과 같이 코드를 작성해야 합니다.
- 해당 코드는 Plain Text 환경으로 작성되었습니다.

```lua
@Component
script UICharacterSelectPopup extends Component

property number discript = 0 -- 캐릭터 선택창 팝업을 띄울때 필요한 기능을 정의한 컴포넌트입니다.

property Entity scrollViewParent = nil

@ExecSpace("ClientOnly")
method void OnBeginPlay()
-- 월드 시작 시 호출되는 메서드입니다.

-- 모델(프리팹)으로 부터 생성되는 오브젝트들의 부모로 설정될 엔티티를 연결합니다.
self.scrollViewParent = _EntityService:GetEntityByPath("/ui/ToastGroup/UITitle/UICharacterSelect/UIPanel/Scroll_Layout")

-- 캐릭터 정보등록을 진행합니다.
self:RegistCharacter()
end

@ExecSpace("ClientOnly")
method void RegistCharacter()
-- 캐릭터 선택 창 내 캐릭터 정보를 등록하는 메서드입니다.
-- 모델(프리팹)으로 부터 캐릭터 정보를 나타내는 팝업이 생성될때 내부에 기입되는 데이터 값들을 설정하기 위함입니다.

-- 기본 모델(프리팹)요소의 모델Id를 가져옵니다.
local modelId = _EntryService:GetModelIdByName("UICharacterSelect")
-- 데이터셋으로부터 캐릭터 정보 테이블을 가져옵니다.
local characterTable = _DataSetLogic.characterInfoTable

-- 테이블 내에 등록된 캐릭터 수만큼 반복문을 수행합니다.
for	i = 1, #characterTable, 1 do
	
	-- 스폰 서비스를 활용해 UI 팝업을 생성하며 캐릭터인포 컴포넌트를 저장합니다.
	local characterInfo = _SpawnService:SpawnByModelId(modelId,"characterInfoPopup",Vector3.zero,self.scrollViewParent).UICharacterInfoComponent

	-- 스킬인포 테이블을 식별하기 위한 skillId값을 불러옵니다.
	local skillId = tonumber(characterTable[i].SkillId)
	-- 캐릭터 인포 내에 설정될 캐릭터 스킬 번호를 설정합니다.
	characterInfo.skillId = skillId
	
	-- 캐릭터 인포 내 설정될 캐릭터명을 설정합니다.
	characterInfo.nameText.Text = characterTable[i].Name
	-- 캐릭터 인포 내 코스튬 바디 RUID값을 설정합니다.
	characterInfo.costumeImage.CustomBodyEquip = characterTable[i].RUID
	-- 캐릭터 인포 내 스킬 명을 설정합니다 (저장된 SkillId 번호를 기반으로 스킬 인포 테이블에서 스킬명 가져옵니다.)
	characterInfo.skillText.Text = _DataSetLogic.skillInfoTable[skillId].Name
	-- 캐릭터 인포 내 속도 텍스트를 설정합니다.
	characterInfo.speedText.Text = characterTable[i].Speed
	
	-- 캐릭터 ID(인덱스)값을 설정합니다.
	characterInfo.characterId = i
end

end

end
```

### 캐릭터 Slot 제작하기
<p align = "Center">
<img src = "../../_images/MSW-100_Running_6_010.png" alt="MSW-100_Running_6_010.png">
</p>

- Slot의 하단에 배치된 캐릭터 선택 기능을 수행할 버튼의 기능을 동작시키기 위한 스크립트를 제작합니다.
- 버튼의 이벤트로 CloseSelectPopup메서드를 호출할 수 있도록합니다.
- CloseSelectPopup 메서드는 다음과 같은 동작을 수행합니다.
  - 게임 시작 시 보이는 타이틀화면을 구성하는 UI요소 비활성화합니다.
  - 플레이어가 사용하는 캐릭터의 외형을 변경합니다.
  - 해당 캐릭터의 스킬 번호를 SkillLogic에게 전달합니다.
  - GameLogic에게 현재 캐릭터가 몇 번의 캐릭터인지 전달하며 게임시작 기능을 호출합니다.
- 다음과 같이 코드를 작성해야 합니다.
- 해당 코드는 Plain Text 환경으로 작성되었습니다.

```lua
@Component
script UICharacterInfoComponent extends Component

property number discript = 0 -- 캐릭터 선택 창에서 각 캐릭터의 정보를 표기하기 위한 컴포넌트입니다.

property TextComponent nameText = nil

property CostumeManagerComponent costumeImage = nil

property TextComponent skillText = nil

property TextComponent speedText = nil

property ButtonComponent selectButton = nil

property number characterId = -1 -- 캐릭터 인덱스를 구분하기 위한 변수입니다.

property number skillId = -1 -- 캐릭터 고유 스킬 번호입니다.

@ExecSpace("ClientOnly")
method void OnBeginPlay()
-- 월드 시작 시 호출되는 메서드입니다.

-- 해당 컴포넌트 초기화를 진행합니다.
self:Init()
end

@ExecSpace("ClientOnly")
method void Init()
-- CharacterInfo 컴포넌트 초기화를 진행하는 메서드입니다.

-- 모델(프리팹) 자식 오브젝트 컴포넌트 연결을 수행합니다.

-- 캐릭터명을 기입할 텍스트 컴포넌트를 자식오브젝트에서 가져옵니다.
self.nameText = self.Entity:GetChildByName("UIName").TextComponent
-- 캐릭터 이미지를 등록 할 코스튬 컴포넌트를 자식오브젝트에서 가져옵니다.
self.costumeImage = self.Entity:GetChildByName("UIImage").CostumeManagerComponent
-- 스킬명을 기입할 텍스트 컴포넌트를 자식오브젝트에서 가져옵니다.
self.skillText = self.Entity:GetChildByName("UISkill",true).TextComponent
-- 속도값을 기입할 텍스트 컴포넌트를 자식오브젝트에서 가져옵니다.
self.speedText = self.Entity:GetChildByName("UISpeed",true).TextComponent
-- 마우스 클릭을 감지할 버튼 컴포넌트를 자식오브젝트에서 가져옵니다.
self.selectButton = self.Entity:GetChildByName("UISelectButton").ButtonComponent

-- 캐릭터 선택버튼 클릭시 수행될 메서드(CloseSelectPopup)를 등록합니다.
self.selectButton.Entity:ConnectEvent(ButtonClickEvent,self.CloseSelectPopup)

end

@ExecSpace("ClientOnly")
method void SetInfo(string name, string ruid, string skill, string speed)
-- 캐릭터 정보창에 필요한 정보,내용을 설정하기 위한 메서드입니다.

-- 매개변수 1 string name : 캐릭터 명을 전달 받습니다.
-- 매개변수 2 sting ruid : 캐릭터 몸통 RUID 값을 전달 받습니다.
-- 매개변수 3 string skill : 캐릭터 스킬명을 전달 받습니다.
-- 매개변수 4 string speed : 캐릭터 스크롤링 속도값을 전달받습니다.

-- 캐릭터 명 텍스트를 설정합니다.
self.nameText.Text = name
-- 코스튬 바디파츠 부분 리소스를 설정합니다.
self.costumeImage.CustomBodyEquip = ruid
-- 스킬 명 텍스트를 설정합니다.
self.skillText.Text = skill
-- 스크롤링 속도값 텍스트를 설정합니다.
self.speedText.Text = speed
end

@ExecSpace("ClientOnly")
method void CloseSelectPopup()
-- 캐릭터 선택창 팝업을 비활성화 시키는 메서드입니다.

-- 게임 시작전 타이틀 패널(시작버튼, 조작법, 캐릭터 선택 창의 부모) 엔티티를 가져옵니다.
local TitlePanel = _EntityService:GetEntityByPath("/ui/ToastGroup/UITitle")

-- 패널 오브젝트를 비활성화합니다.
TitlePanel.Enable = false

-- 캐릭터의 코스튬 바디 리소스값을 설정합니다.
_UserService.LocalPlayer.CostumeManagerComponent.CustomBodyEquip = self.costumeImage.CustomBodyEquip

-- 게임내에서 사용할 캐릭터가 고유한 스킬Id를 설정합니다.
_SkillLogic.skillId = self.skillId

-- 게임 로직에 선택한 캐릭터Id 번호로 게임 진행을 요청합니다.
_GameLogic:GameStart(self.characterId)
end

end
```

## 최소 구현 기준
- 플레이어 캐릭터가 바닥 타일과의 충돌을 정상적으로 감지하며, 공중에서 무한히 도약하지 않고 지면 상태에서만 점프를 수행하도록 제한 기능이 정상적으로 동작해야 합니다.
- 캐릭터 선택 창에서 캐릭터 Slot에 등록된 버튼 입력 시 GameLogic에게 GameStart를 호출할 수 있도록 제작되어야 합니다.
<br>

## 응용 및 확장 방향
- CharacterInfo DataSet에 캐릭터별 스크롤링 속도 외에도 점프력, 최대 체력, 무적 시간 등의 세부 능력치 열을 추가해 캐릭터마다 다른 플레이 스타일을 제공해 볼 수 있습니다.
- 장애물을 파괴하며 달리는 '무적 돌진'이나 주변의 코인을 한 번에 습득하는 등 다채로운 액션 스킬을 기획하고 추가해 봅니다.
- 캐릭터를 선택할 때 바디 코스튬 하나만 교체하는 것을 넘어, 헤어, 얼굴, 망토 등 다양한 파츠 데이터를 묶어 변경하고 스크립트를 고도화할 수 있습니다.
