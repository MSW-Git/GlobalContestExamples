## 撰寫CustomPlayer腳本

- 必須如下撰寫程式碼。
- 該程式碼是在Plain Text環境下撰寫。

```lua
@Component
script CustomPlayerController extends Component

property string discript = "" -- 這是用來定義角色功能與狀態的元件。

property PhysicsRigidbodyComponent rigidBody = nil

property number jumpForce = 730 -- 角色執行跳躍時所需的力量值。

property boolean isGrounded = false -- 確認角色目前是否在地面上的變數。

property StateComponent stateComponent = nil

property number invincibleTimer = 0 -- 用來控制障礙物碰撞時無敵時間的計時器變數。

property boolean isHit = false -- 確認角色目前是否為障礙物受擊狀態的變數。

@ExecSpace("ClientOnly")
method void Init()
-- 執行CustomPlayerController元件初始化的方法。

-- 執行內部變數參照連接。
self:ConnectRefs()

-- 預設狀態設為MOVE狀態。
self:ChangeState("MOVE")

end

@ExecSpace("ClientOnly")
method void ConnectRefs()
-- 連接內部所需成員參照的方法。

-- 連接附加在角色上的Rigidbody元件。
self.rigidBody = self.Entity.PhysicsRigidbodyComponent
-- 連接附加在角色上的State元件。
self.stateComponent = self.Entity.StateComponent
end

@ExecSpace("ClientOnly")
method void Jump()
-- 執行角色跳躍動作的方法。

-- 若未偵測到地面
if self.isGrounded == false then
	-- 中止跳躍執行。
	return
end

-- 將角色y軸目前套用中的速度值設為0，以移除y軸運動。
self.rigidBody:SetLinearVelocityY(0)

-- 透過rigidbody在y軸方向施加Force值。
self.rigidBody:ApplyForce(Vector2.up * self.jumpForce)

-- 執行跳躍邏輯時若不是趴下狀態
if self.stateComponent.CurrentStateName ~= "CROUCH" then
	-- 切換為跳躍狀態。
	self:ChangeState("JUMP")
end

_SoundService:PlaySound("24ee4f2425924a1abd4f655e8faf2791",1)

-- 因執行跳躍，因此設定角色目前不是貼在地面的狀態。
self.isGrounded = false
end

@ExecSpace("ClientOnly")
method void ChangeState(string state)
-- 將角色狀態變更為傳入狀態的方法。(用來控制角色動畫。)

-- 參數1 string state：傳入欲變更的字串狀態。

-- 以成員宣告的state元件設定狀態。
self.stateComponent:ChangeState(state)

-- 調整觸發碰撞箱大小：用來區分角色趴下與站立狀態的碰撞箱狀態。

-- 若傳入的狀態為CROUCH值
if state == "CROUCH" then
	-- 將自身(角色)實體所包含的Trigger元件之觸發Box大小設為Config中設定的尺寸。	
	self.Entity.TriggerComponent.BoxSize = _Config.crouchTriggerSize
	-- 設定Trigger元件的觸發Box距離Pivot(基準點)應偏移多少的offset值。
	self.Entity.TriggerComponent.BoxOffset = _Config.crouchTriggerOffset
	
-- 若傳入的狀態為MOVE值
elseif state == "MOVE" then
	-- 將自身(角色)實體所包含的Trigger元件之觸發Box大小設為Config中設定的尺寸。
	self.Entity.TriggerComponent.BoxSize = _Config.standTriggerSize
	-- 設定Trigger元件的觸發Box距離Pivot(基準點)應偏移多少的offset值。
	self.Entity.TriggerComponent.BoxOffset = _Config.standTriggerOffset
end
end

@ExecSpace("ClientOnly")
method void ContactGround()
-- 偵測到地面時執行的方法。

-- /分支1/ 若目前已經接觸地面中
if self.isGrounded == true then
	-- 不執行任何內容並結束方法。
		return
end
	
-- /分支2/ 若原本未接觸地面狀態下接觸到地面

-- 因目前為地面狀態，變更狀態。
	self.isGrounded = true

-- 偵測到地面時播放移動動畫。
	self:ChangeState("MOVE")
end

@ExecSpace("ClientOnly")
method void ContactItem(Item item)
-- 執行各道具應進行的動作。

-- 參數1 Item item：傳入道具物件內包含的Item元件。

-- 取得道具時播放效果音
_SoundService:PlaySound("ef4235b158414f5283c068bad71f403a",1)

-- 若傳入道具元件的name變數為Coin
if item.name == "Coin" then
	-- 執行GameLogic的加分方法，並傳入該道具的Value值。
	_GameLogic:AddScore(item.value)
end

-- 若傳入道具元件的name變數為Health
if item.name == "Health" then
	-- 執行遊戲邏輯的增加體力方法，並傳入該道具的Value值。
	_GameLogic:AddHP(item.value)
end

-- 若傳入道具元件的name變數為Booster
if item.name == "Booster" then
	-- 呼叫道具邏輯中的使用加速器道具方法。
 	_ItemLogic:UseBooster()
end

-- 取得道具時移動到遠處位置。
-- (因從物件池重新生成時會有問題，所以暫時將道具物件移至看不見的位置。)
item.Entity.TransformComponent.Position = Vector3.one * 100

-- 執行將道具物件回收至物件池的方法。
_ObjectPoolLogic:ReturnItem(item.Entity)
end

@ExecSpace("ClientOnly")
method void ContactObstacle()
-- 接觸障礙物時執行的方法。

-- 若無敵持續時間尚未結束
if self.invincibleTimer > 0 then
	-- 不做任何事並結束。
	return
end

-- 將Config定義的障礙物接觸傷害值傳給遊戲邏輯的增加體力方法。
_GameLogic:AddHP(_Config.obstacleDamage)

-- 設定為受擊中的狀態。
self.isHit = true

-- 將無敵持續時間計時器設為Config中設定的初始值。
self.invincibleTimer = _Config.invincibleTime

-- 播放受擊特效。
self:PlayOnDamagedEffect()

-- 設定紙娃娃渲染的透明度。(用來表現受擊狀態)
self.Entity.AvatarRendererComponent:SetAlpha(0.5)
end

@ExecSpace("ClientOnly")
method void PlayOnDamagedEffect()
-- 播放受擊時出現特效方法。

-- 儲存受擊特效資源ID值。
local effectRUID = "41e13add2ba746d9bbe624a31205a848"
-- 取得地圖實體。
local mapEntity = _EntityService:GetEntityByPath("/maps/MainGame")
-- 設定特效出現位置。
local position = self.Entity.TransformComponent.Position

-- 使用特效服務播放特效。
_EffectService:PlayEffect(effectRUID,mapEntity,position,0,Vector3.one * 3)
end

@ExecSpace("ClientOnly")
method void OnUpdate(number delta)
-- 這是每一幀都會被呼叫的方法。


-- 若為受擊中狀態
if self.isHit == true then
	-- 若受擊計時器尚有剩餘持續時間
	if self.invincibleTimer > 0 then
		-- 減少受擊計時器的持續時間。
		self.invincibleTimer -= delta
	-- 若受擊計時器已無剩餘持續時間
	else
		-- 變更為解除受擊狀態。
		self.isHit = false
		-- 將紙娃娃渲染透明度值初始化為1。
		self.Entity.AvatarRendererComponent:SetAlpha(1)
	end
end
end

@ExecSpace("ClientOnly")
method void ResetCharacter()
-- 遊戲重新開始時設定所需初始值的方法。

-- 將該實體位置絕對座標設為x = 0, y = 1。
self.Entity.TransformComponent.Position = Vector3(0,1,0)
-- 為了動態設定Rigidbody的BodyType，設為Dynamic。
self.rigidBody.BodyType = BodyType.Dynamic

-- 讓角色顯示。
self.Entity.Visible = true
end

@ExecSpace("ClientOnly")
method void OnBeginPlay()
-- 這是在世界執行時呼叫的方法。

-- 執行該元件初始化。
self:Init()
end

@EventSender("Service", "InputService")
handler HandleKeyDownEvent(KeyDownEvent event)
-- 當使用者按下按鍵時呼叫的方法。

-- 若遊戲已結束狀態
if _GameLogic.isEnd == true then
-- 中止執行該方法。
	return
end

-- 儲存輸入的按鍵資訊。
local key = event.key

-- 按下空白鍵時 
if key == KeyboardKey.Space then
	-- 執行跳躍動作。
	self:Jump()
end

-- 按下方向鍵下鍵時
if key == KeyboardKey.DownArrow then
	-- 將角色狀態設為'CROUCH'。
	self:ChangeState("CROUCH")
end

-- 按下'K'鍵時
if key == KeyboardKey.K then
-- 使用角色技能。
	_SkillLogic:UseSkill()
end


end

@EventSender("Service", "InputService")
handler HandleKeyUpEvent(KeyUpEvent event)
-- 當使用者結束按鍵輸入時呼叫的方法。

-- 若遊戲已結束狀態
if _GameLogic.isEnd == true then
-- 中止執行該方法。
	return
end

-- 儲存輸入的按鍵資訊。
local key = event.key
---------------------------------------------------------

-- 放開方向鍵下鍵時
if key == KeyboardKey.DownArrow then
	-- 再次解除Crouch狀態。
	self:ChangeState("MOVE")
end
end

@EventSender("Self")
handler HandleTriggerEnterEvent(TriggerEnterEvent event)
-- 當角色的觸發箱內與其他觸發器碰撞時呼叫。

-- 若遊戲已結束狀態
if _GameLogic.isEnd == true then
-- 中止執行該方法。
	return
end

-- 儲存觸發碰撞的目標實體。
local TriggerBodyEntity = event.TriggerBodyEntity
-- 儲存碰撞觸發器的群組名稱。
local TriggerGroup = tostring(TriggerBodyEntity.TriggerComponent.CollisionGroup.Id)
---------------------------------------------------------

-- 若目標觸發器為道具物件的觸發器
if _Config.triggerGroup[TriggerGroup] == "Item" then
	-- 執行偵測到道具時的邏輯。
   self:ContactItem(TriggerBodyEntity.Item)
end

-- 若目標觸發器為障礙物元素物件的觸發器
if _Config.triggerGroup[TriggerGroup] == "Obstacle" then
	-- 執行偵測到障礙物時的邏輯。
	self:ContactObstacle()
end

-- 若目標觸發器為DeadZone要素物件的觸發器
if _Config.triggerGroup[TriggerGroup] == "DeadZone" then
	
	-- 目前角色與DeadZone相遇的條件為角色掉落至畫面下方時
	-- 因此需要執行遊戲結束命令。
	
	-- 將Rigidbody的BodyType設為Static，使物理運動不再進行。
	self.rigidBody.BodyType = BodyType.Static
	-- 執行遊戲邏輯的遊戲結束方法。
	_GameLogic:EndMainGame()
end
end

@EventSender("Self")
handler HandlePhysicsContactBeginEvent(PhysicsContactBeginEvent event)
-- 偵測到Physics剛體物理碰撞時發生的事件。

-- 取得碰撞剛體的實體。
local ContactedBodyEntity = event.ContactedBodyEntity

-- 取得碰撞剛體父物件的圖塊集元件。
local tileSet = ContactedBodyEntity.Parent.TileSet

-- 若從接觸物件中偵測到圖塊集
if tileSet ~= nil then
	
	-- 取得接觸物件的Transform元件。
	local contactTransform = ContactedBodyEntity.TransformComponent
	-- 取得角色的Transform元件。
	local playerTransform = self.Entity.TransformComponent
	
	-- 若碰撞體目標物件的位置y值高於角色位置y值
	if contactTransform.Position.y >= playerTransform.Position.y then
		-- 中止執行該方法。
		return
	end
	
	-- 傳遞目前為站在地面的狀態。
	self.isGrounded = true
	-- 若目前狀態為'JUMP'
	if self.stateComponent.CurrentStateName == "JUMP" then
		-- 切換為'MOVE'狀態。
		self:ChangeState("MOVE")
	end
end
end

end
```

### 理解跳躍功能執行過程
- 在已設定的環境中，可使用的跳躍功能之一，是利用登錄在角色上的PhysicsRigidbody物理功能。
- 在CustomPlayerController元件內，為了控制角色的物理功能，會宣告PhysicsRigidbodyComponent型態的變數，並連接元件參照。
- 若要對實體施加往上方向的力量，會使用PhysicsRigidbody的ApplyForce方法。
- 為了讓人知道角色處於跳躍狀態，會播放跳躍動畫。

```lua
角色跳躍功能製作範例

PhysicsRigidbodyComponent rigidBody;
StateComponent state;

number jumpForce = 730
boolean isGrounded = false


void OnBeginPlay()
{
    -- 連接要施加物理力量的PhysicsRigidbodyComponent
    self.rigidBody = self.Entity.PhysicsRigidbodyComponent
    -- 連接用來切換角色狀態的StateComponent
    self.state = self.Entity.StateComponent
}

void Jump()
{
    if self.isGround == false then 
        return -- 若角色不在地面上，則不執行跳躍功能並結束。 
    end

    -- 當開始執行跳躍時，消除角色目前受到的Y軸速度、力量。
    self.rigidBody:SetLinearVelocityY(0)

    -- 往Vector2.up角色的上方方向施加jumpForce大小的力量。
    self.rigidBody:ApplyForce(Vector2.up * self.jumpForce)

    -- 為了套用跳躍動畫，向StateComponent請求切換為Jump狀態。
    self.state:ChangeState("Jump")

    -- 因為要執行跳躍，所以切換狀態為角色目前不在地面上。
    self.isGround = false;
}

HandleKeyDownEvent(KeyDownEvent event)
{
    -- 這是在鍵盤按鍵被按下時呼叫的方法。

    -- 儲存輸入的按鍵資訊。
    local key = event.key

    -- 如果輸入的按鍵是Space鍵
    if key == KeyboardKey.Space then
    -- 則執行Jump方法。
        self:Jump()
    end
}

```

### 偵測到地面時切換角色狀態的方式
- 為了防止角色連續跳躍，使用了boolean isGround狀態變數。
- 當角色開始跳躍時，會將該變數切換為false。
- 當跳躍完成，並因重力落地到地面時，必須再將isGrounded狀態變數切換為true，恢復為可跳躍狀態。
- 可透過PhysicsCollider與EventHandler進行狀態切換。

```lua
地面碰撞狀態切換範例程式碼

StateComponent state

void OnBeginPlay()
{
    -- 連接用來切換角色狀態的StateComponent
    state = self.Entity.StateComponent
}

HandlePhysicsContactBeginEvent(PhysicsContactBeginEvent event)
{
    -- 這是在透過PhysicsCollider偵測到與其他碰撞體發生碰撞時呼叫的方法。

    -- 儲存碰撞體對象的實體。
    local ContactedBodyEntity = event.ContactedBodyEntity
    -- 從碰撞體對象取得圖塊集元件。
    local tileSet = ContactedBodyEntity.TileSet

    -- 若確認到圖塊集元件(若確認為地板圖塊)
    if tileSet ~= nil then
        -- 則將isGround狀態切換為true。
        isGround = true
        -- 將原本為跳躍狀態的動畫重新切換為步行動畫。
        self.state.ChangeState("Move")
    end
}
```

## Skill Logic的UseSkill運作原理
- 在CustomPlayerController的HandleKeyDownEvent方法中，當輸入技能發動鍵(K key)時，會呼叫SkillLogic的UseSkill方法。
- CustomPlayerController只負責接收按鍵輸入並向SkillLogic傳遞技能發動命令，不會處理具體的技能製作邏輯，以明確區分角色。
- Skill Logic會直接發動與自身登錄變數skillId編號對應的技能。
- skillId編號會在標題畫面選擇角色時變更。

```lua
Skill Logic的UseSkill原理

	local skillName = _DataSetLogic.skillInfoTable[self.skillId].Name

	elseif skillName == "Bonus" then
	-- 使用Bonus技能時，定義具體動作過程
	end 
}
```

## 撰寫SkillLogic腳本
- 必須如下撰寫程式碼。
- 該程式碼是在Plain Text環境下撰寫。

```lua
@Logic
script SkillLogic extends Logic

property number discript = 0 -- 這是為了在使用角色專屬技能時執行具體動作而宣告的邏輯。

property number skillId = 0 -- 這是用來設定角色專屬SkillId值的變數。

property boolean useable = true -- 這是用來儲存技能可使用狀態的狀態變數。

@ExecSpace("ClientOnly")
method void UseSkill()
-- 這是執行目前已登錄(SkillId)之具體動作的方法。

-- 若呼叫該方法時不是可使用的狀態
if self.useable == false then
	-- 就結束該方法。
	return
end

-- 則儲存目前角色技能名稱。
local skillName = _DataSetLogic.skillInfoTable[self.skillId].Name
-- 儲存目前角色技能固有數值。
local skillValue = _DataSetLogic.skillInfoTable[self.skillId].Value

-- 若技能名稱是'GetLife'
if skillName == "Heal" then
-- 則依技能固有數值恢復體力。
	_GameLogic:AddHP(skillValue)
	

-- 若技能名稱是'GetScore'
elseif skillName == "Bonus" then
-- 則依技能固有數值獲得額外分數。
	_GameLogic:AddScore(skillValue)
end

-- 播放技能特效。
self:PlaySkillEffect()

-- 播放技能效果音
_SoundService:PlaySound("56c1b5e3c7fc4abda561c1cc00813bda",1)

-- 因為已使用技能，所以切換為不可使用狀態。
self.useable = false

-- 將技能圖示UI渲染器的透明度值設低。
_UIManagerLogic.skillRenderer.Color.a = 0.3
end

@ExecSpace("ClientOnly")
method void Reset()
-- 這是初始化內部狀態的方法。

-- 切換為可使用技能。
self.useable = true
end

@ExecSpace("ClientOnly")
method void PlaySkillEffect()
-- 這是播放技能特效的方法。

-- 從SkillInfoTable取得儲存的技能特效RUID值。
local effect = _DataSetLogic.skillInfoTable[self.skillId].EffectRUID

-- 利用EffectService附加到角色物件上並播放技能特效。
_EffectService:PlayEffectAttached(effect,_UserService.LocalPlayer,Vector3.zero,0,Vector3.one)
end

end
```

## 在角色Slot中登錄各角色資訊
- 在執行世界時，會預先生成所有角色的Slot道具。
- 各槽位必須輸出CharaterInfo DataSet的所有要素與角色的全部資料。
- 將角色資訊登錄到Slot的過程如下。
  - 世界執行 -> 設定為在自動呼叫的OnBeginPlay內呼叫RegistCharacter方法。
  - RegistCharacter方法的功能如下。

```lua
UICharacterSelectPopup的RegistCharacter方法

Entity scrollViewEntity; -- Slot道具應配置的父實體

void RegistCharacter()
{
	-- 儲存登錄在Workspace中的Slot道具模型ID。
	local modelId = _EntryService:GetModelIdByName("UICharacterSelect")
	-- 為了讓呼叫角色資訊表的程式碼撰寫得更簡潔，將其儲存為區域變數。
	local characterTable = _DataSetLogic.characterInfoTable

	-- 依角色資訊表中登錄的數量進行巡覽。(CharacterInfo DataSet中登錄的角色數量)
	for i = 1, #characterTable, 1 do

		-- 使用SpawnService的SpawnByModelID方法生成一個Slot道具。
		local characterInfo = _SpawnService:SpawnByModelID(
			modelId,"characterInfoPopup",Vector3.zero,scrollViewEntity).UICharacterInfoComponent

		-- 將顯示角色名稱的UI文字設定為角色名稱。
		characterInfo.nameText.Text = characterTable[i].Name

		-- 設定預先登錄在Slot中的角色外型(CustomBodyEquip)資源ID值。
		characterInfo.costumImage.CustomBodyEquip = characterTable[i].RUID

		-- 取出儲存在CharacterInfo DataSet中的SkillId編號。
		local skillId = tonumber(characterTable[i].SkillId)
		-- 載入儲存在SkillInfo DataSet中的技能資訊，並登錄該技能的名稱。
		characterInfo.skillText.Text = _DataSetLogic.skillInfoTable[skillId].Name

		-- 將顯示角色捲動畫面速度的UI文字設定為角色專屬捲動畫面速度。
		characterInfo.speedText.Text = characterTable[i].Speed

	end
}
```

### 製作角色選擇視窗
<p align = "Center">
<img src = "https://develrocket-bucket.s3.ap-northeast-2.amazonaws.com/learning/MSW-100/Running/MSW-100_Running_6_009.png" alt="MSW-100_Running_6_009.png">
</p>

- 在角色選擇視窗最上層實體上，登錄用來在世界執行時執行角色Slot生成功能的元件。
- 製作完成後，在玩家點擊開始遊戲按鈕前，該彈窗不應顯示，因此先將實體停用。
- 必須如下撰寫程式碼。
- 該程式碼是在Plain Text環境下撰寫。

```lua
@Component
script UICharacterSelectPopup extends Component

property number discript = 0 -- 這是定義顯示角色選擇彈跳視窗時所需功能的元件。

property Entity scrollViewParent = nil

@ExecSpace("ClientOnly")
method void OnBeginPlay()
-- 這是在世界開始時呼叫的方法。

-- 連接將作為從模型(Prefab)生成之物件父層設定的實體。
self.scrollViewParent = _EntityService:GetEntityByPath("/ui/ToastGroup/UITitle/UICharacterSelect/UIPanel/Scroll_Layout")

-- 進行角色資訊登錄。
self:RegistCharacter()
end

@ExecSpace("ClientOnly")
method void RegistCharacter()
-- 在角色選擇視窗內登錄角色資訊的方法。
-- 為了在從模型(Prefab)生成顯示角色資訊的彈窗時，設定內部要填入的資料值。

-- 取得基本模型(Prefab)要素的模型Id。
local modelId = _EntryService:GetModelIdByName("UICharacterSelect")
-- 從資料集取得角色資訊表。
local characterTable = _DataSetLogic.characterInfoTable

-- 依表格內登錄的角色數量執行迴圈。
for	i = 1, #characterTable, 1 do
	
	-- 使用生成服務生成UI彈窗，並儲存角色資訊元件。
	local characterInfo = _SpawnService:SpawnByModelId(modelId,"characterInfoPopup",Vector3.zero,self.scrollViewParent).UICharacterInfoComponent

	-- 載入用來識別技能資訊表的skillId值。
	local skillId = tonumber(characterTable[i].SkillId)
	-- 設定要登錄在角色資訊內的角色技能編號。
	characterInfo.skillId = skillId
	
	-- 設定要登錄在角色資訊內的角色名稱。
	characterInfo.nameText.Text = characterTable[i].Name
	-- 設定要登錄在角色資訊內的服裝身體RUID值。
	characterInfo.costumeImage.CustomBodyEquip = characterTable[i].RUID
	-- 設定角色資訊內的技能名稱(根據儲存的SkillId編號從技能資訊表中取得技能名稱。)
	characterInfo.skillText.Text = _DataSetLogic.skillInfoTable[skillId].Name
	-- 設定角色資訊內的速度文字。
	characterInfo.speedText.Text = characterTable[i].Speed
	
	-- 設定角色ID(索引)值。
	characterInfo.characterId = i
end

end

end
```

### 製作角色Slot
<p align = "Center">
<img src = "https://develrocket-bucket.s3.ap-northeast-2.amazonaws.com/learning/MSW-100/Running/MSW-100_Running_6_010.png" alt="MSW-100_Running_6_010.png">
</p>

- 製作腳本，以執行配置在Slot下方的角色選擇按鈕功能。
- 使按鈕事件可以呼叫CloseSelectPopup方法。
- CloseSelectPopup方法會執行以下動作。
  - 停用建構遊戲開始時可見標題畫面的UI元素。
  - 變更玩家所使用角色的外型。
  - 將該角色的技能編號傳遞給SkillLogic。
  - 向GameLogic傳遞目前角色是第幾號角色，並呼叫開始遊戲功能。
- 必須如下撰寫程式碼。
- 該程式碼是在Plain Text環境下撰寫。

```lua
@Component
script UICharacterInfoComponent extends Component

property number discript = 0 -- 這是用來在角色選擇視窗中顯示各角色資訊的元件。

property TextComponent nameText = nil

property CostumeManagerComponent costumeImage = nil

property TextComponent skillText = nil

property TextComponent speedText = nil

property ButtonComponent selectButton = nil

property number characterId = -1 -- 這是用來區分角色索引的變數。

property number skillId = -1 -- 這是角色專屬技能編號。

@ExecSpace("ClientOnly")
method void OnBeginPlay()
-- 這是在世界開始時呼叫的方法。

-- 進行該元件初始化。
self:Init()
end

@ExecSpace("ClientOnly")
method void Init()
-- 執行CharacterInfo元件初始化的方法。

-- 執行模型(Prefab)子物件元件連接。

-- 從子物件取得用來填入角色名稱的文字元件。
self.nameText = self.Entity:GetChildByName("UIName").TextComponent
-- 從子物件取得用來登錄角色圖片的服裝元件。
self.costumeImage = self.Entity:GetChildByName("UIImage").CostumeManagerComponent
-- 從子物件取得用來填入技能名稱的文字元件。
self.skillText = self.Entity:GetChildByName("UISkill",true).TextComponent
-- 從子物件取得用來填入速度值的文字元件。
self.speedText = self.Entity:GetChildByName("UISpeed",true).TextComponent
-- 從子物件取得用來偵測滑鼠點擊的按鈕元件。
self.selectButton = self.Entity:GetChildByName("UISelectButton").ButtonComponent

-- 登錄在點擊角色選擇按鈕時將執行的方法(CloseSelectPopup)。
self.selectButton.Entity:ConnectEvent(ButtonClickEvent,self.CloseSelectPopup)

end

@ExecSpace("ClientOnly")
method void SetInfo(string name, string ruid, string skill, string speed)
-- 用來設定角色資訊視窗所需資訊、內容的方法。

-- 參數1 string name：傳入角色名稱。
-- 參數2 sting ruid：傳入角色身體RUID值。
-- 參數3 string skill：傳入角色技能名稱。
-- 參數4 string speed：傳入角色捲動畫面速度值。

-- 設定角色名稱文字。
self.nameText.Text = name
-- 設定服裝身體部位資源。
self.costumeImage.CustomBodyEquip = ruid
-- 設定技能名稱文字。
self.skillText.Text = skill
-- 設定捲動畫面速度值文字。
self.speedText.Text = speed
end

@ExecSpace("ClientOnly")
method void CloseSelectPopup()
-- 這是停用角色選擇彈跳視窗的方法。

-- 取得遊戲開始前的標題面板(開始按鈕、操作方式、角色選擇視窗的父層)實體。
local TitlePanel = _EntityService:GetEntityByPath("/ui/ToastGroup/UITitle")

-- 停用面板物件。
TitlePanel.Enable = false

-- 設定角色的服裝身體資源值。
_UserService.LocalPlayer.CostumeManagerComponent.CustomBodyEquip = self.costumeImage.CustomBodyEquip

-- 設定遊戲中要使用的角色之專屬技能Id。
_SkillLogic.skillId = self.skillId

-- 請求遊戲邏輯以所選角色Id編號進行遊戲。
_GameLogic:GameStart(self.characterId)
end

end
```

## 最低製作標準
- 玩家角色必須能正常偵測與地板圖塊的碰撞，並且不能在空中無限跳躍，而是僅限於地面狀態下才能跳躍，此限制功能必須正常運作。
- 在角色選擇視窗中，按下登錄在角色Slot中的按鈕輸入時，必須製作成可以向GameLogic呼叫GameStart。
<br>

## 應用與進階應用
- 可在CharacterInfo DataSet中，除了各角色的捲動畫面速度外，再新增跳躍力、最大體力、無敵時間等細部能力值欄位，藉此提供各角色不同的遊玩風格。
- 嘗試企劃並新增各種動作技能，例如一邊破壞障礙物一邊奔跑的「無敵突進」，或一次取得周圍所有金幣等。
- 在選擇角色時，不只是替換一個身體服裝，而是可將頭髮、臉部、披風等多種部位資料綁定變更，並進一步強化腳本。
