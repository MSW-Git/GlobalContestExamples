## Composing a Monster Summon DataSet
- Compose a DataSet that can specify the monster that must be summoned in each wave.
- The sample world has a DataSet with the name WaveData.
- WaveData is created with the following structure.
- `WaveIndex`
    - The value that determines the wave's level.
- `SpawnMonster`
    - The value that determines the monster that must be summoned in each wave.
    - Enters the MonsterID value from MonsterData for use.
- `SpawnMonsterCount`
    - The value that determines the number of monsters that need to be spawned for each wave.

### Composing Logic
- Register the SpawnData that was created to MonsterSpawnLogic, which was already created.
- Compose a function that returns the monster that matches the information of the current wave when a monster spawn is requested.
- The following is the MonsterSpawnLogic code.

```lua
@Logic
script MonsterSpawnLogic extends Logic

property table monsterData = {}

property table waveData = {}

@ExecSpace("ClientOnly")
method void OnBeginPlay()
self:InitMonsterData()
self:InitWaveData()
end

@ExecSpace("ClientOnly")
method void InitMonsterData()
-- Determines whether monsterData is unusable.
if not isvalid(self.monsterData) or not next(self.monsterData) then
	-- Confirms whether MonsterData DataSet exists normally.
	local findDataSet = _DataService:GetTable("MonsterData")
	-- Leaves an error log if the load was a failure.
	if not isvalid(findDataSet) then
		log_error("Failed to load MonsterData DataSet.")
		return
	end
	-- If the search was successful, loads the rowCount.
	local rowCount = findDataSet:GetRowCount()
	-- Repeats a number of times equal to rowCount and searches for data in order and registers them.
	for i = 1, rowCount do
		local monsterID = tostring(findDataSet:GetCell(i, "MonsterID"))
		-- Determines whether MonsterID exists normally.
		if not _UtilLogic:IsNilorEmptyString(monsterID) then
			-- If MonsterID exists normally, starts to load data after determining that it's a normal value.
			local getData = MonsterData()
			getData.MonsterID = monsterID
			getData.OneCycleMoveTime = tonumber(findDataSet:GetCell(i, "OneCycleMoveTime"))
			getData.StartHP = tonumber(findDataSet:GetCell(i, "StartHP"))
			getData.ModelID = tostring(findDataSet:GetCell(i, "ModelID"))
			if not isvalid(self.monsterData[monsterID]) then
				self.monsterData[monsterID] = {}
			end
			self.monsterData[monsterID] = getData
		end
	end
end
end

@ExecSpace("ClientOnly")
method void InitWaveData()
-- Determines whether waveData is in a usable state.
if not isvalid(self.waveData) or not next(self.waveData) then
	local findDataSet = _DataService:GetTable("WaveData")
	-- Confirms whether DataSet was found normally.
	if isvalid(findDataSet) then
		-- Loads rowCount if data exists normally.
		local rowCount = findDataSet:GetRowCount()
		-- Navigates data repetitively up to the number of rowCount.
		for i = 1, rowCount do
			local getData = WaveData()
			getData.WaveIndex = tonumber(findDataSet:GetCell(i, "WaveIndex"))
			getData.SpawnMonster = tostring(findDataSet:GetCell(i, "SpawnMonster"))
			getData.SpawnMonsterCount = tonumber(findDataSet:GetCell(i, "SpawnMonsterCount"))
			if not isvalid(self.waveData[i]) then
				self.waveData[i] = {}
			end
			self.waveData[i] = getData
		end
	end
end
end

@ExecSpace("ClientOnly")
method MonsterData GetMonsterDataByMonsterID(string MonsterID)
-- Checks if there is data that has a MonsterID as a Key.
if isvalid(self.monsterData[MonsterID]) then
	-- If data exists normally, returns the target data.
	return self.monsterData[MonsterID]
end
-- If there is no data, returns nil with an error log.
log_warning(MonsterID.."is Nil Data!")
return nil
end

@ExecSpace("ClientOnly")
method WaveData GetSpawnList(integer WaveIndex)
-- Checks if there is a WaveIndex value.
if isvalid(self.waveData[WaveIndex]) then
	-- Returns the target Data if there is a valid value.
	return self.waveData[WaveIndex]
end
end

end
```

### Creating a MonsterSpawnController
- MonsterSpawnController manages its own current wave, monster creation, and the number of remaining monsters.
- The following is the full code for MonsterSpawnLogic.

```lua
@Component
script MonsterSpawnController extends Component

property table monsterList = {}

property integer waveIndex = 0

property number standbyTime = 5

property Vector3 spawnPos = Vector3(-2.5,2.5,0)

property number timerID = 0

property number spawnEntityCount = 50

property number maxSpawnCount = 0

property number currentSpawnCount = 0

property number remainMonsterCount = 0

property number maxRemainMonsterCount = 0

property boolean isFinishData = false

property boolean pauseSpawn = false

@ExecSpace("ClientOnly")
method void OnBeginPlay()
-- Resets waveIndex to 1 at the beginning.
self.waveIndex = 1
-- Resets the current number of monsters.
_UIManageLogic:UpdateMonsterCount(self.remainMonsterCount, self.maxRemainMonsterCount)
-- Requests a wave execution.
self:InitWave()
end

@ExecSpace("ClientOnly")
method void InitWave()
-- Updates the current wave count to the UI.
_UIManageLogic:UpdateWaveCount(self.waveIndex)
-- Loads the spawn data.
local getSpawnData = _MonsterSpawnLogic:GetSpawnList(self.waveIndex)
if not isvalid(getSpawnData) then
	-- If there is no following spawn data, converts to a state without data.
	self.isFinishData = true
	-- Calls the results window, since there is no spawn data.
	_GameLogic:NeedResult()
	return 
end
-- Registers the max number of monsters spawned to the monster spawn number in the spawn data.
self.maxRemainMonsterCount = getSpawnData.SpawnMonsterCount
-- Registers the number of monsters that need to be spawned to the number of monsters spawned in the spawn data.
self.maxSpawnCount = getSpawnData.SpawnMonsterCount
-- Loads MonsterID from spawn data.
local monsterID = getSpawnData.SpawnMonster
-- Resets the available spawns to 0.
local canSpawnCount = 0
-- Registers to the monsterList that has monsters that needs to use spawnList as Keys.
local spawnList = self.monsterList[monsterID]

-- Determines and creates a monster pool.
if not isvalid(spawnList) or not next(spawnList) then
	-- If the pool is empty, creates a number equal to the amount in the early setting.
	for i = 1, self.spawnEntityCount do
		self:SpawnMonster(monsterID)
	end
else
	-- Deactivates if there's already a pool and analyzes the number of immediately-possible spawns.
	for i = 1, #spawnList do
		if not spawnList[i].Enable then 
			canSpawnCount = canSpawnCount + 1
            end
	end
        
	-- If the available number is lower than the wave's max spawn amount, an amount equal to the difference is additionally spawned.
	if canSpawnCount < self.maxSpawnCount then
		local needCount = self.maxSpawnCount - canSpawnCount
		for i = 1, needCount do
			self:SpawnMonster(monsterID)
		end
	end
end

-- Uses Timer for monster generation so that they are generated at a set time.
self.timerID = _TimerService:SetTimerRepeat(function()
self:PlayMonster(monsterID) end, 1.0, self.standbyTime)	
end

@ExecSpace("ClientOnly")
method void SpawnMonster(string MonsterID)
-- Loads monster information based on the received MonsterID.
local getMonsterData = _MonsterSpawnLogic:GetMonsterDataByMonsterID(MonsterID)
-- Halts the code execution if unusable MonsterData comes in.
if not isvalid(getMonsterData) then 
	return 
end
-- Searches for and spawns the monster Model based on the found data.
local modelEntity = _SpawnService:SpawnByModelId(getMonsterData.ModelID, getMonsterData.MonsterID, self.spawnPos, self.Entity)
    
-- Hides it from the screen first before putting it into the pool.
modelEntity:SetEnable(false) 
-- Registers the monster information to the created modelEntity.
modelEntity.MonsterController:InitData(getMonsterData, self.spawnPos, self)
-- Resets if no monsterList that has MonsterIDs as Keys exists.
if not isvalid(self.monsterList[MonsterID]) then
	self.monsterList[MonsterID] = {}
end
-- Registers the created monster entity to monsterList using MonsterID as the Key.
table.insert(self.monsterList[MonsterID], modelEntity)
end

@ExecSpace("ClientOnly")
method void PlayMonster(string MonsterID)
-- Registers the elements that use MonsterIDs and Keys delivered from monsterList to the spawnList.
local spawnList = self.monsterList[MonsterID]
    
-- When the timer is called every second, searches for a single monster that is on standby and spawns it.
for i = 1, #spawnList do
	local targetMonster = spawnList[i]
        
	-- Searches for a monster that's not on the current screen and is resting.
	if not targetMonster.Enable then
		targetMonster:SetEnable(true)
		targetMonster.MonsterController:SetEnableState() 
		-- Increases the current monster spawn count by increasing the spawn count.
		self.currentSpawnCount = self.currentSpawnCount + 1 
		-- Increases remainMonsterCount, which is a value needed for the UI display.
		self.remainMonsterCount = self.remainMonsterCount + 1
		-- Requests UIManageLogic to update the current status.
		_UIManageLogic:UpdateMonsterCount(self.remainMonsterCount, self.maxRemainMonsterCount)
		-- Only spawns one and ends the loop.
		break 
	end
end

-- Confirms the conditions for the conversion of the next wave if all of the target numbers of the wave have been spawned.
if self.currentSpawnCount >= self.maxSpawnCount then
	self.currentSpawnCount = 0
	-- Stops the timer that manages the spawns.
	_TimerService:ClearTimer(self.timerID)
        
	if self.remainMonsterCount <= 0 then
		-- Moves onto the next wave immediately if there are no more monsters left on the field.
		self.waveIndex = self.waveIndex + 1
		self:InitWave()
	else
		-- Enters the spawn standby state (pauseSpawn) if there are any monsters left on the field.
		self.pauseSpawn = true
		_GameLogic:NeedTimer()
	end
end
end

@ExecSpace("ClientOnly")
method void UpdateRemainMonsterCount(number ChangeValue)
-- Adds the value received to remainMonsterCount. (Normally -1 is delivered when a monster is defeated.)
self.remainMonsterCount = self.remainMonsterCount + ChangeValue
    
-- It's safer to add defensive code so that the remaining monster count doesn't drop below 0.
if self.remainMonsterCount < 0 then 
	self.remainMonsterCount = 0 
end

-- Requests that the changed value be updated to the UI.
_UIManageLogic:UpdateMonsterCount(self.remainMonsterCount, self.maxRemainMonsterCount)

Starts the next wave if all remaining monsters have been defeated (0) during a spawn standby status (pauseSpawn).
if self.pauseSpawn == true and self.remainMonsterCount == 0 then
	self.pauseSpawn = false -- Disable standby state
	self.waveIndex = self.waveIndex + 1 -- Increase wave
	_GameLogic:DisableTimer()
	self:InitWave() -- Start the next wave spawn
end
end

@ExecSpace("ClientOnly")
method void ChangeResultUI()
-- Iterates the entire monsterList.
for monsterID, spawnList in pairs(self.monsterList) do
	-- Iterates the entities that are contained within each ID list.
	for i = 1, #spawnList do
		local monster = spawnList[i]
		-- Confirms whether the entity is valid and then deactivates it.
		if isvalid(monster) then
			monster:SetEnable(false)
		end
	end
end
_TimerService:ClearTimer(self.timerID)
end

@ExecSpace("ClientOnly")
method void CallRestart()
-- Executes InitWave after resetting all values.
self.isFinishData = false
self.pauseSpawn = false
self.remainMonsterCount = 0
self.currentSpawnCount = 0
self.waveIndex = 1
self:InitWave()
end

end
```
