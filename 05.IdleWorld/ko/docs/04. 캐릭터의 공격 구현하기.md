## 스킬 데이터 제작하기

- 캐릭터가 스킬을 사용하기 위한 DataSet을 제작합니다.
- 앞서 제작한 CharacterData DataSet의 SkillData 항목에 값을 입력하면 스킬을 변경할 수 있도록 구성합니다.
- 이를 통해 캐릭터의 스킬을 DataSet 변경만으로 부여할 수 있도록 구조를 설계합니다.

<br>

### DataSet 구성요소
- `SkillID`
    - 스킬을 구분하기 위한 값입니다.
    - 해당 값을 CharacterData DataSet의 SkillData에 입력하여 사용합니다.
- `SkillEffect`
    - 캐릭터가 적에게 스킬을 시전할 때, 적 위치에 출력되는 이펙트입니다.
    - 스킬의 이펙트 리소스 ID를 등록하기 위한 값입니다.
- `CharacterSkillEffect`
    - 캐릭터가 적에게 스킬을 시전할 때, 캐릭터에게 출력되는 이펙트입니다.
    - 스킬의 이펙트 리소스 ID를 등록하기 위한 값입니다.
- `SkillAttack`
    - 스킬의 공격력을 지정하기 위한 값입니다.
- `SkillCoolTime`
    - 스킬의 쿨타임을 지정하기 위한 값입니다.

## 캐릭터의 공격 스크립트 살펴보기

- 캐릭터의 Event Handler를 통해 몬스터가 캐릭터의 공격 범위에 있는지, 벗어났는지 관리합니다.
- 아래는 캐릭터를 관리하는 UnitController Component의 코드입니다.

```lua
@ExecSpace("ClientOnly")
@EventSender("Self")
handler HandleTriggerEnterEvent(TriggerEnterEvent event)
-- 충돌 범위에 들어온 엔티티가 CollisionGroups, Monster일 경우 이를 등록합니다.
if event.TriggerBodyEntity.TriggerComponent.CollisionGroup == CollisionGroups.Monster then
	if #self.enemyTable >= 1 then
		local isAlreadyHave = false
		for i = 1, #self.enemyTable do
			if self.enemyTable[i] == event.TriggerBodyEntity then
				isAlreadyHave = true
				break
			end
		end
		if isAlreadyHave == false then
			table.insert(self.enemyTable, event.TriggerBodyEntity)
		end
	else
		table.insert(self.enemyTable, event.TriggerBodyEntity)
	end
end
end

@ExecSpace("ClientOnly")
@EventSender("Self")
handler HandleTriggerLeaveEvent(TriggerLeaveEvent event)
-- 충돌 범위를 벗어난 엔티티가 Monster일 경우 공격 리스트에서 제거합니다.
if event.TriggerBodyEntity.TriggerComponent.CollisionGroup == CollisionGroups.Monster then
	if #self.enemyTable >= 1 then
		local listCount = #self.enemyTable
		for i = listCount, 1, -1 do
			if self.enemyTable[i] == event.TriggerBodyEntity then
				table.remove(self.enemyTable, i)
				break
			end
		end
	end
end
end
```

> 해당 코드는 Plain Text를 기준으로 작성된 코드입니다.

- `HandleTriggerEnterEvent`를 통해 몬스터가 캐릭터의 공격 범위에 들어왔는지 체크합니다.
- `HandleTriggerExitEvent`를 통해 몬스터가 공격 범위를 벗어날 경우 공격 가능 리스트에서 제거합니다.

```lua
@ExecSpace("ClientOnly")
method void OnUpdate(number delta)
-- 델타타임은 조건 없이 항상 더해줍니다.
self.skillDelta = self.skillDelta + delta
self.basicAtkDelta = self.basicAtkDelta + delta

-- 스킬 공격 쿨타임을 확인하여 조건을 충족할 경우 스킬 공격을 수행합니다.
if self.skillDelta >= self.characterSkillData.SkillCoolTime then
	self:SkillAttack()
end

-- 만약 스킬 공격이 성공하여 쿨타임이 0으로 초기화되었다면, 이번 프레임엔 기본 공격을 쉬어갑니다.
if self.skillDelta == 0 then
	return
end

-- 기본 공격 쿨타임을 확인하여 조건을 충족할 경우 기본 공격을 수행합니다.
if self.basicAtkDelta >= self.characterData.BasicAttackCoolTime then
	self:BasicAttack()
end
end

@ExecSpace("ClientOnly")
method void BasicAttack()
-- 캐릭터의 StateComponent를 가져옵니다.
local bodySelector = self.Entity.StateComponent
-- 캐릭터의 현재 상태 이름을 가져옵니다.
local currentState = bodySelector.CurrentStateName
-- 현재 상태가 IDLE이 아닐 경우, 기본 공격 수행을 하지 않습니다.
if currentState ~= "IDLE" then
	return
end
---@type Entity
local targetEntity = nil
-- 테이블을 역순으로 순회하며 유효하지 않은 엔티티를 안전하게 제거합니다.
for i = #self.enemyTable, 1, -1 do
	local enemy = self.enemyTable[i]
	-- 엔티티가 아직 월드에 존재하는지 확인합니다.
	if isvalid(enemy) then
		-- 엔티티에 MonsterController가 있고 살아있는지 확인합니다.
		if enemy.MonsterController and enemy.MonsterController.isAlive then
			targetEntity = enemy
			break
		end
	else
		-- 이미 파괴된 엔티티라면 테이블에서 제거합니다.
		table.remove(self.enemyTable, i)
	end
end
-- 만약 지정된 targetEntity가 사용 불가능할 경우 공격을 수행하지 않습니다.
if not isvalid(targetEntity) then
	return
end
-- 공격을 수행했다 가정하여 basicAtkDelta값을 0으로 설정합니다.
self.basicAtkDelta = 0
-- 공격 애니메이션을 출력시키기 위해 event를 생성, 전달합니다.
local event = ActionStateChangedEvent(self.characterData.AttackAnimationID, self.characterData.AttackAnimationID, 1, SpriteAnimClipPlayType.Onetime)
self.avatarBodyComponent:SendEvent(event)
-- 출력해야 하는 캐릭터쪽 이펙트가 있을 경우 이를 출력합니다.
if not _UtilLogic:IsNilorEmptyString(self.characterData.BasicAttackEffectID) then
	_EffectService:PlayEffect(self.characterData.BasicAttackEffectID, self.Entity, self.Entity.TransformComponent.WorldPosition, 0, Vector3.one)	
end
-- 적에게 출력해야 하는 이펙트가 있을 경우 이를 출력합니다.
if not _UtilLogic:IsNilorEmptyString(self.characterData.BasicAttackHitEffectID) then
	_EffectService:PlayEffect(self.characterData.BasicAttackHitEffectID, targetEntity, targetEntity.TransformComponent.WorldPosition, 0, Vector3.one)		
end
-- 타겟이 된 적의 MonsterController를 통해 GetDamage를 수행하도록 요청합니다.
targetEntity.MonsterController:GetDamage(self.characterData.BasicAttackValue)
end

@ExecSpace("ClientOnly")
method void SkillAttack()
-- 애니메이션 재생을 위한 bodySelector를 찾아 등록합니다.
local bodySelector = self.Entity.StateComponent
-- 현재 상태를 찾기 위해 CurrentStateName을 등록합니다.
local currentState = bodySelector.CurrentStateName
-- 현재 상태가 IDLE이 아닐 경우 코드 실행을 중단합니다.
if currentState ~= "IDLE" then
	return
end
---@type Entity
local targetEntity = nil
-- 테이블을 역순으로 순회하며 유효하지 않은 엔티티를 안전하게 제거합니다.
for i = #self.enemyTable, 1, -1 do
	local enemy = self.enemyTable[i]
	if isvalid(enemy) then
		if enemy.MonsterController and enemy.MonsterController.isAlive then
			targetEntity = enemy
			break
		end
	else
		table.remove(self.enemyTable, i)
	end
end
-- 만약 targetEntity가 nil일 경우 코드 실행을 중단합니다.
if not isvalid(targetEntity) then
	return
end
-- 스킬 사용을 성공했다 가정하여 skillDelta값을 0으로 초기화합니다.
self.skillDelta = 0
-- 공격 애니메이션을 재생합니다.
local event = ActionStateChangedEvent(self.characterData.AttackAnimationID, self.characterData.AttackAnimationID, 1, SpriteAnimClipPlayType.Onetime)
self.avatarBodyComponent:SendEvent(event)
-- 스킬 사용시 자기 자신에게 출력해야 하는 이펙트가 있을 경우 이펙트를 출력합니다.
if not _UtilLogic:IsNilorEmptyString(self.characterSkillData.CharacterSkillEffect) then
	_EffectService:PlayEffect(self.characterSkillData.CharacterSkillEffect, self.Entity, self.Entity.TransformComponent.WorldPosition, 0, Vector3.one)	
end
-- 피격 대상에게 출력해야 하는 이펙트가 있을 경우 이펙트를 출력합니다.
if not _UtilLogic:IsNilorEmptyString(self.characterSkillData.SkillEffect) then
	_EffectService:PlayEffect(self.characterSkillData.SkillEffect, targetEntity, targetEntity.TransformComponent.WorldPosition, 0, Vector3.one)		
end
-- 대상이 된 targetEntity의 MonsterContorller에게 GetDamage를 수행하도록 요청합니다.
targetEntity.MonsterController:GetDamage(self.characterSkillData.SkillAttack)
end
```

> 해당 코드는 Plain Text로 작성된 코드입니다.

- OnUpdate를 통해 기본 공격의 쿨타임, 스킬 공격의 쿨타임을 계산합니다.
- 만약 스킬 쿨타임이 일정 시간에 도달한다면 스킬 공격을 우선합니다.
- 현재 스킬 공격을 수행중일 경우 기본 공격을 하지 않도록 예외처리를 진행합니다.
- 만약 스킬 공격이 대기 상태이며 기본 공격이 가능할 경우 기본 공격을 수행합니다.

## 몬스터의 피격 살펴보기

- 몬스터의 행동을 제어하는 MonsterController 내에 몬스터가 피격될 때 호출되는 Method가 존재합니다.
- 해당 코드는 MonsterController의 코드입니다.

```lua
@ExecSpace("ClientOnly")
method void GetDamage(number AtkValue)
-- 받은 데미지를 표기합니다.
_DamageSkinService:Play(self.Entity, "3271c3e79bf04ecba9a107d55495970d", 0.05, {AtkValue}, DamageSkinTweenType.Default, false)
-- 현재 체력에서 전달받은 데미지만큼 감소시킵니다.
self.currentHP = self.currentHP - AtkValue
-- 만약 현재 체력이 0 이하일 경우 처치 관련 처리를 진행합니다.
if self.currentHP <= 0 then
	-- 오버플로우 방지를 위해 0으로 초기화합니다.
	self.currentHP = 0
	-- 현재 생존 여부를 false로 변경합니다.
	self.isAlive = false
	-- 몬스터의 이동을 중단시킵니다.
	self.tweenLineComponent:Pause()
	-- 몬스터의 애니메이션을 사망으로 변경합니다.
	self.monsterSpriteRenderer.SpriteRUID = self.dieAnimationKey
	-- HPBar의 크기를 0으로 초기화합니다.
	self.hpBar.TransformComponent.Scale.x = 0
	-- 자기 자신이 처치되었으므로 MonsterSpawnManager에게 잔존 수를 감소하도록 요청합니다.
	self.spawnController:UpdateRemainMonsterCount(-1)
	-- GameLogic에게 소환 토큰을 1 증가시키도록 요청합니다.
	_GameLogic:UpdateSummonTokken(1)
else
	-- 이외의 경우에는 설정한 크기에 맞춰 UI가 변경되도록 계산한 값을 적용합니다.
	self.hpBar.TransformComponent.Scale.x = self.currentHP / self.monsterData.StartHP * 20
end
end
```

> 해당 코드는 Plain Text를 기준으로 작성된 코드입니다.

- AtkValue로 캐릭터가 주는 데미지를 받아옵니다.
- 자기 자신의 체력에서 전달 받은 데미지만큼 감소시킵니다.
- 현재 체력이 0일 경우, 사망과 관련된 처리를 수행합니다.
- HPBar를 감소시켜 남은 체력을 표기합니다.

## 최소 구현 기준

- 캐릭터가 몬스터를 감지하고 공격할 수 있습니다.
- 데미지 계산을 통해 몬스터의 상태를 생존과 사망상태로 전환할 수 있습니다.
- 캐릭터가 스킬을 사용하여 공격을 할 수 있습니다.

<br>

## 응용 및 확장 방향

- 캐릭터의 스킬을 범위형 공격을 제작하여 범위내 적들에게 공격하는 기능을 제작합니다.
- 캐릭터의 공격에 따라 사운드의 출력, 이펙트의 변경 등 시각적, 청각적 효과를 강화합니다.
- 몬스터를 공격하는 기준을 자신과 가장 가까운 적으로 공격 상대를 결정하도록 기능을 수정합니다.
