## 構成技能實體
- 為了製作技能Model，必須構成實體。
- 如下在技能Entity上構成Component。

</p>

- **SpriteRendererComponent**：用來顯示技能圖片的Component。
- **TriggerComponent**：用來檢查技能是否發生碰撞的Component。必須將該Component的CollisionGroup設為Skill。
- **ProjectileSkillComponent**：用來控制投射體技能資料與行為的Component。必須透過Add Component來製作。
<br>

## 構成腳本
- 必須如下構成ProjectileSkillComponent。
- 以下程式碼是以Plain Text為基準撰寫。

```lua
@Component
script ProjectileSkillComponent extends Component

property table collideTable = {} -- 用來管理技能投射體正在碰撞中的實體的table。

property number atkValue = 0 -- 用來儲存技能投射體攻擊力的Property。

property number hitInterval = 0 -- 用來調整投射體技能多段Hit間隔的Property。

property number maxHitCount = 0 -- 用來調整投射體技能最大Hit間隔的Property。

property boolean isPierce = false -- 用來決定投射體技能是否可貫穿的Property。

property number lifeTime = 0 -- 用來決定投射體技能持續時間的Property。

property number moveSpeed = 0 -- 用來決定投射體技能移動速度的Property。

property string hitEffectRUID = "" -- 用來儲存被投射體技能命中時要顯示的特效ID的Property。

property number deltaTimer = 0 -- 用來計算投射體技能持續時間的Property。

property Vector2 fireDirection = Vector2(0,0) -- 用來儲存投射體應前進方向的Property。

@ExecSpace("Client")
method void InitData(table SkillData)
--檢查傳入的技能資料是否有效。
if not isvalid(SkillData) or next(SkillData) == nil then
	log("傳入的SkillData是nil！")
	return
end
-- 進行技能資料初始化。
--初始化用來檢查投射體技能壽命的deltaTimer。
self.deltaTimer = 0 
-- 初始化存放已碰撞怪物的table。
self.collideTable = {} 
-- 初始化技能攻擊力數值。
self.atkValue = SkillData.AtkValue 
-- 初始化技能攻擊延遲數值。
self.hitInterval = SkillData.HitInterval
 -- 初始化技能最大攻擊次數。
self.maxHitCount = SkillData.MaxHitCount
 -- 初始化技能持續時間。
self.lifeTime = SkillData.LifeTime 
-- 初始化技能移動速度。
self.moveSpeed = SkillData.MoveSpeed
 -- 檢查技能是否可貫穿。
self.isPierce = SkillData.IsPierce 
-- 初始化命中技能時要顯示的特效RUID。
self.hitEffectRUID = SkillData.HitEffectRUID 
-- 先停用投射體技能的使用狀態。
self.Entity.Enable = false
end

@ExecSpace("Client")
method table GetOrCreateCollideEntity(Entity CollideEntity)
-- 若CollideEntity可使用，則執行以下程式碼。
if isvalid(CollideEntity) then
	-- 取得CollideEntity的ID。
	local entityID = CollideEntity.Id
	-- 若collideTable中沒有持有該ID值的實體，則新登錄到collideTable中。
	if not isvalid(self.collideTable[entityID]) or not next(self.collideTable[entityID]) then
		self.collideTable[entityID] = {
			Target = CollideEntity,
			HitCount = 0,
			LastHitTime = nil
		}
	end
	-- 回傳持有CollideEntity ID的實體。
	return self.collideTable[entityID]
end
end

@ExecSpace("Client")
method void ApplyDamage(table CheckData)
-- 若已死亡而不存在實體，則將該ID的Table Value改為nil。
if not isvalid(CheckData.Target) then
	self.collideTable[CheckData.Target.Id] = nil
	return
end

-- 檢查目標是否有MonsterState 元件後，對該元件執行傷害計算。
if isvalid(CheckData.Target.MonsterState) then
	CheckData.Target.HitComponent:OnHit(self.Entity, self.atkValue, false, self.hitEffectRUID, 1)
end

-- 登錄最後命中時間。
CheckData.LastHitTime = _UtilLogic.ElapsedSeconds
-- 該實體可承受的傷害次數增加1。
CheckData.HitCount = CheckData.HitCount + 1
end

@ExecSpace("ClientOnly")
method void OnUpdate(number delta)
-- 若遊戲的Play狀態為false，則中止執行以下程式碼。
if not _GameLogic.IsStillPlay then
	return
end
-- 請求投射體技能移動。
self:Move(delta)
-- 請求計算投射體的持續時間。
self:RunLifeTimer(delta)

end

@ExecSpace("Client")
method void RunLifeTimer(number DeltaTime)
-- 將投射體技能的deltaTimer增加DeltaTime。
self.deltaTimer = self.deltaTimer + DeltaTime
-- 若deltaTimer超過lifeTime，則停用使用狀態。
if self.deltaTimer >= self.lifeTime then
	self.Entity.Enable = false
end

end

@ExecSpace("Client")
method void Move(number DeltaTime)
-- 取得實體目前位置。
local transform = self.Entity.TransformComponent
-- 以投射體技能的移動速度與DeltaTime來計算位置。
local moveVec = Vector3(self.fireDirection.x, self.fireDirection.y, 0) * self.moveSpeed * DeltaTime
-- 變更投射體技能的位置。
transform.Position = transform.Position + moveVec
end

@ExecSpace("Client")
method void FireSkill(Vector2 FireDirection, Vector3 FireStartPos)
-- 將用於計算投射體技能持續時間的deltaTimer值初始化為0。
self.deltaTimer = 0
-- 初始化投射體技能的發射方向。
self.fireDirection = FireDirection
-- 初始化投射體技能的發射位置。
local transform = self.Entity.TransformComponent  
if transform then
	transform.Position = FireStartPos
end
-- 將投射體的使用狀態切換為true。
self.Entity.Enable = true
end

@EventSender("Self")
handler HandleTriggerStayEvent(TriggerStayEvent event)
-- 確認正在碰撞的Entity是否正常存在。
if isvalid(event.TriggerBodyEntity) then
	--若正在碰撞的Entity是投射體，則刪除敵方投射體。
	if isvalid(event.TriggerBodyEntity.MissileComponent) and not event.TriggerBodyEntity.MissileComponent.isShotPlayer then
		event.TriggerBodyEntity.MissileComponent:ReturnSelf()
		return
	end
	--若正在碰撞的實體是敵人，則在滿足Hit次數的情況下執行傷害計算。
	if isvalid(event.TriggerBodyEntity.MonsterState) then
		-- 從collideTable中回傳與正在碰撞的實體相同的實體。
		local collideTable = self:GetOrCreateCollideEntity(event.TriggerBodyEntity)
		-- 若回傳的實體不可使用，則中止執行程式碼。
		if not isvalid(collideTable) then
			return
		end
		-- 若HitCount超過最大值，則中止執行程式碼。
		if collideTable.HitCount >= self.maxHitCount then
			return
		end
		-- 取得目前時間。
		local currentTime = _UtilLogic.ElapsedSeconds
		-- 若一次傷害計算都尚未進行，則立即造成傷害。
		if collideTable.LastHitTime == nil then
			self:ApplyDamage(collideTable)
		-- 若受傷時間大於多段Hit時間，則嘗試進行傷害計算。
		elseif currentTime - collideTable.LastHitTime >= self.hitInterval then
			self:ApplyDamage(collideTable)
		end
	end
end
end

@EventSender("Self")
handler HandleTriggerEnterEvent(TriggerEnterEvent event)
-- 若event不正常，則輸出錯誤日誌。
if not isvalid(event) then
	log("event不是正常的event！")
	return
end
-- 若碰撞的實體持有MissileComponent，且發射主體不是玩家，則停用該碰撞實體。
if isvalid(event.TriggerBodyEntity.MissileComponent) and not event.TriggerBodyEntity.MissileComponent.isShotPlayer then 
	event.TriggerBodyEntity.MissileComponent:ReturnSelf()
	return
end
-- 當碰撞的實體持有MonsterState時，執行以下程式碼。
if isvalid(event.TriggerBodyEntity.MonsterState) then
	-- 檢查collideTable中是否有與碰撞實體相同的實體。
	local findList = self:GetOrCreateCollideEntity(event.TriggerBodyEntity)

	-- 若碰撞實體受到傷害的次數已超過最大值，則中止執行程式碼。
	if findList.HitCount >= self.maxHitCount then return end

	-- 立即請求對碰撞實體進行傷害處理。
	self:ApplyDamage(findList)
end
end

@EventSender("Self")
handler HandleTriggerLeaveEvent(TriggerLeaveEvent event)
-- 若離開碰撞範圍的實體為不可使用狀態，則中止執行程式碼。
if not isvalid(event) then 
	return
end
-- 尋找離開碰撞範圍的實體是否存在於collideTable內。
local record = self.collideTable[event.TriggerBodyEntity.Id]
-- 若該實體存在於collideTable內，則初始化最後命中時間，讓它再次進入碰撞範圍時可立即受到傷害。
if record then
    record.LastHitTime = nil
end
end

end
```

- 完成上述作業後，必須將技能實體模型化，並將Model的EntryID登錄到SkillData的ModelRUID中。
<br>

## HandleTrigger事件的呼叫時機
- **EnterEvent**：當2個以上持有TriggerComponent的實體之碰撞範圍重疊的瞬間呼叫。
- **StayEvent**：當2個以上持有TriggerComponent的實體之碰撞範圍持續重疊時，會在每一幀呼叫。
- **LeaveEvent**：當2個以上持有TriggerComponent的實體原本正在碰撞，而碰撞範圍重疊解除的瞬間呼叫。
<br>

## 最低實作標準
- 玩家可以使用技能實體。
- 技能實體可以消除敵方投射體。
- 投射體技能可以攻擊多個敵人，並能依指定次數攻擊敵人。
