## Creating a DataSet
- You need 2 types of DataSets when creating a function related to item drops.
- There are the ItemData and ItemDropData sets.
    - **ItemData**: The DataSet that determines an item's effect or its efficacy.
    - **ItemDropData**: The DataSet that determines an item's chance of dropping.

## Creating an ItemSpawnManager
- The Component that's in charge of actual item creation.
- This element is created in the form of an entity, and it is created in InGame in the sample world.
- Create an entity and Component and register the following code to the ItemSpawnManager.
- The code is written based on Plain Text.

```lua
@Component
script ItemSpawnManager extends Component

property string itemModelID = "model://0b6340f2-19d9-4a27-b18a-70feb94ce2ae" -- The Model ID for use as an item within Workspace. 

property table itemObjectPool = {} -- Table for searching and re-using usable entities.

@ExecSpace("Client")
method void CreateItem(string ItemKey, Vector3 ItemSpawnPosition)
-- Determines whether the itemObjectPool is nil.
if self.itemObjectPool[ItemKey] == nil then
	-- Resets to {} if the itemObjectPool is nil.
	self.itemObjectPool[ItemKey] = {}
end

-- Registers ItemKey entities received from within the pool to the pool.
local pool = self.itemObjectPool[ItemKey]
-- Sets returnEntity to nil to register usable entities.
local returnEntity = nil

-- Starts the navigation if there is at least 1 pool element.
if #pool > 0 then
-- Starts searches equal to the number of pool elements.
	for i = 1, #pool do
		-- Returns the entity at index i if the index i entity is in a usable state.
		if isvalid(pool[i]) and not pool[i].Enable then
			returnEntity = pool[i]
			break
		end
	end
end

-- Searches for values based on the ItemKey value from DataTableLogic.
local itemData = _DataTableLogic:GetItemDataByItemID(ItemKey)
-- Determines that there are no usable entities if the returnEntity has been nil so far.
if returnEntity == nil then
	-- Creates an item entity that was created with Model on the map through SpawnService.
    returnEntity = _SpawnService:SpawnByModelId(self.itemModelID, itemData.ItemID, Vector3.zero, self.Entity)
	-- Registers itself to the ItemComponent's poolManager, so the created item can be returned again.
	returnEntity.ItemComponent.poolManager = self
	-- Changes the entity's Enable to true just in case the item is in an unusable state.
	returnEntity.Enable = true
	-- Adds the element created in itemObjectPool ItemKey value to Key and entity to Value.
	table.insert(self.itemObjectPool[ItemKey], returnEntity)
else
    -- Changes Enable to true if there is a returnEntity, so it can be used.
    returnEntity.Enable = true
end
-- Resets the created item entity's data.
returnEntity.ItemComponent:InitData(itemData, ItemSpawnPosition)
end

@ExecSpace("Client")
method void ReturnItemEntity(Entity ItemEntity)
-- Exits code through return if the received itemEntity is an unusable entity.
if not isvalid(ItemEntity) then return end
-- Deactivates Enable and converts to standby mode, so it can be used if the ItemEntity can be used.
ItemEntity.Enable = false
end

end
```

## Writing an ItemComponent Script
- Write the following code to ItemComponent.
- The code is written based on Plain Text.

```lua
@Component
script ItemComponent extends Component

property SpriteRendererComponent imageSpriteRenderer = nil

property number itemID = 0 -- Property that saves the ItemData's ID value.

property number valueType = 0 -- Property that classifies the ItemData's application type.

property string targetPlayerStateName = "" -- Property that holds the name of the Property that needs to be applied in the PlayerState component.

property number numberValue = 0 -- Property that holds the number values when number format values must be applied.

property string stringValue = "" -- Property that holds text values when text format values must be applied.

property number moveSpeed = 3.0 -- Property that saves the item's movement speed in the map.

property number lifeTime = 3.0 -- Property that holds the amount of time the item can stay on the map.

property string gainEffect = "" -- Property that holds the effect ID that's generated upon obtaining the item.

property any poolManager = nil

property number deltaTimer = 0 -- Property that calculates an item's duration.

@ExecSpace("Client")
method void InitData(table ItemData, Vector3 SpawnPosition)
-- Starts a reset if the Property that manages the item's image sprite is nil.
if not isvalid(self.imageSpriteRenderer) then
	-- Searches for a SpriteRendererComponent from an entity that has the component attached to it.
	local findImageSpriteRenderer = self.Entity.SpriteRendererComponent
	if not isvalid(findImageSpriteRenderer) then
		-- Generates a log that says the search failed if the SpriteRendererComponent wasn't found.
		log("SpriteRendererComponent doesn't exist in the item model!")
		return
	end
	-- If the SpriteRenderer is found, the found component is registered to Property.
	self.imageSpriteRenderer = findImageSpriteRenderer
end

-- Returns an error log when the received ItemData is nil and does not attempt a reset.
if not isvalid(ItemData) or not next(ItemData) then
	log("The ItemData received from Init is nil!")
	return
end

-- Resets the deltaTimer to 0 and calculates the duration.
self.deltaTimer = 0
-- Registers the received ItemData data to the Component's Property.
-- Registers the item icon image RUID.
self.imageSpriteRenderer.SpriteRUID = ItemData.SpriteRUID
-- Registers the Value that determines the item's influence. 
self.valueType = ItemData.ValueType
-- Registers values with names such as the PlayerState's Property.
self.targetPlayerStateName = ItemData.TargetPlayerStateName 
-- Registers the increased amount for items that have changes in number-type values.
self.numberValue = ItemData.NumberValue
-- Registers the String value for items that have changes in the String type Key values, such as default missiles.
self.stringValue = ItemData.StringValue
-- Registers the ID value of the effect that will be generated upon obtaining an item. 
self.gainEffect = ItemData.GainEffect
 -- Receives the location where the monster died and spawns at that location.
self.Entity.TransformComponent.Position = SpawnPosition 
-- If the entity is in an inactive state, activates it.
if not self.Entity.Enable then
	self.Entity.Enable = true
end
end

@ExecSpace("ClientOnly")
method void OnUpdate(number delta)
-- If the entity is in an inactive state, the script below isn't executed through the Return immediately.
if not self.Entity.Enable then
	return
end

-- If the game's play state isn't in-play, the script below isn't executed through the Return immediately.
if not _GameLogic.IsStillPlay then
	return
end
-- Deactivates an item if the deltaTimer goes over the item's survival duration.
if self.deltaTimer > self.lifeTime then
	self:ReturnSelf()
	return
else
	-- Increases the deltaTimer equal to the amount elapsed if the deltaTimer hasn't gone over the lifeTime.
	self.deltaTimer = self.deltaTimer + delta	
end

-- Calls the item's movement function every frame.
self:ItemMove(delta)
end

@ExecSpace("Client")
method void ItemMove(number DeltaTime)
-- Loads the TransformComponent that manages the item's position.
local transform = self.Entity.TransformComponent

-- Calculates the distance that the item must travel.
local moveVec = Vector3(-1, 0, 0) * self.moveSpeed * DeltaTime

-- Moves the item the distance it must travel.
transform.Position = transform.Position + moveVec
end

@ExecSpace("Client")
method void ReturnSelf()
    -- Changes the item's Enable to false when the LifeTime is over.
    if self.poolManager and isvalid(self.poolManager) then
        self.poolManager:ReturnItemEntity(self.Entity)
    else
	-- Destroys the item entity if there is no poolManager.
        _EntityService:Destroy(self.Entity)
    end
end

@EventSender("Self")
handler HandleTriggerEnterEvent(TriggerEnterEvent event)
-- Processes item use if the collided entity has a PlayerState.
if isvalid(event.TriggerBodyEntity.PlayerState) then
	-- Executes the AppliedItem method of the collided player entity.
	event.TriggerBodyEntity.PlayerState:AppliedItem(self.valueType, self.targetPlayerStateName, self.stringValue, self.numberValue, self.gainEffect)
	-- Deactivates the target entity.
	self:ReturnSelf()
end 
end

end
```

## Creating an ItemHelper
- The ItemHelper is a Logic that requests item creation when a monster dies.
- When the ItemHelper receives a request, and if an item needs to be dropped, then an item creation request is sent to the ItemSpawnManager.
- The Logic was created with the name ItemHelper in the sample world.
- The code is written based on Plain Text.

```lua
@Logic
script ItemHelper extends Logic

property ItemSpawnManager itemSpawnManager = nil

method void CalcDropItemData(number ItemDropRate, string MustDropItemKey, Vector3 SpawnPosition)
-- Returns nil without executing the probability calculation, since the item drop rate is 0.
if not isvalid(ItemDropRate) or ItemDropRate <= 0.0 then
	return
end

-- Determines the item drop probability in any other case.
if ItemDropRate >= 1.0 then
	-- No additional calculations are made, since the item must be dropped.
else
	-- Executes random based on the ItemDropRate value.
	local randomValue = _UtilLogic:RandomDouble()
	if ItemDropRate < randomValue then
		-- Since the random value is higher than the Item's drop probability, it considers the drop to be a failure and returns nil.
		return
	end
end

-- The target item is always returned if a MustDropItemKey is set.
if not _UtilLogic:IsNilorEmptyString(MustDropItemKey) then
	-- Requests data from DataTablelogic based on MustDropItemKey.
	local findData = _DataTableLogic:GetItemDataByItemID(MustDropItemKey)
	-- Creates an item if there is data that has a MustDropItemKey value.
	if isvalid(findData) and next(findData) then
		self:SendSpawnItem(findData.ItemID, SpawnPosition)
		return
	else
		-- Leaves an error log saying that the item couldn't be found if there are no such items.
		log("Couldn't find an item that is MustDropItemKey!")
		return
	end
end

-- Executes random, so a random item is provided if MustDropItemKey is blank.
local getDropData = _DataTableLogic:GetItemDropData()
local totalValue = 0
-- Attempts to calculate the probability after adding all of the probabilities in the item drop table.
for _, data in pairs(getDropData) do
	totalValue = totalValue + data.DropRate
end
-- Returns an item that has greater timing than the randomized value by determining the value between 0 and totalValue and then adding DropRate from 1 as the winner.
local randomValue = _UtilLogic:RandomIntegerRange(0, totalValue)
local resultValue = 0 
for _, result in pairs(getDropData) do
	-- Adds DropRate to resultValue.
	resultValue = resultValue + result.DropRate
	-- If the resultValue value is higher than the determined randomValue, it is processed as the winner.
	if resultValue >= randomValue then
		-- Searches the ItemData from DataTablelogic with ItemID.
		local findData = _DataTableLogic:GetItemDataByItemID(result.ItemID)
		-- Returns the data that was searched if the data is normally loaded.
		if isvalid(findData) and next(findData) then
			self:SendSpawnItem(findData.ItemID, SpawnPosition)
			return
		else
		-- Leaves a log and returns nil if the data that was found is nil or is data that can't be used.
			log("Couldn't find a value with "..result.ItemID.." as the name from ItemData!")
			return
		end
	end
end
end

method void SendSpawnItem(string ItemID, Vector3 SpawnPosition)
-- Generates an error log if no ItemSpawnManager is found InGame.
if not isvalid(self.itemSpawnManager) then
	log("The ItemHelper's itemSpawnManager is nil!")
	return
end
-- Directs the InGame's ItemSpawnManager to create an item.
self.itemSpawnManager:CreateItem(ItemID, SpawnPosition)
end

method void InitData()
-- Searches for the InGame ItemSpawnManager if the InGame ItemSpawnManager couldn't be found.
if not isvalid(self.itemSpawnManager) then
	local findEntity = _EntityService:GetEntityByPath("/maps/InGame/ItemSpawnManager")
	-- Generates an error log if the ItemSpawnManager couldn't be found.
	if not isvalid(findEntity.ItemSpawnManager) then
		log("Either the ItemSpawnManager entity or the ItemSpawnManager component doesn't exist in the InGame map!")
		return
	end
	-- Registers the found ItemSpawnManager to property.
	self.itemSpawnManager = findEntity.ItemSpawnManager
end
end

method void ReleaseData()
-- Accesses the ItemSpawnManager from the Title screen and disables it, so it can't be used.
self.itemSpawnManager = nil
end

end
```
