## Creating a UIInventoryController
- Create a `UIInventoryController`, a Component that manages elements in `InventoryUI`.
- Register the created Component to `InventoryUI` with `AddComponent`.

```lua
@Component
script UIInventoryController extends Component

property Entity infoEntity = "21856f11-da9c-4d39-869f-38679e45396a"

property Entity slotLayoutEntity = "413114e4-bc44-4702-8c59-04c097ccab27"

property table itemSlotTable = {}

property SpriteGUIRendererComponent infoFrameItemIcon = "a114218d-e9af-4529-a361-c40ff4b026a0"

property TextGUIRendererComponent infoFrameName = "cfae0263-8821-4d04-9d51-7cbf70f1f963"

property TextGUIRendererComponent infoFrameDesc = "651d2c82-7c70-4f17-bf92-43ab69b4fa5d"

property string itemSlotModelID = "model://b99d554b-893e-4e9b-9afd-fd91ad4b9805"

property number maxSlotCount = 8

@ExecSpace("ClientOnly")
method void OnBeginPlay()
-- Verifies whether the InfoFrame that outputs the item's information is usable.
if not isvalid(self.infoEntity) then
	-- Generates an error log and halts code execution if the InfoFrame that outputs the item's information is nil.
	log_warning("There is no registered InfoFrame to show the information on InventoryUI!")
	return
end
-- Deactivates infoEntity if usable.
self.infoEntity.Enable = false

-- Creates an inventory slot if there is less than 1 slot in the inventory.
if #(self.itemSlotTable) < 1 then
	for i = 1, self.maxSlotCount do
		-- Creates 1 empty inventory slot.
		local slotEntity = _SpawnService:SpawnByModelId(self.itemSlotModelID, tostring("ItemSlot_"..i), Vector3.zero, self.slotLayoutEntity)
		-- Attempts to reset to the initial state if an entity was created normally.
		if isvalid(slotEntity) then
			slotEntity.InventorySlotComponent:SetDefault()
			--Registers created entity to itemSlotTable.
			self.itemSlotTable[i] = slotEntity.InventorySlotComponent
		end
	end
end

-- Deactivates the inventory UI to stop the inventory UI from being output when the game starts.
self.Entity.Visible = false
self.Entity.Enable = false
end

@ExecSpace("ClientOnly")
method void ShowInventory()
-- Changes its own state to visible.
self.Entity.Visible = true
self.Entity.Enable = true
end

@ExecSpace("ClientOnly")
method void HideInventory()
-- Changes its own state to hidden.
self.infoEntity.Enable = false
self.Entity.Visible = false
self.Entity.Enable = false
end

@ExecSpace("ClientOnly")
method void UpdateItemSlot(OnPlayerLootItem event)
-- Attempts to update the inventory when the player obtains an item.
for i = 1, #(self.itemSlotTable) do
	-- Searches whether the same item exists in the existing inventory slot.
	local slotItemName = self.itemSlotTable[i]:ReturnItemID()
	-- Checks if the fetched ItemName is nil.
	if not _UtilLogic:IsNilorEmptyString(slotItemName) then
		-- Increases the received itemID's Count if there is an ItemName.
		if slotItemName == event.ItemID then
			self.itemSlotTable[i]:UpdateItemCount(event.ItemCount)
			return
		end
	end
end
-- Searches for and updates an empty slot, since there is no ItemSlot of the same name after navigating one full lap.
for i = 1, #(self.itemSlotTable) do
	-- Loads the ItemID value registered to index i.
	local slotItemName = self.itemSlotTable[i]:ReturnItemID()
	-- Registers if the fetched value is nil, since the slot is empty.
	if _UtilLogic:IsNilorEmptyString(slotItemName) then
		-- Loads the DataSet value from ItemLogic based on ItemID value.
		local getTableItemData = _ItemLogic:ReturnItemData(event.ItemID)
		-- Creates a value to be delivered in an ItemType format.
		local inputItemData = ItemType()
		-- Registers the ItemID value used to classify items and apply values to inventory slots.
		inputItemData.ItemID = event.ItemID
		-- Registers the ItemImage value used to output the item icon.
		inputItemData.ItemImage = getTableItemData.ItemImage
		-- Registers the ItemName value used to output the item name.
		inputItemData.ItemNameID = getTableItemData.ItemNameID
		-- Registers the ItemTypeValue value used to designate the item's type.
		inputItemData.ItemTypeValue = getTableItemData.ItemTypeValue
		-- Registers the ItemDescID value used to output the item description.
		inputItemData.ItemDescID = getTableItemData.ItemDescID
		-- Registers the saved data to the slot.
		self.itemSlotTable[i]:InitItemData(getTableItemData, event.ItemCount)
		self.itemSlotTable[i]:UpdateItemCount(event.ItemCount)
		return
	end
end
end

@ExecSpace("ClientOnly")
method void ShowItemInfo(ShowInventoryItemInfoEvent ItemData)
-- Activates the item info window.
self.infoEntity.Enable = true
self.infoEntity.Visible = true
-- Outputs the item image in the item information window.
self.infoFrameItemIcon.ImageRUID = ItemData.ItemImage
-- Outputs the item information window's name value that was translated through Localize.
self.infoFrameName.Text = _LocalizationService:GetText(ItemData.ItemNameID)
-- Outputs the item information window's description value that was translated through Localize.
self.infoFrameDesc.Text = _LocalizationService:GetText(ItemData.ItemDescID)
end

@ExecSpace("ClientOnly")
method void UpdateLoadData()

end

@EventSender("Entity", "cfbc60a4-3eb4-4d8b-968f-bb02558514c1")
handler HandleButtonClickEvent(ButtonClickEvent event)
-- Deactivates the inventory UI.
self:HideInventory()
end

@EventSender("LocalPlayer")
handler HandleShowInventoryEvent(ShowInventoryUIEvent event)
self:ShowInventory()
end

@EventSender("LocalPlayer")
handler HandleOnPlayerLootItem(OnPlayerLootItem event)
self:UpdateItemSlot(event)
end

@EventSender("Self")
handler HandleShowInventoryItemInfoEvent(ShowInventoryItemInfoEvent event)
self:ShowItemInfo(event)
end

end
```

> This code was written based on Plain Text.

- Write the above code to `UIInventoryController`.
- Property is composed of the following elements.
- Registers the entities for each element.
  - **infoEntity**: The Property needed to register the `InfoFrame` entity for `InventoryUI`.
  - **slotLayoutEntity**: The Property needed to register the `SlotLayout` entity for the `InventoryFrame` entity.
  - **infoFrameItemIcon**: Outputs the item icon to the item information window by registering the `ItemIconFrame` and `ItemIcon` for the `InfoFrame`.
  - **infoFrameName**: Outputs the item name in the item information window by registering the `NameFrame` and `NameText` for the `InfoFrame`.
  - **infoFrameDesc**: Outputs the item information in the item information window by registering the `DescriptionFrame` and `DescriptionText` for the `InfoFrame`.
  - **itemSlotModelID**: The Property needed to register the previously-created `InventoryItemSlot` Entry ID.
    - Right-click Model from the `InventoryItemSlot` that was created in `Workspace` and `MyDesk` and click `Copy Entry ID` to register the copied value.
  - **maxSlotCount**: Registers the number of inventory slots that will be created. A value of 8 has been input to create 8 slots in the sample world.

## Creating InventorySlotComponent
- The Component that manages `IventoryItemSlot` that's used as slots in the inventory.
- This Component registers the item information and updates the UI upon receiving item information.
- The function is composed to output the item information window when it is clicked.

```lua
@Component
script InventorySlotComponent extends Component

property SpriteGUIRendererComponent spriteComponent = "09988a83-db6c-4895-a2a8-f13daae52688"

property ItemType itemData = ItemType()

property number itemCount = 0

@ExecSpace("ClientOnly")
method void OnBeginPlay()
-- Resets to Default state when the game starts.
self:SetDefault()
end

@ExecSpace("ClientOnly")
method void SetDefault()
-- Searches for Icon from the target entity's child entity if spriteComponent is empty.
if self.spriteComponent == nil then
	local findIcon = self.Entity:GetChildByName("Icon")
	if isvalid(findIcon.SpriteGUIRendererComponent) then
		self.spriteComponent = findIcon.SpriteGUIRendererComponent
	end
end
-- Converts the Visible state of the entity that outputs the icon to false, so only an empty inventory is visible.
self.spriteComponent.Entity.Visible = false
-- Resets the value of itemData that holds item information to an empty state.
self.itemData = ItemType()
end

@ExecSpace("ClientOnly")
method string ReturnItemID()
-- Returns the ItemID of the target slot.
return self.itemData.ItemID
end

@ExecSpace("ClientOnly")
method void InitItemData(ItemType ItemData)
-- Verifies whether the data received can be used normally.
if not isvalid(ItemData) or ItemData.ItemID == "" or ItemData.ItemID == nil then
	-- Generates an error log if the received data is unusable.
	log_error("The received item data is nil or unusable!")
	return
end
-- Loads data based on the data received, since the received data is usable.
self.itemData = ItemData
local getData = _ItemLogic:ReturnItemData(self.itemData.ItemID)
-- Enters the image RUID from the found DataSet and outputs the item image.
self.spriteComponent.ImageRUID = getData.ItemImage
self.spriteComponent.Entity.Visible = true
self.spriteComponent.Entity.Enable = true
end

@ExecSpace("ClientOnly")
method void UpdateItemCount(number Count)
-- Increases the current Count by the amount received.
self.itemCount = self.itemCount + Count
end

@EventSender("Self")
handler HandleButtonClickEvent(ButtonClickEvent event)
-- The item info window doesn't pop up when there is no item ID registered to the slot.
if _UtilLogic:IsNilorEmptyString(self.itemData.ItemID) then
	return
-- Calls an Event to output the item info window if there is an item ID in the slot.
else
	-- Enters all of the event's Property values for delivery.
	local itemData = ShowInventoryItemInfoEvent()
	itemData.ItemNameID = self.itemData.ItemNameID
	itemData.ItemImage = self.itemData.ItemImage
	itemData.ItemDescID = self.itemData.ItemDescID
	-- Searches for InventoryUI and delivers the event to the relevant UIGroup.
	local inventoryUI = _EntityService:GetEntityByPath("/ui/InventoryUI")
	inventoryUI:SendEvent(itemData)
end
end

end
```
> This code is written based on Plain Text.

- Write the above code to InventorySlotComponent.
- The Methods fulfill the following roles.
- **OnBeginPlay**: The Method that executes SetDefault the moment the world is launched.
- **SetDefault**: The Method used to set an inventory slot to its initial state.
- **ReturnItemID**: The Method that returns the ItemID that's held by the relevant inventory slot.
- **InitItemData**: Updates the slot's UI based on the received ItemData.
- **UpdateItemCount**: Increases `itemCount` by the amount of received quantity count.
- **HandleButtonClickEvent**: Searches for `InventoryUI` and delivers the item information that's registered to itself to call the Event to output the item information window when it is clicked.

## Creating ItemLogic
- `ItemLogic` is a Logic used to manage the `ItemData` DataSet.
- Upon receiving an `ItemID` value based on `ItemData`, executes a function that returns the relevant value's information.
- Write the code below after creating `ItemLogic`.
```lua
@Logic
script ItemLogic extends Logic

property table itemData = {}

@ExecSpace("ClientOnly")
method void OnBeginPlay()
-- Calls a method when the game starts to load the ItemData DataSet.
self:InitItemData()
end

@ExecSpace("ClientOnly")
method void InitItemData()
-- Determines whether there is any nil or missing data for itemData.
if not isvalid(self.itemData) or not next(self.itemData) then
	-- Attempts a reset if the itemData is nil or if there is no data.
	local findData = _DataService:GetTable("ItemData")
	-- Generates an error log and halts code execution if findData is nil.
	if not isvalid(findData) then
		log_error("ItemData DataSet not found!")
		return
	end
	-- Registers data if the DataSet is found normally.
	-- Loads the number of ItemData rows.
	local rowCount = findData:GetRowCount()
	-- Navigates rows equal to the number of ItemData rows and registers data.
	for i = 1, rowCount do
		-- Loads the ItemID value from index i.
		local itemKey = tostring(findData:GetCell(i, "ItemID"))
		-- Generates an error log and halts code execution if the ItemID received is nil or "".
		if not isvalid(itemKey) or itemKey == "" then
			log_error("The Item ID value for the ItemData DataSet in index i is nil or blank!")
			break
		end
		-- Attempts to register if there is a normal value.
		local dataValue = ItemType()
		-- Registers the item info's Index as value i.
		dataValue.Index = i
		-- Registers the item info's ItemID as the previously-found Key value.
		dataValue.ItemID = itemKey
		-- Searches for and registers the item info's ItemTypeValue from the Cell in row i.
		dataValue.ItemTypeValue = tostring(findData:GetCell(i, "ItemTypeValue"))
		-- Searches for and registers the item info's ItemNameID from the Cell in row i.
		dataValue.ItemNameID = tostring(findData:GetCell(i, "ItemNameID"))
		-- Searches for and registers the item info's ItemDescID from the Cell in row i.
		dataValue.ItemDescID = tostring(findData:GetCell(i, "ItemDescID"))
		-- Searches for and registers the item info's ItemImage from the Cell in row i.
		dataValue.ItemImage = tostring(findData:GetCell(i, "ItemImage"))
		-- Registers the entered data.
		self.itemData[itemKey] = dataValue
	end
	-- Generates a log if the registration is completed normally.
	log("Registration to ItemData ItemLogic complete!")
	return
end
-- The relevant content is delivered as a log, since item parsing wasn't attempted.
log("ItemData not registered to ItemLogic. Either parsing is already complete or it failed to read DataSet normally!")
end

@ExecSpace("ClientOnly")
method table ReturnItemData(string ItemID)
-- Returns the relevant value if there is itemData that has ItemID as a Key.
if isvalid(self.itemData[ItemID]) or next(self.itemData[ItemID]) then
	return self.itemData[ItemID]
else
	-- Generates an error log if there is no itemData that has ItemID as a Key.
	log("There is no ItemData that has "..ItemID.." as a Key!")
	return nil
end
end

end
```
