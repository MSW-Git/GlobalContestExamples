## 영상 타임라인
- 강의 영상내 아래의 타임라인에서 학습할 수 있습니다.
- 퀘스트 시스템 구현하기: `01:12:04`

<br>

## 학습 목표
- 플레이어의 행동을 유도하는 퀘스트 시스템을 제작하는 방법에 대해 학습합니다.
- 샘플 월드가 제공하는 방식을 학습하고, 게임을 제작할 때 어떤 요소를 추가하여 변경할 수 있는지 학습합니다.
- 퀘스트 시스템을 이해하고, 퀘스트를 유기적으로 연결하여 게임의 흐름을 제어할 수 있습니다.
- 퀘스트에 따른 보상을 지급하여 플레이어가 퀘스트에 대한 동기를 부여할 수 있도록 보상 시스템을 제작하는 방법을 학습합니다.

<br>

## 해당 영상 시청 후 수행해 볼 내용
- 퀘스트 시스템을 구현하여 조건을 충족하면 다음 퀘스트로 연계되는 기능을 제작합니다.
- 플레이어가 퀘스트를 클리어 할 경우 아이템을 지급하며, 인벤토리가 업데이트되도록 기능을 구현합니다.
- 퀘스트에 대한 정보를 출력하는 UI의 구성을 변경하여 직관적인 UI로 구성합니다.

## 퀘스트 시스템이란?

<p align="center">
<img src="https://develrocket-bucket.s3.ap-northeast-2.amazonaws.com/learning/MSW-100/Adventure/MSW-100_Adventure_6_001.jpg" alt="MSW-100_Adventure_6_001.jpg">
</p>

> 메이플스토리의 퀘스트 UI입니다. 플레이어가 수행해야 하는 행동과 보상이 출력됩니다.

- 퀘스트는 플레이어의 행동을 제약하거나 유도하는 수단입니다.
- 플레이어가 어떤 행위를 해야 하는지, 이를 통해 보상을 얻을 수 있다는 학습을 수행시킵니다.
- 플레이어는 학습된 경험을 기반으로 퀘스트를 수행하면 보상이 있고, 이는 캐릭터의 성장을 통해 플레이의 난이도가 완화될 수 있다는 행동 사이클을 생성합니다.
- 따라서 정교하게 짜여진 퀘스트일수록 플레이어는 퀘스트에 몰입하게 되며 개발자는 플레이어를 원하는 방향으로 게임의 흐름을 유도할 수 있습니다.

<p align="center">
<img src="https://develrocket-bucket.s3.ap-northeast-2.amazonaws.com/learning/MSW-100/Adventure/MSW-100_Adventure_6_002.jpg" alt="MSW-100_Adventure_6_002.jpg">
</p>

> 젤다의 전설 야생의 숨결은 플레이어가 어떤 행동을 하면 그에 따른 보상이 주어집니다. 강제성은 없지만 플레이어는 이를 학습하여 해당 행위를 반복하게 됩니다.

- 시각적으로는 보이지 않지만 퀘스트 시스템을 활용하여 게임의 사이클을 만들 수 있습니다.
- `젤다의 전설: 야생의 숨결`에서는 코로그 열매를 얻기 위해 퍼즐을 풀거나 특정 위치에 도착해야 합니다. 코로그 열매를 통해 인벤토리의 슬롯을 늘릴 수 있습니다.
- 이런 행동 역시 하나의 퀘스트로 작용되며, 플레이어는 인벤토리 슬롯을 늘리기 위해 코로그를 찾는 행동, 퍼즐을 해결하는 행동을 반복하게 됩니다.

## QuestLogic 제작하기
- 앞서 제작한 `QuestData` DataSet을 관리하는 Logic이 필요합니다.
- QuestLogic은 `QuestData`를 캐싱한 뒤, QuestID값에 따라 필요한 형태의 데이터를 전달합니다.
- `Workspace`, `MyDesk`에서 Logic을 생성한 뒤 생성한 Logic의 이름을 `QuestLogic`으로 설정합니다.
```lua
@Logic
script QuestLogic extends Logic

property table questData = {}

@ExecSpace("ClientOnly")
method void OnBeginPlay()
-- QuestData DataSet을 가져오기 위해 초기화를 시도합니다.
self:InitQuestData()
end

@ExecSpace("ClientOnly")
method void InitQuestData()
-- questData가 비어있는 상태인지 체크합니다.
if not isvalid(self.questData) or not next(self.questData) then
	-- 비어있는 상태일 경우, Workspace에서 QuestData를 가져옵니다.
	local getData = _DataService:GetTable("QuestData")
	-- 만약 getData가 nil일 경우, DataSet을 찾지 못했으므로 로그를 남기고 코드 실행을 중단합니다.
	if not isvalid(getData) then
		log_error("QuestData를 찾지 못했습니다!")
		return
	end
	-- QuestData의 RowCount만큼 반복하여 DataSet내의 값들을 모두 탐색합니다.
	for i = 1, getData:GetRowCount() do
		local findQuestID = getData:GetCell(i, "QuestID")
		-- 만약 찾은 QuestID가 Nil일 경우, DataSet에 문제가 있다고 판단하고 탈출합니다.
		if _UtilLogic:IsNilorEmptyString(findQuestID) then
			log_error("QuestData내 QuestID를 찾을 수 없습니다!")
			return			
		end
		-- 이외에는 정상적으로 값이 있다고 판단하여 값 입력을 시작합니다.
		local rowData = QuestDataType()
		rowData.QuestID = findQuestID
		rowData.QuestNameID = getData:GetCell(i, "QuestNameID")
		rowData.InfoTextID = getData:GetCell(i, "InfoTextID")
		rowData.ConditionType = getData:GetCell(i, "ConditionType")
		rowData.TargetID = getData:GetCell(i, "TargetID")
		rowData.TargetCount = tonumber(getData:GetCell(i, "TargetCount"))
		rowData.NeedDialogue = getData:GetCell(i, "NeedDialogue")
		rowData.RewardID = getData:GetCell(i, "RewardID")
		rowData.NextQuestID = getData:GetCell(i, "NextQuestID")
		-- 모든 값들을 가져왔다면, questData에 해당 값을 저장합니다.
		self.questData[findQuestID] = rowData
	end
	log("QuestData 파싱 완료!")
end
end

@ExecSpace("ClientOnly")
method void SetQuestNewGame()
-- 로컬 플레이어를 찾습니다.
local localPlayer = _UserService.LocalPlayer
-- 정상적으로  찾았는지 확인합니다.
if not isvalid(localPlayer) then
	--만약 playerQuestController가 사용 불가능하다면 에러 로그를 남기고 코드 실행을 중단합니다.
	log_error("localPlayer를 찾을 수 없습니다!")
	return
end
-- questData를 정상적으로 사용할 수 있는지 점검합니다.
if not isvalid(self.questData) or not next(self.questData) then
	-- 만약 정상 사용이 불가능할 경우 초기화를 시도합니다.
	self:InitQuestData()
end
-- DataSet에서 가장 첫 번째에 있는 QuestID를 찾습니다.
local firstQuestID = _DataService:GetTable("QuestData"):GetCell(1, "QuestID")
-- 해당 ID값을 기준으로, questData에서 데이터를 가져옵니다.
local findData = self.questData[firstQuestID]
-- 초기화 후 전달을 위해 Event를 제작합니다.
local sendEvent = OnUpdatePlayerQuestData()
sendEvent.QuestID = findData.QuestID
sendEvent.QuestNameID = findData.QuestNameID
sendEvent.InfoTextID = findData.InfoTextID
sendEvent.ConditionType = findData.ConditionType
sendEvent.TargetID = findData.TargetID
sendEvent.NeedDialogue = findData.NeedDialogue
sendEvent.RewardID = findData.RewardID
sendEvent.NextQuestID = findData.NextQuestID
-- 완성된 Event를 localPlayer에게 전달합니다.
localPlayer:SendEvent(sendEvent)
end

@ExecSpace("ClientOnly")
method void UpdateNextQuest(string NextQuestID)
-- LocalPlayer를 찾습니다.
local localPlayer = _UserService.LocalPlayer
-- 정상적으로  찾았는지 확인합니다.
if not isvalid(localPlayer) then
	--만약 playerQuestController가 사용 불가능하다면 에러 로그를 남기고 코드 실행을 중단합니다.
	log_error("localPlayer를 찾을 수 없습니다!")
	return
end
-- 다음 퀘스트의 정보를 찾습니다.
local findData = self.questData[NextQuestID]
local sendEvent = OnUpdatePlayerQuestData()
sendEvent.QuestID = findData.QuestID
sendEvent.QuestNameID = findData.QuestNameID
sendEvent.InfoTextID = findData.InfoTextID
sendEvent.ConditionType = findData.ConditionType
sendEvent.TargetID = findData.TargetID
sendEvent.NeedDialogue = findData.NeedDialogue
sendEvent.RewardID = findData.RewardID
sendEvent.NextQuestID = findData.NextQuestID
-- LocalPlayer에게 이벤트를 전달합니다.
localPlayer:SendEvent(sendEvent)
end

@ExecSpace("ClientOnly")
method void SetQuestDataByQuestID(string QuestID)
-- 로컬 플레이어를 찾습니다.
local localPlayer = _UserService.LocalPlayer
-- 정상적으로  찾았는지 확인합니다.
if not isvalid(localPlayer) then
	--만약 playerQuestController가 사용 불가능하다면 에러 로그를 남기고 코드 실행을 중단합니다.
	log_error("localPlayer를 찾을 수 없습니다!")
	return
end
-- questData를 정상적으로 사용할 수 있는지 점검합니다.
if not isvalid(self.questData) or not next(self.questData) then
	-- 만약 정상 사용이 불가능할 경우 초기화를 시도합니다.
	self:InitQuestData()
end
-- 해당 ID값을 기준으로, questData에서 데이터를 가져옵니다.
local findData = self.questData[QuestID]
-- 초기화 후 전달을 위해 Event를 제작합니다.
local sendEvent = OnUpdatePlayerQuestData()
sendEvent.QuestID = findData.QuestID
sendEvent.QuestNameID = findData.QuestNameID
sendEvent.InfoTextID = findData.InfoTextID
sendEvent.ConditionType = findData.ConditionType
sendEvent.TargetID = findData.TargetID
sendEvent.NeedDialogue = findData.NeedDialogue
sendEvent.RewardID = findData.RewardID
sendEvent.NextQuestID = findData.NextQuestID
-- 완성된 Event를 localPlayer에게 전달합니다.
localPlayer:SendEvent(sendEvent)

end
```

> 다음은 Plain Text를 기준으로 작성된 코드입니다.

- 위의 코드를 `QuestLogic`에 작성합니다.
- 각 Method의 역할은 다음과 같습니다.
    - `OnBeginPlay`: 월드가 실행되면 `InitQuestData` Method를 실행합니다.
    - `InitQuestData`: 제작한 `QuestData`를 찾은 뒤 `questData` Property에 등록하는 기능을 수행합니다.
    - `SetQuestNewGame`: 게임을 최초로 실행했을 경우, `QuestData`의 최상단에 있는 퀘스트를 플레이어가 수행해야 하는 퀘스트로 등록하는 기능을 수행합니다.
    - `UpdateNextQuest`: 전달받은 NextQuestID값을 기준으로 동일한 QuestID값으로 플레이어가 수행해야 하는 퀘스트로 등록하는 기능을 수행합니다.
    - `SetQuestDataByQuestID`: 이어하기를 통해 시작한 플레이어의 경우, 기존 데이터를 가져온 뒤, 해당 Quest정보로 플레이어가 수행해야 하는 퀘스트로 등록하는 기능을 수행합니다.

## PlayerQuestComponent 제작하기
- 플레이어가 퀘스트 정보를 가지기 위한 Component입니다.
- `Workspace`에서 `MyDesk`에 `PlayerQuestComponent`를 제작한 뒤 `Workspace`의 `DefaultPlayer`에 추가합니다.

```lua
@Component
script PlayerQuestComponent extends Component

property QuestDataType currentQuestData = QuestDataType()

@EventSender("Self")
handler HandleOnUpdatePlayerQuestData(OnUpdatePlayerQuestData event)
-- 전달받은 퀘스트 데이터로 currentQuestData를 업데이트합니다.
self.currentQuestData.QuestID = event.QuestID
self.currentQuestData.QuestNameID = event.QuestNameID
self.currentQuestData.InfoTextID = event.InfoTextID
self.currentQuestData.ConditionType = event.ConditionType
self.currentQuestData.TargetID = event.TargetID
self.currentQuestData.NeedDialogue = event.NeedDialogue
self.currentQuestData.RewardID = event.RewardID
self.currentQuestData.NextQuestID = event.NextQuestID
end

@EventSender("Self")
handler HandleCheckQuestClearEvent(CheckQuestClearEvent event)
-- 전달받은 event의 값들이 현재 퀘스트의 클리어 조건을 충족하는지 체크합니다.
if event.ConditionType == self.currentQuestData.ConditionType and event.TargetID == self.currentQuestData.TargetID then
	-- 조건을 충족했을 경우, 클리어에 따른 처리를 진행합니다.
	if not (_UtilLogic:IsNilorEmptyString(self.currentQuestData.NeedDialogue)) then
		-- 대화창이 출력되야 하므로 DialogueUI를 찾습니다.
		local dialogueUIGroup = _EntityService:GetEntityByPath("/ui/DialogueUI")
		-- Event에 출력해야 하는 DialogueID를 입력합니다.
		local eventValue = ShowDialogueUIEvent()
		eventValue.DialogueID = self.currentQuestData.NeedDialogue
		-- DialogueUI에 Event를 전달합니다.
		dialogueUIGroup:SendEvent(eventValue)
	end
	-- 지급해야 하는 보상이 있을 경우, 해당 보상을 지급합니다.
	if not _UtilLogic:IsNilorEmptyString(self.currentQuestData.RewardID) then
		local lootItemEvent = OnPlayerLootItem()
		lootItemEvent.ItemID = self.currentQuestData.RewardID
		lootItemEvent.ItemCount = 1
		self.Entity:SendEvent(lootItemEvent)
	end
	-- 출력해야 하는 Dialogue가 있을 경우, Dialogue를 출력합니다.
	if not _UtilLogic:IsNilorEmptyString(self.currentQuestData.NeedDialogue) then
		local findDialogueUI = _EntityService:GetEntityByPath("/ui/DialogueUI")
		local showDialogueEvent = ShowDialogueUIEvent()
		showDialogueEvent.DialogueID = self.currentQuestData.NeedDialogue
		findDialogueUI:SendEvent(showDialogueEvent)
	end
	
	-- 연계 퀘스트가 있을 경우, Quest 업데이트를 요청합니다.
	if not _UtilLogic:IsNilorEmptyString(self.currentQuestData.NextQuestID) then
		_QuestLogic:UpdateNextQuest(self.currentQuestData.NextQuestID)
	end
end
end

end
```

> 해당 코드는 Plain Text를 기준으로 작성된 코드입니다.

- 작성된 코드들은 다음의 기능을 수행합니다.
    - `HandleOnUpdatePlayerQuestData`: 전달받은 event의 값으로 `currentQuestData` Property를 초기화합니다.
    - `HandleCheckQuestClearEvent`: 전달받은 event를 기준으로 클리어 조건을 점검합니다. 만약 조건을 충족했을 경우 다음의 기능들을 수행합니다.
        - 만약 Dialogue를 출력해야 할 경우, 전달받은 `DialogueID`를 출력하도록 요청합니다.
        - 지급해야 하는 보상이 있을 경우, `RewardID`값을 아이템으로 지급합니다.
        - 연계 퀘스트가 있을 경우, 퀘스트 업데이트를 요청합니다.

## PlayerItemComponent 제작하기
- 플레이어가 아이템을 관리할 수 있도록 하는 Component입니다.
- 플레이어가 아이템을 습득할 경우, `OnPlayerLootItem`을 수신받아 아이템을 등록하는 기능을 수행합니다.
- `Workspace`에서 `MyDesk`에 Component를 제작합니다.
- 제작한 Component의 이름을 `PlayerItemComponent`로 등록합니다.
- 제작한 Component를 `Workspace`의 `DefaultPlayer`에 추가합니다.

```lua
@Component
script PlayerItemComponent extends Component

property table playerInventory = {}

@ExecSpace("ClientOnly")
@EventSender("Service", "InputService")
handler HandleKeyDownEvent(KeyDownEvent event)
-- i키를 누르는 이벤트를 감지합니다.
if event.key == KeyboardKey.I then
	-- i키를 눌렀다면, 인벤토리 UI를 엽니다.
	self.Entity:SendEvent(ShowInventoryUIEvent())
end
end

@EventSender("Self")
handler HandleOnPlayerLootItem(OnPlayerLootItem event)
log("아이템 획득 수신 완료!")
local checkClearQuestEvent = CheckQuestClearEvent()
checkClearQuestEvent.ConditionType = "Item"
checkClearQuestEvent.TargetID = event.ItemID
self.Entity:SendEvent(checkClearQuestEvent)
-- 1. 먼저 인벤토리에 같은 아이템이 있는지 확인합니다.
for i = 1, #self.playerInventory do
    if self.playerInventory[i] and self.playerInventory[i].ItemID == event.ItemID then
        self.playerInventory[i].ItemCount = self.playerInventory[i].ItemCount + event.ItemCount
        return -- 아이템을 찾아서 더했으므로 함수를 종료합니다.
    end
end

-- 2. 같은 아이템이 없다면, 첫 번째로 비어있는 슬롯(nil)을 찾아 추가합니다.
-- 최대 8번 슬롯까지만 체크 (itemIndex < 9 조건 반영)
for i = 1, 8 do
    if self.playerInventory[i] == nil then
        self.playerInventory[i] = {ItemID = event.ItemID, ItemCount = event.ItemCount}
        return -- 아이템을 추가했으므로 함수를 종료합니다.
    end
end
end

end
```

> 해당 코드는 Plain Text를 기준으로 작성된 코드입니다.

- `HandleKeyDownEvent`: 플레이어가 키보드의 I 키를 누를 경우 인벤토리 UI를 출력하도록 요청하는 Method입니다.
- `HandleOnPlayerLootItem`: 전달받은 event의 값을 기준으로 플레이어의 인벤토리에 추가하고 업데이트 하는 역할을 수행합니다.
